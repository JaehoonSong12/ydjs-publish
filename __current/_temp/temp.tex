
  \item Let $p$, $q$, and $r$ be propositions.  Determine whether the 
  following pairs of compound propositions are logically equivalent.
  Justify your answer \textbf{without using a full truth table}.
  \begin{enumerate}
    \item $(p \THEN q) \AND (p \THEN r)$ and $p \THEN (q \AND r)$.
    \begin{subprove}
      Let \(p,q,r\) be propositions. 
      \[
        (p\THEN q)\AND(p\THEN r) \equiv (\NOT p\OR q)\AND(\NOT p\OR r)
      \]
      by applying the \textbf{Implication Equivalence} \(p\THEN q\equiv~\NOT p\OR q\) to both implications.
      Moreover, by \textbf{Distributivit Laws} in the form \((p\OR q)\AND(p\OR r)\equiv p\OR(q\AND r)\), 
      \[
        (\NOT p\OR q)\AND(\NOT p\OR r) \equiv \NOT p\OR(q\AND r).
      \]
      Lastly, applying the \textbf{Implication equivalence} in reverse,
      \[
        \NOT p\OR(q\AND r)\equiv~p\THEN(q\AND r).
      \]
      Thus, \((p\THEN q)\AND(p\THEN r)\equiv p\THEN(q\AND r)\), so the two formulas are logically equivalent. 
    \end{subprove}

    \item $(p \THEN r) \AND (q \THEN r)$ and $(p \OR q) \THEN r$.
    \begin{subprove}
      Let \(p,q,r\) be propositions. 
      We begin by applying the \textbf{Implication Equivalence} ($A \THEN B \equiv \NOT A \OR B$) to both implications on the left-hand side:
      \[
      (p \THEN r) \AND (q \THEN r) \equiv (\NOT p \OR r) \AND (\NOT q \OR r).
      \]
      By the \textbf{Distributive Law} in the form $(A \OR B) \AND (A \OR C) \equiv A \OR (B \AND C)$, we have
      \[
      (\NOT p \OR r) \AND (\NOT q \OR r) \equiv (\NOT p \AND \NOT q) \OR r.
      \]
      Now we consider the right-hand side. By applying the \textbf{Implication Equivalence}, we get
      \[
      (p \OR q) \THEN r \equiv \NOT(p \OR q) \OR r.
      \]
      Lastly, by \textbf{De Morgan's Law} ($\NOT(A \OR B) \equiv \NOT A \AND \NOT B$), we have
      \[
      \NOT(p \OR q) \OR r \equiv (\NOT p \AND \NOT q) \OR r.
      \]
      Since both expressions simplify to $(\NOT p \AND \NOT q) \OR r$, the two formulas are logically equivalent. $\blacksquare$
    \end{subprove}

    \item $(p \THEN q) \THEN r$ and $p \THEN (q \THEN r)$.
    \begin{subprove}
      \textbf{They are NOT logically equivalent.}
      We simplify the left-hand side expression first by applying the \textbf{Implication Equivalence} ($A \THEN B \equiv \NOT A \OR B$):
      \[
      (p \THEN q) \THEN r \equiv \NOT(p \THEN q) \OR r \equiv \NOT(\NOT p \OR q) \OR r.
      \]
      Next, by applying \textbf{De Morgan's Law} ($\NOT(\NOT A \OR B) \equiv A \AND \NOT B$), we obtain
      \[
      \NOT(\NOT p \OR q) \OR r \equiv (p \AND \NOT q) \OR r.
      \]
      Now we consider the right-hand side. By applying the \textbf{Implication Equivalence} twice, we get
      \[
      p \THEN (q \THEN r) \equiv \NOT p \OR (\NOT q \OR r).
      \]
      Since $(p \AND \NOT q) \OR r$ is not logically equivalent to $\NOT p \OR \NOT q \OR r$, the two expressions are not equivalent.
      For instance, if we take $p=\FALSE$, $q=\TRUE$, and $r=\FALSE$, the left-hand expression is
      \[
      (\FALSE \THEN \TRUE) \THEN \FALSE \equiv \TRUE \THEN \FALSE \equiv \FALSE,
      \]
      while the right-hand expression is
      \[
      \FALSE \THEN (\TRUE \THEN \FALSE) \equiv \FALSE \THEN \FALSE \equiv \TRUE.
      \]
      As the truth values differ for the same assignment, the two compound propositions are not logically equivalent. $\blacksquare$
    \end{subprove}

    \item $(p \OR q) \THEN r$ and $(p \THEN r) \OR (q \THEN r)$.
    \begin{subprove}
      \textbf{They are NOT logically equivalent.}
      We begin by applying the \textbf{Implication Equivalence} ($A \THEN B \equiv \NOT A \OR B$) to the left-hand side expression:
      \[
      (p \OR q) \THEN r \equiv \NOT(p \OR q) \OR r.
      \]
      Then, by applying \textbf{De Morgan's Law} ($\NOT(A \OR B) \equiv \NOT A \AND \NOT B$), we have
      \[
      \NOT(p \OR q) \OR r \equiv (\NOT p \AND \NOT q) \OR r.
      \]
      Now, we apply the \textbf{Implication Equivalence} to both implications on the right-hand side expression:
      \[
      (p \THEN r) \OR (q \THEN r) \equiv (\NOT p \OR r) \OR (\NOT q \OR r).
      \]
      By applying the \textbf{Associative Law} ($(A \OR B) \OR C \equiv A \OR (B \OR C)$), we group the terms:
      \[
      (\NOT p \OR r) \OR (\NOT q \OR r) \equiv \NOT p \OR \NOT q \OR r.
      \]
      Since the simplified forms, $(\NOT p \AND \NOT q) \OR r$ and $\NOT p \OR \NOT q \OR r$, are not the same, the two expressions are not logically equivalent.
      For instance, if we take $p=\TRUE$, $q=\FALSE$, and $r=\FALSE$, the left-hand expression becomes:
      \[
      (\TRUE \OR \FALSE) \THEN \FALSE \equiv \TRUE \THEN \FALSE \equiv \FALSE
      \]
      while the right-hand expression becomes:
      \[
      (\TRUE \THEN \FALSE) \OR (\FALSE \THEN \FALSE) \equiv \FALSE \OR \TRUE \equiv \TRUE.
      \]
      As the truth values differ for the same assignment, the two compound propositions are not logically equivalent. $\blacksquare$
    \end{subprove}
  \end{enumerate}
