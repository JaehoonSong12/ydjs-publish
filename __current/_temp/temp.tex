
  \item Let $p$, $q$, and $r$ be propositions.  Determine whether the 
  following pairs of compound propositions are logically equivalent.
  Justify your answer \textbf{without using a full truth table}.
  \begin{enumerate}
    \item $(p \THEN q) \AND (p \THEN r)$ and $p \THEN (q \AND r)$.
    \begin{subprove}
      Let \(p,q,r\) be propositions. 
      \[
        (p\THEN q)\AND(p\THEN r) \equiv (\NOT p\OR q)\AND(\NOT p\OR r)
      \]
      by applying the \textbf{Implication Equivalence} \(p\THEN q\equiv~\NOT p\OR q\) to both implications.
      Moreover, by \textbf{Distributivit Laws} in the form \((p\OR q)\AND(p\OR r)\equiv p\OR(q\AND r)\), 
      \[
        (\NOT p\OR q)\AND(\NOT p\OR r) \equiv \NOT p\OR(q\AND r).
      \]
      Lastly, applying the \textbf{Implication equivalence} in reverse,
      \[
        \NOT p\OR(q\AND r)\equiv~p\THEN(q\AND r).
      \]
      Thus, \((p\THEN q)\AND(p\THEN r)\equiv p\THEN(q\AND r)\), so the two formulas are \textbf{logically equivalent}. 
    \end{subprove}

    \item $(p \THEN r) \AND (q \THEN r)$ and $(p \OR q) \THEN r$.
    \begin{subprove}
      Let \(p,q,r\) be propositions. 
      \[
        (p\THEN r)\AND(q\THEN r) \equiv (\NOT p\OR r)\AND(\NOT q\OR r)
      \]
      by applying the \textbf{Implication Equivalence} \(p\THEN q\equiv~\NOT p\OR q\) to both implications.
      Moreover, by \textbf{Distributive Laws} in the form \((p\OR q)\AND(p\OR r)\equiv p\OR(q\AND r)\), 
      \[
        (\NOT p\OR r)\AND(\NOT q\OR r) \equiv (\NOT p\AND\NOT q)\OR r.
      \]
      For the right side, applying the \textbf{Implication Equivalence} first,
      \[
        (p\OR q)\THEN r \equiv \NOT(p\OR q)\OR r.
      \]
      Then by \textbf{De Morgan's Laws} \(\overline{p\vee q}\equiv\bar{p}\wedge\bar{q}\),
      \[
        \NOT(p\OR q)\OR r \equiv (\NOT p\AND\NOT q)\OR r.
      \]
      Thus, both expressions simplify to \((\NOT p\AND\NOT q)\OR r\), so the two formulas are \textbf{logically equivalent}.
    \end{subprove}

    \item $(p \THEN q) \THEN r$ and $p \THEN (q \THEN r)$.
    \begin{subprove}
      Let \(p,q,r\) be propositions. 
      \[
        (p\THEN q)\THEN r \equiv \NOT(\NOT p\OR q)\OR r
      \]
      by applying the \textbf{Implication Equivalence} \(p\THEN q\equiv~\NOT p\OR q\) to the inner implication.
      Next, by \textbf{De Morgan's Laws} \(\overline{p\vee q}\equiv\bar{p}\wedge\bar{q}\), we obtain
      \[
        \NOT(\NOT p\OR q)\OR r \equiv (p\AND\NOT q)\OR r.
      \]
      For the right side, applying the \textbf{Implication Equivalence} twice,
      \[
        p\THEN(q\THEN r) \equiv \NOT p\OR(\NOT q\OR r).
      \]
      By the \textbf{Associative Laws} \((p\vee q)\vee r\equiv p\vee(q\vee r)\),
      \[
        \NOT p\OR(\NOT q\OR r) \equiv \NOT p\OR\NOT q\OR r.
      \]
      Since \((p\AND\NOT q)\OR r\not\equiv\NOT p\OR\NOT q\OR r\), the two expressions are not equivalent.
      For instance, if we take \(p=\FALSE\), \(q=\TRUE\), and \(r=\FALSE\), the left expression is
      \[
        (\FALSE\THEN\TRUE)\THEN\FALSE \equiv \TRUE\THEN\FALSE \equiv \FALSE,
      \]
      while the right expression is
      \[
        \FALSE\THEN(\TRUE\THEN\FALSE) \equiv \FALSE\THEN\FALSE \equiv \TRUE.
      \]
      As the truth values differ for the same assignment, the two compound propositions are \textbf{not logically equivalent}.
    \end{subprove}

    \item $(p \OR q) \THEN r$ and $(p \THEN r) \OR (q \THEN r)$.
    \begin{subprove}
      Let \(p,q,r\) be propositions. 
      \[
        (p\OR q)\THEN r \equiv \NOT(p\OR q)\OR r
      \]
      by applying the \textbf{Implication Equivalence} \(p\THEN q\equiv~\NOT p\OR q\).
      Then, by \textbf{De Morgan's Laws} \(\overline{p\vee q}\equiv\bar{p}\wedge\bar{q}\), we have
      \[
        \NOT(p\OR q)\OR r \equiv (\NOT p\AND\NOT q)\OR r.
      \]
      For the right side, applying the \textbf{Implication Equivalence} to both implications,
      \[
        (p\THEN r)\OR(q\THEN r) \equiv (\NOT p\OR r)\OR(\NOT q\OR r).
      \]
      By the \textbf{Associative Laws} \((p\vee q)\vee r\equiv p\vee(q\vee r)\), we group the terms:
      \[
        (\NOT p\OR r)\OR(\NOT q\OR r) \equiv \NOT p\OR\NOT q\OR r.
      \]
      Since the simplified forms, \((\NOT p\AND\NOT q)\OR r\) and \(\NOT p\OR\NOT q\OR r\), are not the same, the two expressions are not equivalent.
      For instance, if we take \(p=\TRUE\), \(q=\FALSE\), and \(r=\FALSE\), the left expression becomes:
      \[
        (\TRUE\OR\FALSE)\THEN\FALSE \equiv \TRUE\THEN\FALSE \equiv \FALSE
      \]
      while the right expression becomes:
      \[
        (\TRUE\THEN\FALSE)\OR(\FALSE\THEN\FALSE) \equiv \FALSE\OR\TRUE \equiv \TRUE.
      \]
      As the truth values differ for the same assignment, the two compound propositions are \textbf{not logically equivalent}.
    \end{subprove}
  \end{enumerate}
