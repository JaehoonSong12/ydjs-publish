  \begin{enumerate}
    \item $(p \THEN q) \AND (p \THEN r)$ and $p \THEN (q \AND r)$.\\
    \begin{subanswer}
      \textbf{They are logically equivalent.} We prove this step by step using the logical equivalences from the reference table:
      
      \textbf{Step 1:} Apply the \textbf{implication equivalence} ($p \rightarrow q \equiv \bar{p} \vee q$) to both implications:
      \[
        (p \THEN q) \AND (p \THEN r)
        \equiv (\NOT p \OR q) \AND (\NOT p \OR r)
      \]
      
      \textbf{Step 2:} Apply the \textbf{distributive law} ($p \wedge (q \vee r) \equiv (p \wedge q) \vee (p \wedge r)$) in reverse:
      \[
        (\NOT p \OR q) \AND (\NOT p \OR r)
        \equiv \NOT p \OR (q \AND r)
      \]
      
      \textbf{Step 3:} Apply the \textbf{implication equivalence} again to the right side:
      \[
        \NOT p \OR (q \AND r) \equiv p \THEN (q \AND r)
      \]
      
      Therefore, $(p \THEN q) \AND (p \THEN r) \equiv p \THEN (q \AND r)$, so the two formulas are logically equivalent.
    \end{subanswer}

    \item $(p \THEN r) \AND (q \THEN r)$ and $(p \OR q) \THEN r$.\\
    \begin{subanswer}
      \textbf{They are logically equivalent.} We prove this step by step using the logical equivalences from the reference table:
      
      \textbf{Step 1:} Apply the \textbf{implication equivalence} ($p \rightarrow q \equiv \bar{p} \vee q$) to both implications:
      \[
        (p \THEN r) \AND (q \THEN r)
        \equiv (\NOT p \OR r) \AND (\NOT q \OR r)
      \]
      
      \textbf{Step 2:} Apply the \textbf{distributive law} ($p \wedge (q \vee r) \equiv (p \wedge q) \vee (p \wedge r)$) in reverse:
      \[
        (\NOT p \OR r) \AND (\NOT q \OR r)
        \equiv (\NOT p \AND \NOT q) \OR r
      \]
      
      \textbf{Step 3:} For the right side, apply the \textbf{implication equivalence} first:
      \[
        (p \OR q) \THEN r \equiv \NOT(p \OR q) \OR r
      \]
      
      \textbf{Step 4:} Apply \textbf{De Morgan's law} ($\overline{p \vee q} \equiv \bar{p} \wedge \bar{q}$):
      \[
        \NOT(p \OR q) \OR r \equiv (\NOT p \AND \NOT q) \OR r
      \]
      
      Since both expressions simplify to $(\NOT p \AND \NOT q) \OR r$, the two formulas are logically equivalent.
    \end{subanswer}

    \item $(p \THEN q) \THEN r$ and $p \THEN (q \THEN r)$.\\
    \begin{subanswer}
      \textbf{They are NOT logically equivalent.} We prove this step by step using the logical equivalences from the reference table:
      
      \textbf{Step 1:} For the left side, apply the \textbf{implication equivalence} ($p \rightarrow q \equiv \bar{p} \vee q$) to the inner implication:
      \[
        (p \THEN q) \THEN r \equiv \NOT(\NOT p \OR q) \OR r
      \]
      
      \textbf{Step 2:} Apply \textbf{De Morgan's law} ($\overline{p \vee q} \equiv \bar{p} \wedge \bar{q}$) to $\NOT(\NOT p \OR q)$:
      \[
        \NOT(\NOT p \OR q) \OR r \equiv (p \AND \NOT q) \OR r
      \]
      
      \textbf{Step 3:} For the right side, apply the \textbf{implication equivalence} to both implications:
      \[
        p \THEN (q \THEN r) \equiv \NOT p \OR (\NOT q \OR r)
      \]
      
      \textbf{Step 4:} Apply the \textbf{associative law} ($(p \vee q) \vee r \equiv p \vee (q \vee r)$) to the right side:
      \[
        \NOT p \OR (\NOT q \OR r) \equiv \NOT p \OR \NOT q \OR r
      \]
      
      Since $(p \AND \NOT q) \OR r \not\equiv \NOT p \OR \NOT q \OR r$, the two formulas are not logically equivalent. A counterexample is $p=\FALSE,\ q=\TRUE,\ r=\FALSE$:
      \[
        (p \THEN q) \THEN r = ( \FALSE \THEN \TRUE ) \THEN \FALSE = \TRUE \THEN \FALSE = \FALSE,
      \]
      whereas
      \[
        p \THEN (q \THEN r) = \FALSE \THEN (\TRUE \THEN \FALSE) = \FALSE \THEN \FALSE = \TRUE.
      \]
      Therefore, the two formulas are not logically equivalent.
    \end{subanswer}

    \item $(p \OR q) \THEN r$ and $(p \THEN r) \OR (q \THEN r)$.\\
    \begin{subanswer}
      \textbf{They are NOT logically equivalent.} We prove this step by step using the logical equivalences from the reference table:
      
      \textbf{Step 1:} For the left side, apply the \textbf{implication equivalence} ($p \rightarrow q \equiv \bar{p} \vee q$):
      \[
        (p \OR q) \THEN r \equiv \NOT(p \OR q) \OR r
      \]
      
      \textbf{Step 2:} Apply \textbf{De Morgan's law} ($\overline{p \vee q} \equiv \bar{p} \wedge \bar{q}$):
      \[
        \NOT(p \OR q) \OR r \equiv (\NOT p \AND \NOT q) \OR r
      \]
      
      \textbf{Step 3:} For the right side, apply the \textbf{implication equivalence} to both implications:
      \[
        (p \THEN r) \OR (q \THEN r) \equiv (\NOT p \OR r) \OR (\NOT q \OR r)
      \]
      
      \textbf{Step 4:} Apply the \textbf{associative law} ($(p \vee q) \vee r \equiv p \vee (q \vee r)$) to group the terms:
      \[
        (\NOT p \OR r) \OR (\NOT q \OR r) \equiv \NOT p \OR \NOT q \OR r
      \]
      
      Since $(\NOT p \AND \NOT q) \OR r \not\equiv \NOT p \OR \NOT q \OR r$, the two formulas are not logically equivalent. Note that $(\NOT p \AND \NOT q) \OR r \implies \NOT p \OR \NOT q \OR r$ holds (by the \textbf{domination law}), but the converse fails.
      
      A concrete counterexample is $p=\FALSE,\ q=\TRUE,\ r=\FALSE$:
      \[
        (p \OR q) \THEN r = (\FALSE \OR \TRUE) \THEN \FALSE = \TRUE \THEN \FALSE = \FALSE,
      \]
      while
      \[
        (p \THEN r) \OR (q \THEN r) = (\FALSE \THEN \FALSE) \OR (\TRUE \THEN \FALSE)
        = \TRUE \OR \FALSE = \TRUE.
      \]
      Therefore, the two formulas are not logically equivalent; moreover, $(p \OR q) \THEN r$ implies $(p \THEN r) \OR (q \THEN r)$, but not conversely.
    \end{subanswer}
  \end{enumerate}

  Prove the following directly from the two definitions and three axioms
  given in ICA2 on 8/20.
  \begin{define}{ICA2 Definitions and Axioms}
    Here are the ICA2 Definitions and Axioms as follows.
    \begin{enumerate}
      \item \textbf{Definition:} An integer $n$ is \underbar{even} if there exists some integer $k$ such that
      \[
      n = 2k.
      \]
      \item \textbf{Definition:} An integer $n$ is \underbar{odd} if there exists some integer $k$ such that
      \[
      n = 2k + 1.
      \]
      \item \textbf{Axiom 1:} The sum of two integers is again an integer. (Closure under addition)
      \item \textbf{Axiom 2:} The product of two integers is again an integer. (Closure under multiplication)
      \item \textbf{Axiom 3:} An integer is odd if and only if it is not even.
    \end{enumerate}
  \end{define}
  You may not use any other results without proving them first.
  Every statement in your proof must be fully justified.
  See \textbf{Elements of (Proof) Style} guidelines in Canvas for information.
  \begin{enumerate}
  \item Direct proof: For all integers $n$, if $n$ is odd, then $9n + 5$ is even.\\
  \begin{subprove}
    Let $n$ be an arbitrary integer then assume $n$ is odd. By the definition of odd, 
    there exists an integer $k$ such that
    \[
      n = 2k + 1.
    \]
    Substitute into $9n + 5$:
    \[
      9n + 5 = 9(2k + 1) + 5 = 18k + 9 + 5 = 18k + 14.
    \]
    By Axiom 2 (closure under multiplication) and since $9$ and $k$ are integers, $9k$ is an integer; by repeated application of Axiom 2 and Axiom 1 the product $18k$ and sums $18k+9$ and $18k+14$ are integers. Factor a $2$:
    \[
      18k + 14 = 2(9k + 7).
    \]
    By Axiom 1 and Axiom 2, $9k+7$ is an integer. Thus $9n+5$ has the form $2(\text{integer})$, hence by the definition of even $9n+5$ is even. This completes the direct proof.
  \end{subprove}

  \item Contrapositive proof: For all integers $n$, if $7n + 5$ is odd, then $n$ is even.\\
  \begin{subanswer}
    \textbf{Claim.} For every integer $n$, if $7n+5$ is odd then $n$ is even.\\

    \textbf{Remark (justifying the contrapositive method).} The implication $P \THEN Q$ is logically equivalent to its contrapositive $\NOT Q \THEN \NOT P$, because both are false exactly in the case $P$ is true and $Q$ is false. Hence proving the contrapositive suffices.\\

    \textbf{Proof (by contrapositive).} The contrapositive of the statement is: \inlinequote{If $n$ is odd, then $7n+5$ is even.} Assume $n$ is odd. By definition of odd there exists integer $k$ with $n = 2k + 1$. Then
    \[
      7n + 5 = 7(2k+1) + 5 = 14k + 7 + 5 = 14k + 12 = 2(7k + 6).
    \]
    By Axiom 2 and Axiom 1, $7k+6$ is an integer, so $7n+5$ equals $2$ times an integer. By the definition of even, $7n+5$ is even. Thus the contrapositive is proved, and therefore the original implication holds: if $7n+5$ is odd then $n$ is even.
  \end{subanswer}

  \item Proof by contradiction: For all integers $n$, if $3n + 2$ is even, then $n$ is even.\\
  \begin{subanswer}
    \textbf{Claim.} For every integer $n$, if $3n+2$ is even then $n$ is even.\\

    \textbf{Proof (by contradiction).} Assume, for contradiction, that $3n+2$ is even and $n$ is \emph{not} even. By Axiom 3, $n$ is not even implies $n$ is odd. Thus there exists integer $k$ with $n = 2k + 1$. Compute:
    \[
      3n + 2 = 3(2k+1) + 2 = 6k + 3 + 2 = 6k + 5.
    \]
    Write $6k+5 = 2(3k+2) + 1$, so $6k+5$ is of the form $2(\text{integer})+1$, hence odd by the definition of odd. This contradicts the assumption that $3n+2$ is even. Therefore the assumption that $n$ is not even is false, so $n$ must be even. This completes the proof by contradiction.
  \end{subanswer}
  \end{enumerate}

\item \textbf{Logical Equivalences without Truth Tables} (10 points)\\
Let $p$, $q$, and $r$ be propositions. Determine whether the
following pairs of compound propositions are logically equivalent.
Justify your answer \textbf{without using a full truth table}.
\begin{enumerate}[label=\arabic*.]
 \item $(p \THEN q) \AND (p \THEN r)$ and $p \THEN (q \AND r)$.
 \begin{subanswer}
  \textbf{They are logically equivalent.} Using the equivalence $A \THEN B \equiv \NOT A \OR B$, we can transform both expressions.
  \begin{itemize}
   \item Left side: $(p \THEN q) \AND (p \THEN r) \equiv (\NOT p \OR q) \AND (\NOT p \OR r)$. By the distributive law, this is equivalent to $\NOT p \OR (q \AND r)$.
   \item Right side: $p \THEN (q \AND r) \equiv \NOT p \OR (q \AND r)$.
  \end{itemize}
  Since both expressions simplify to the same form, they are logically equivalent.
 \end{subanswer}
 \item $(p \THEN r) \AND (q \THEN r)$ and $(p \OR q) \THEN r$.
 \begin{subanswer}
  \textbf{They are logically equivalent.} Using the equivalence $A \THEN B \equiv \NOT A \OR B$, we can transform both expressions.
  \begin{itemize}
   \item Left side: $(p \THEN r) \AND (q \THEN r) \equiv (\NOT p \OR r) \AND (\NOT q \OR r)$. By the distributive law, this is equivalent to $(\NOT p \AND \NOT q) \OR r$. By De Morgan's laws, this is equivalent to $\NOT (p \OR q) \OR r$.
   \item Right side: $(p \OR q) \THEN r \equiv \NOT (p \OR q) \OR r$.
  \end{itemize}
  Since both expressions simplify to the same form, they are logically equivalent.
 \end{subanswer}
 \item $(p \THEN q) \THEN r$ and $p \THEN (q \THEN r)$.
 \begin{subanswer}
  \textbf{They are NOT logically equivalent.} Consider the case where $p=\FALSE, q=\TRUE, r=\FALSE$.
  \begin{itemize}
   \item Left side: $(p \THEN q) \THEN r \equiv (\FALSE \THEN \TRUE) \THEN \FALSE \equiv \TRUE \THEN \FALSE \equiv \FALSE$.
   \item Right side: $p \THEN (q \THEN r) \equiv \FALSE \THEN (\TRUE \THEN \FALSE) \equiv \FALSE \THEN \FALSE \equiv \TRUE$.
  \end{itemize}
  Since the two expressions have different truth values for the same assignment, they are not logically equivalent.
 \end{subanswer}
 \item $(p \OR q) \THEN r$ and $(p \THEN r) \OR (q \THEN r)$.
 \begin{subanswer}
  \textbf{They are NOT logically equivalent.} Consider the case where $p=\FALSE, q=\FALSE, r=\FALSE$.
  \begin{itemize}
   \item Left side: $(p \OR q) \THEN r \equiv (\FALSE \OR \FALSE) \THEN \FALSE \equiv \FALSE \THEN \FALSE \equiv \TRUE$.
   \item Right side: $(p \THEN r) \OR (q \THEN r) \equiv (\FALSE \THEN \FALSE) \OR (\FALSE \THEN \FALSE) \equiv \TRUE \OR \TRUE \equiv \TRUE$.
  \end{itemize}
  Consider the case where $p=\FALSE, q=\TRUE, r=\TRUE$.
  \begin{itemize}
   \item Left side: $(p \OR q) \THEN r \equiv (\FALSE \OR \TRUE) \THEN \TRUE \equiv \TRUE \THEN \TRUE \equiv \TRUE$.
   \item Right side: $(p \THEN r) \OR (q \THEN r) \equiv (\FALSE \THEN \TRUE) \OR (\TRUE \THEN \TRUE) \equiv \TRUE \OR \TRUE \equiv \TRUE$.
  \end{itemize}
  Let's try another case. Consider $p=\TRUE, q=\TRUE, r=\FALSE$.
  \begin{itemize}
   \item Left side: $(p \OR q) \THEN r \equiv (\TRUE \OR \TRUE) \THEN \FALSE \equiv \TRUE \THEN \FALSE \equiv \FALSE$.
   \item Right side: $(p \THEN r) \OR (q \THEN r) \equiv (\TRUE \THEN \FALSE) \OR (\TRUE \THEN \FALSE) \equiv \FALSE \OR \FALSE \equiv \FALSE$.
  \end{itemize}
  The expressions are not equivalent. Let's find a counterexample.
  Using De Morgan's laws on the left side, we get $\NOT(p \OR q) \OR r \equiv (\NOT p \AND \NOT q) \OR r$.
  On the right side, we get $(\NOT p \OR r) \OR (\NOT q \OR r) \equiv \NOT p \OR \NOT q \OR r$.
  Since $(\NOT p \AND \NOT q) \OR r$ is not equivalent to $\NOT p \OR \NOT q \OR r$ (as shown by a truth table), the original expressions are not logically equivalent. For example, let $p = \TRUE, q = \FALSE, r = \FALSE$.
  \begin{itemize}
   \item Left side: $(\TRUE \OR \FALSE) \THEN \FALSE \equiv \TRUE \THEN \FALSE \equiv \FALSE$.
   \item Right side: $(\TRUE \THEN \FALSE) \OR (\FALSE \THEN \FALSE) \equiv \FALSE \OR \TRUE \equiv \TRUE$.
  \end{itemize}
  Since the two expressions have different truth values, they are not logically equivalent.
 \end{subanswer}
\end{enumerate}

\item \textbf{Direct Proof, Contrapositive Proof, Proof by Contradiction} (10 points)\\
Prove the following directly from the two definitions and three axioms
given in ICA2 on 8/20.
\begin{define}{ICA2 Definitions and Axioms}
 Here are the ICA2 Definitions and Axioms as follows.
 \begin{enumerate}
  \item \textbf{Definition:} An integer $n$ is \underbar{even} if there exists some integer $k$ such that
  \[
  n = 2k.
  \]
  \item \textbf{Definition:} An integer $n$ is \underbar{odd} if there exists some integer $k$ such that
  \[
  n = 2k + 1.
  \]
  \item \textbf{Axiom 1:} The sum of two integers is again an integer. (Closure under addition)
  \item \textbf{Axiom 2:} The product of two integers is again an integer. (Closure under multiplication)
  \item \textbf{Axiom 3:} An integer is odd if and only if it is not even.
  \end{enumerate}
\end{define}
You may not use any other results without proving them first.
Every statement in your proof must be fully justified.
See \textbf{Elements of (Proof) Style} guidelines in Canvas for information.
\begin{enumerate}[label=\arabic*.]
 \item Direct proof: For all integers $n$, if $n$ is odd, then $9n + 5$ is even.
 \begin{subprove}
  \textbf{Claim:} For all integers $n$, if $n$ is odd, then $9n+5$ is even.
  \textbf{Proof:} Let $n$ be an odd integer. By the definition of an odd integer, there exists some integer $k$ such that $n = 2k+1$.
  Now, consider the expression $9n+5$.
  \[ 9n+5 = 9(2k+1) + 5 \]
  \[ = 18k + 9 + 5 \]
  \[ = 18k + 14 \]
  \[ = 2(9k + 7) \]
  By Axiom 2, the product of integers 9 and $k$ is an integer. By Axiom 1, the sum of integers $9k$ and 7 is an integer. Let $m = 9k+7$. Thus, $m$ is an integer.
  The expression $9n+5$ can be written as $2m$ for some integer $m$. By the definition of an even integer, $9n+5$ is even.
  Therefore, if $n$ is odd, then $9n+5$ is even. $\blacksquare$
 \end{subprove}
 \item Contrapositive proof: For all integers $n$, if $7n + 5$ is odd, then $n$ is even.
 \begin{subprove}
  \textbf{Claim:} For all integers $n$, if $7n+5$ is odd, then $n$ is even.
  \textbf{Proof (by contraposition):} The contrapositive of the statement "if $7n+5$ is odd, then $n$ is even" is "if $n$ is not even, then $7n+5$ is not odd." By Axiom 3, this is equivalent to "if $n$ is odd, then $7n+5$ is even." We will prove this contrapositive statement directly.
  Let $n$ be an odd integer. By the definition of an odd integer, there exists some integer $k$ such that $n = 2k+1$.
  Now, consider the expression $7n+5$.
  \[ 7n+5 = 7(2k+1) + 5 \]
  \[ = 14k + 7 + 5 \]
  \[ = 14k + 12 \]
  \[ = 2(7k + 6) \]
  By Axiom 2, the product of integers 7 and $k$ is an integer. By Axiom 1, the sum of integers $7k$ and 6 is an integer. Let $m = 7k+6$. Thus, $m$ is an integer.
  The expression $7n+5$ can be written as $2m$ for some integer $m$. By the definition of an even integer, $7n+5$ is even.
  Since the contrapositive is true, the original statement is also true. $\blacksquare$
 \end{subprove}
 \item Proof by contradiction: For all integers $n$, if $3n + 2$ is even, then $n$ is even.
 \begin{subprove}
  \textbf{Claim:} For all integers $n$, if $3n+2$ is even, then $n$ is even.
  \textbf{Proof (by contradiction):} Assume the negation of the statement is true. That is, there exists an integer $n$ such that $3n+2$ is even and $n$ is not even.
  By Axiom 3, if $n$ is not even, then $n$ must be odd.
  By the definition of an odd integer, there exists some integer $k$ such that $n = 2k+1$.
  By the definition of an even integer, since $3n+2$ is even, there exists some integer $m$ such that $3n+2 = 2m$.
  Now we substitute our expression for $n$:
  \[ 3(2k+1) + 2 = 2m \]
  \[ 6k + 3 + 2 = 2m \]
  \[ 6k + 5 = 2m \]
  \[ 5 = 2m - 6k \]
  \[ 5 = 2(m - 3k) \]
  By Axiom 2, the product of integers 3 and $k$ is an integer. By Axiom 1, the difference of integers $m$ and $3k$ is an integer. Let $j = m-3k$. Thus, $j$ is an integer.
  We have $5 = 2j$. This means that 5 is an even integer by the definition of an even integer. However, 5 is an odd integer, which, by Axiom 3, means it cannot be even.
  This is a contradiction. Therefore, our initial assumption must be false.
  Thus, for all integers $n$, if $3n+2$ is even, then $n$ is even. $\blacksquare$
 \end{subprove}
\end{enumerate}
