\section*{Lecture Note}
Logic notations should never appear in a proof. Instead of Implication/conditional, $\,p \rightarrow q \equiv \bar{q} \to \bar{p}$,
\begin{enumerate}
    \item If $p$, then $q$
    \item $q$ if $p$
    \item $q$ only if $p$
    \item $p$ implies $q$
    \item $q$ whenever $p$
    \item $q$ is necessary for $p$
    \item $p$ is sufficient for $q$
\end{enumerate}

Instead of biconditional, $p \leftrightarrow q \equiv \left(p \to q\right) \wedge \left(q \to p\right)$
\begin{enumerate}
    \item $p$ if and only if $q$
    \item $p$ is necessary and sufficient for $q$
\end{enumerate}

Predicate is a proposition with 1 or more variables, for example, $$x>2$$.

Almost always a predicate becomes a proposition through qualification.

\begin{enumerate}
    \item $\mathbb{Z}$: the set of all integers
    \item $\mathbb{Q}$: the set of all rational numbers
    \item $\mathbb{R}$: the set of all realnumber
    \item $\mathbb{N}$ or $\mathbb{Z}_{>0}$: the set of all natual \inlinequote{numbers} which are the set of positive integers (not including $0$)
\end{enumerate}


\begin{state}{Even Integers}
    An integer $n$ is \underbar{even} if there exists \footnote{existential quantification logic $\exists$} an integer $k$ such that $$n=2k$$
\end{state}



\begin{state}{Axiom: Sum of Integers}
    The sum of two integers\footnote{Universal quantification for all, for every, for each} is again an integer .
\end{state}


\begin{state}{Claim}
    The sum of two even integers is again an even integer.
\end{state}
\begin{subprove}
    Let $x$ be an integer abd $y$ be an interger \footnote{$\forall x,y \in \mathbb{Z}$}. 
    Suppose $x$ is even and $y$ is even. Then, by definition, there exists an integer 
    $k$ and an integer $j$ such that $x=2k$ and $y=2j$. Then, 
    $x+y=\left(2k\right)+\left(2j\right)=2\left(k+j\right)$. By the axiom, since $k$ and 
    $j$ are integers, then so is $\left(k+j\right)$. Hence, by definition, $x+y$ is an 
    even integer.
\end{subprove}
