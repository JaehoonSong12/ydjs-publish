% ------------------------------------------------------------------------------
% Resume Header Environment
% ------------------------------------------------------------------------------
% DESCRIPTION:
%   Environment for creating a standardized resume header with contact information
%   and social media links. Provides consistent formatting across resume documents.
%
% FEATURES:
%   - Centered layout with proper spacing
%   - Social media icons with hyperlinks (input between begin/end)
%   - Contact information formatting
%   - Consistent typography and spacing
%
% DEPENDENCIES:
%   \usepackage{graphicx}      % for \raisebox, \resizebox
%   \usepackage{fontawesome5}  % for social media icons
%   \usepackage{hyperref}      % for clickable links
%
% USAGE EXAMPLES:
%   \begin{resume-header}{John Doe}{+1-234-567-8900}{john@email.com}{johnwebsite.com}{New York, NY - 10001, USA}
%   \socialicon{\faLinkedin} \href{https://www.linkedin.com/in/john-profile/}{john-profile} | 
%   \socialicon{\faGithub} \href{https://github.com/john-username}{john-username} |
%   \socialicon{\faTwitter} \href{https://twitter.com/john-handle}{john-handle}
%   \end{resume-header}

%   \begin{resume-header}
%     {John Doe}{www.johnwebsite.com}
%     {New York, NY - 10001, USA}{+1-234-567-8900}{john@email.com}
%     \socialicon{\faLinkedin} \href{https://www.linkedin.com/in/john-profile/}{john-profile} | 
%     \socialicon{\faGithub} \href{https://github.com/john-username}{john-username} |
%     \socialicon{\faTwitter} \href{https://twitter.com/john-handle}{john-handle} |
%     \socialicon{\faLinkedin} \href{https://www.linkedin.com/in/#5/}{john-profile} \\
%     \socialicon{\faInstagram} \href{https://www.instagram.com/#8/}{john-instagram} |
%     \socialicon{\faFacebook} \href{https://www.facebook.com/#9/}{john-facebook} |
%     \socialicon{\faYoutube} \href{https://www.youtube.com/@#10}{john-youtube}
%   \end{resume-header}
% ------------------------------------------------------------------------------
\newcommand{\socialicon}[1]{\raisebox{-0.05em}{\resizebox{!}{1em}{#1}}}
\NewDocumentEnvironment{resume-header}{m m m m m}{%
\begin{center}
    {\LARGE\bfseries \href{#1}{\socialicon{\faIcon{github}}} \textbf{#2}} \\ \vspace{0.5em}%\Huge?
    {\small #3 | \socialicon{\faIcon{mobile-alt}} #4 | \socialicon{\faIcon{envelope}} \href{mailto:#5}{#5}} \\ \vspace{0.5em}
}{%
\end{center}
}







% Underlined, all-caps section title macro (no indent; use \hrule for the line)
\newcommand{\resumetitle}[1]{%
  {\noindent\bfseries\MakeUppercase{#1}\par\nobreak\vspace{-0.6em}%
  \noindent\rule{\linewidth}{0.6pt}\par}
}

%-------------------------------------
% Environment: resume-itemize
%-------------------------------------
\newenvironment{resume-itemize}[4]{%
  % Header formatting
%   \vspace{1.0em}
  \noindent\textbf{#1} \,|\, #2 \hfill \textit{#4}\\
  \noindent\textit{#3}
  \vspace{-0.4em}
  \begin{itemize}
    \setlength{\itemsep}{0.2em}
    \setlength{\parskip}{0pt}
}{%
  \end{itemize}
}
