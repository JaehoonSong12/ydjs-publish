The following definitions are presented from standard mathematical sources (\textbf{Sets}) together with the author's own (or the student, myself, Jaehoon Song) clarifications and labels where applicable.

\begin{define}{Basic Set Concepts}
  \begin{enumerate}
    \item \textbf{Universal Set:} A \underbar{universal set} $\setUNIV$ is a 
    set that contains all objects under consideration in a particular context or problem.
    \item \textbf{Empty Set:} The \underbar{empty set}, denoted $\setNULL$, 
    is the unique set that contains no elements.
    \item \textbf{Subset:} Let $A$ and $B$ be sets. We say $A$ is a 
    \underbar{subset} of $B$ ($A \subseteq B$) if (and only if) 
    for all $x \in A$, $x \in B$.
    \item \textbf{Set Equality:} Let $A$ and $B$ be sets. We say $A$ and $B$ are 
    \underbar{equal} ($A = B$) if $A \subseteq B$ and $B \subseteq A$.
  \end{enumerate}
\end{define}

\begin{define}{Set Operations}
  \begin{enumerate}
    \item \textbf{Set Difference:} Let $A$, $B$ be subsets of a universal set $\setUNIV$. The set difference $A \setminus B$ is defined to be the set $\{ x \in \setUNIV \LDIV x \in A \text{ and } x \notin B\}$. \textit{It is sometimes called the relative complement of $B$ in $A$, and denoted in the textbook as $A - B$.}
    \item \textbf{Cartesian Product:} Let $A$ and $B$ be sets. The \underbar{Cartesian product} of $A$ and $B$, denoted $A \times B$, is the set consisting of all ordered pairs:
    \[
    A \times B = \{(a, b) \LDIV a \in A, b \in B\}.
    \]
  \end{enumerate}
\end{define}

The following definitions and claims extend the understanding of set operations and properties.

\begin{define}{Set Properties}
  Let $A, B \subseteq \setUNIV$ where $\setUNIV$ is a universal set.
  \begin{enumerate}
    \item \textbf{Union and Subset Property:} $A \subseteq A \cup B$
    \item \textbf{Empty Set Property:} $\setNULL \subseteq A$
    \item \textbf{Union Equality Condition:} $A \cup B = A$ if and only if $B \subseteq A$
  \end{enumerate}
\end{define}
