\begin{enumerate}
  \item Let $A, B$ be subsets of a universal set $\mathcal{U}$. Prove the following using the element method:
  \begin{enumerate}
  \item $A \setminus B \subseteq A$.
  \begin{proof}
    Let $x \in A \backslash B$ be arbitrary, for the sake of direct proof. By definition of set difference, $x \in A$ and $x \notin B$. Thus, $x \in A$ (trivial). Therefore, $A \backslash B \subseteq A$ by definition of Subset.
  \end{proof}

  \item $A \setminus B = A \cap \bar{B}$.
  \begin{proof}
    By definition, the equality $A \backslash B=A \cap \bar{B}$ requires showing both 
    directions of subset relations, $A \backslash B \subseteq A \cap \bar{B}$ 
    and $A \backslash B \supseteq A \cap \bar{B}$.\vspace*{10px}\\
    $(\subseteq)$ Let $x \in A \backslash B$ be arbitrary. By definition of set difference, 
    $x \in A$ and $x \notin B$. Since $x \notin B$ by definition of complementary set, it 
    follows that $x \in \bar{B}$. Therefore, $x \in A$ and $x \in \bar{B}$, so $x \in A \cap \bar{B}$. 
    Hence, $A \backslash B \subseteq A \cap \bar{B}$.\vspace*{10px}\\
    $(\supseteq)$ Let $x \in A \cap \bar{B}$ be arbitrary. By definition of intersection, 
    $x \in A$ and $x \in \bar{B}$. Since $x \in \bar{B}$, it follows that $x \notin B$ by 
    definition of complementary set. Therefore, $x \in A$ and $x \notin B$, so $x \in A \backslash B$. 
    Hence, $A \cap \bar{B} \subseteq A \backslash B$.\vspace*{10px}\\
    Therefore, by ( $\subseteq$ ) and ( $\supseteq$ ), $A \backslash B=A \cap \bar{B}$.
  \end{proof}

  \item $A \cap (B \setminus A) = \emptyset$.
  \begin{proof}
    Assume there exists an element $x \in A \cap(B \backslash A)$ by negation 
    of empty set definition for the sake of contradiction. Then $x \in A$ and 
    $x \in B \backslash A$. Since $x \in B \backslash A$, it follows that 
    $x \in B$ and $x \notin A$, which contradicts the assumption, $x \in A$. 
    Therefore, $A \cap(B \backslash A)=\emptyset$.
  \end{proof}
  \end{enumerate}



  \item Let $A, B$, and $C$ be subsets of a universal set $\mathcal{U}$. Prove or disprove the following statements:
  \begin{enumerate}
  \item If $A \cap B = A \cap C$, then $B = C$.
  \begin{proof}[Disproof]
    Let $\mathcal{U}=\{1,2,3,4\}, A=\{1,2\}, B=\{1,3\}$, and $C=\{1,4\}$ be 
    subsets of a universal set $\mathcal{U}$ for the sake of counterexample. 
    Then, $A \cap B=\{1\}$ and $A \cap C=\{1\}$, that is, $A \cap B=A \cap C$. 
    However, $B=\{1,3\} \neq\{1,4\}=C$. Therefore, it is \textbf{false} 
    that $A \cap B=A \cap C$ only if $B=C$.
  \end{proof}

  \item If $A \cup B = A \cup C$, then $B = C$.
  \begin{proof}[Disproof]
    Let $\mathcal{U}=\{1,2,3,4\}, A=\{1,2\}, B=\{2,3\}$, and $C=\{3\}$ be subsets of a 
    universal set $\mathcal{U}$ for the sake of counterexample. Then $A \cup B=\{1,2,3\}$ 
    and $A \cup C=\{1,2,3\}$, so $A \cup B=A \cup C$. However, $B=\{2,3\} \neq\{3\}=C$. 
    Therefore, it is \textbf{false} that $A \cup B=A \cup C$ only if $B=C$.
  \end{proof}

  \item If $A \cap B = A \cap C$ and $A \cup B = A \cup C$, then $B = C$.
  \begin{proof}
    Assume $A \cap B=A \cap C$ and $A \cup B=A \cup C$ for the sake of direct proof. 
    By definition of set equality, both directions of subset relations are required as follows.\vspace*{10px}\\
    $(\subseteq)$ Let $x \in B$ be arbitrary. Since $x \in B$, it follows that 
    $x \in A \cup B=A \cup C$. Therefore, $x \in A$ or $x \in C$ (trivial). 
    If $x \in A$, then $x \in A \cap B=A \cap C$, so $x \in C$. Hence, $B \subseteq C$.\vspace*{10px}\\
    $(\supseteq)$ Let $x \in C$ be arbitrary. Since $x \in C$, it follows that 
    $x \in A \cup C=A \cup B$. Therefore, $x \in A$ or $x \in B$ (trivial). 
    If $x \in A$, then $x \in A \cap C=A \cap B$, so $x \in B$. Hence, $C \subseteq B$.\vspace*{10px}\\
    Therefore, by $(\subseteq)$ and $(\supseteq), B=C$ when $A \cap B=A \cap C$ and $A \cup B=A \cup C$.
  \end{proof}
  \end{enumerate}

  \item Let $A, B, C$, and $D$ be subsets of a universal set $\mathcal{U}$. Prove the following:
  \begin{enumerate}
  \item $(A \cup B)=(A \cap B) \cup(A \backslash B) \cup(B \backslash A)$.
  \begin{proof}
    By definition, the equality $(A \cup B)=(A \cap B) \cup(A \backslash B) \cup(B \backslash A)$ requires showing both directions of subset relations, $(A \cup B) \subseteq (A \cap B) \cup(A \backslash B) \cup(B \backslash A)$ and $(A \cup B) \supseteq (A \cap B) \cup(A \backslash B) \cup(B \backslash A)$.\vspace*{10px}\\
    $(\subseteq)$ Let $x \in A \cup B$ be arbitrary. Then $x \in A$ or $x \in B$ by definition of union. We consider two cases:\vspace*{5px}\\
    \textit{Case 1:} If $x \in A$ and $x \in B$, then $x \in A \cap B$.\vspace*{5px}\\
    \textit{Case 2:} If $x \in A$ and $x \notin B$, then $x \in A \backslash B$.\vspace*{5px}\\
    \textit{Case 3:} If $x \notin A$ and $x \in B$, then $x \in B \backslash A$.\vspace*{5px}\\
    In all cases, $x \in (A \cap B) \cup(A \backslash B) \cup(B \backslash A)$. Hence, $(A \cup B) \subseteq (A \cap B) \cup(A \backslash B) \cup(B \backslash A)$.\vspace*{10px}\\
    $(\supseteq)$ Let $x \in (A \cap B) \cup(A \backslash B) \cup(B \backslash A)$ be arbitrary. Then $x \in A \cap B$ or $x \in A \backslash B$ or $x \in B \backslash A$. In each case:\vspace*{5px}\\
    \textit{Case 1:} If $x \in A \cap B$, then $x \in A$ and $x \in B$, so $x \in A \cup B$.\vspace*{5px}\\
    \textit{Case 2:} If $x \in A \backslash B$, then $x \in A$, so $x \in A \cup B$.\vspace*{5px}\\
    \textit{Case 3:} If $x \in B \backslash A$, then $x \in B$, so $x \in A \cup B$.\vspace*{5px}\\
    Therefore, $(A \cap B) \cup(A \backslash B) \cup(B \backslash A) \subseteq A \cup B$.\vspace*{10px}\\
    Therefore, by $(\subseteq)$ and $(\supseteq)$, $(A \cup B)=(A \cap B) \cup(A \backslash B) \cup(B \backslash A)$.
  \end{proof}

  \item $(A \cap B)$, $(A \backslash B)$, and $(B \backslash A)$ are all pairwise disjoint.
  \begin{proof}
    We need to show that any two of these sets have empty intersection.\vspace*{10px}\\
    \textit{Part 1:} Show $(A \cap B) \cap (A \backslash B) = \emptyset$.\vspace*{5px}\\
    Assume there exists $x \in (A \cap B) \cap (A \backslash B)$ for the sake of contradiction. Then $x \in A \cap B$ and $x \in A \backslash B$. Since $x \in A \cap B$, we have $x \in A$ and $x \in B$. Since $x \in A \backslash B$, we have $x \in A$ and $x \notin B$. This contradicts $x \in B$. Therefore, $(A \cap B) \cap (A \backslash B) = \emptyset$.\vspace*{10px}\\
    \textit{Part 2:} Show $(A \cap B) \cap (B \backslash A) = \emptyset$.\vspace*{5px}\\
    Assume there exists $x \in (A \cap B) \cap (B \backslash A)$ for the sake of contradiction. Then $x \in A \cap B$ and $x \in B \backslash A$. Since $x \in A \cap B$, we have $x \in A$ and $x \in B$. Since $x \in B \backslash A$, we have $x \in B$ and $x \notin A$. This contradicts $x \in A$. Therefore, $(A \cap B) \cap (B \backslash A) = \emptyset$.\vspace*{10px}\\
    \textit{Part 3:} Show $(A \backslash B) \cap (B \backslash A) = \emptyset$.\vspace*{5px}\\
    Assume there exists $x \in (A \backslash B) \cap (B \backslash A)$ for the sake of contradiction. Then $x \in A \backslash B$ and $x \in B \backslash A$. Since $x \in A \backslash B$, we have $x \in A$ and $x \notin B$. Since $x \in B \backslash A$, we have $x \in B$ and $x \notin A$. This contradicts both $x \notin B$ and $x \notin A$ simultaneously. Therefore, $(A \backslash B) \cap (B \backslash A) = \emptyset$.\vspace*{10px}\\
    Therefore, $(A \cap B)$, $(A \backslash B)$, and $(B \backslash A)$ are all pairwise disjoint.
  \end{proof}
  \end{enumerate}
  
  This proves that the three subsets are a partition of $(A \cup B)$. Partitions will be very important to understanding equivalence relations.

  \item Let $A, B, C$, and $D$ be subsets of a universal set $\mathcal{U}$. Prove or disprove the following statements:
  \begin{enumerate}
  \item $(A \times B) \cap(C \times D) \subseteq(A \cap C) \times(B \cap D)$.
  \begin{proof}
    Let $(x,y) \in (A \times B) \cap(C \times D)$ be arbitrary. Then $(x,y) \in A \times B$ and $(x,y) \in C \times D$. Since $(x,y) \in A \times B$, we have $x \in A$ and $y \in B$. Since $(x,y) \in C \times D$, we have $x \in C$ and $y \in D$. Therefore, $x \in A \cap C$ and $y \in B \cap D$, so $(x,y) \in (A \cap C) \times(B \cap D)$. Hence, $(A \times B) \cap(C \times D) \subseteq(A \cap C) \times(B \cap D)$.
  \end{proof}

  \item $(A \cap C) \times(B \cap D) \subseteq(A \times B) \cap(C \times D)$.
  \begin{proof}
    Let $(x,y) \in (A \cap C) \times(B \cap D)$ be arbitrary. Then $x \in A \cap C$ and $y \in B \cap D$. Since $x \in A \cap C$, we have $x \in A$ and $x \in C$. Since $y \in B \cap D$, we have $y \in B$ and $y \in D$. Therefore, $(x,y) \in A \times B$ and $(x,y) \in C \times D$, so $(x,y) \in (A \times B) \cap(C \times D)$. Hence, $(A \cap C) \times(B \cap D) \subseteq(A \times B) \cap(C \times D)$.
  \end{proof}

  \item $(A \times B) \cup(C \times D) \subseteq(A \cup C) \times(B \cup D)$.
  \begin{proof}[Disproof]
    Let $A=\{1\}, B=\{2\}, C=\{3\}, D=\{4\}$ be subsets of a universal set $\mathcal{U}$ for the sake of counterexample. Then $(A \times B) \cup(C \times D)=\{(1,2)\} \cup \{(3,4)\}=\{(1,2),(3,4)\}$ and $(A \cup C) \times(B \cup D)=\{1,3\} \times \{2,4\}=\{(1,2),(1,4),(3,2),(3,4)\}$. Since $(1,4) \in (A \cup C) \times(B \cup D)$ but $(1,4) \notin (A \times B) \cup(C \times D)$, we have $(A \times B) \cup(C \times D) \not\subseteq (A \cup C) \times(B \cup D)$. Therefore, the statement is \textbf{false}.
  \end{proof}

  \item $(A \cup C) \times(B \cup D) \subseteq(A \times B) \cup(C \times D)$.
  \begin{proof}[Disproof]
    Let $A=\{1\}, B=\{2\}, C=\{3\}, D=\{4\}$ be subsets of a universal set $\mathcal{U}$ for the sake of counterexample. Then $(A \cup C) \times(B \cup D)=\{1,3\} \times \{2,4\}=\{(1,2),(1,4),(3,2),(3,4)\}$ and $(A \times B) \cup(C \times D)=\{(1,2)\} \cup \{(3,4)\}=\{(1,2),(3,4)\}$. Since $(1,4) \in (A \cup C) \times(B \cup D)$ but $(1,4) \notin (A \times B) \cup(C \times D)$, we have $(A \cup C) \times(B \cup D) \not\subseteq (A \times B) \cup(C \times D)$. Therefore, the statement is \textbf{false}.
  \end{proof}
  \end{enumerate}

  Likewise, Cartesian product of sets (above) and the power set of a set (below) will be very important in the next part of the course.

  \item Let $A, B$ be subsets of a universal set $\mathcal{U}$. Prove or disprove the following statements:
  \begin{enumerate}
  \item $\mathcal{P}(A \cap B) \subseteq \mathcal{P}(A) \cap \mathcal{P}(B)$.
  \begin{proof}
    Let $X \in \mathcal{P}(A \cap B)$ be arbitrary. Then $X \subseteq A \cap B$ by definition of power set. Since $X \subseteq A \cap B$, we have $X \subseteq A$ and $X \subseteq B$. Therefore, $X \in \mathcal{P}(A)$ and $X \in \mathcal{P}(B)$, so $X \in \mathcal{P}(A) \cap \mathcal{P}(B)$. Hence, $\mathcal{P}(A \cap B) \subseteq \mathcal{P}(A) \cap \mathcal{P}(B)$.
  \end{proof}

  \item $\mathcal{P}(A) \cap \mathcal{P}(B) \subseteq \mathcal{P}(A \cap B)$.
  \begin{proof}[Disproof]
    Let $A=\{1,2\}, B=\{2,3\}$ be subsets of a universal set $\mathcal{U}$ for the sake of counterexample. Then $\mathcal{P}(A)=\{\emptyset,\{1\},\{2\},\{1,2\}\}$ and $\mathcal{P}(B)=\{\emptyset,\{2\},\{3\},\{2,3\}\}$, so $\mathcal{P}(A) \cap \mathcal{P}(B)=\{\emptyset,\{2\}\}$. However, $\mathcal{P}(A \cap B)=\mathcal{P}(\{2\})=\{\emptyset,\{2\}\}$. Since $\{1\} \in \mathcal{P}(A) \cap \mathcal{P}(B)$ but $\{1\} \notin \mathcal{P}(A \cap B)$, we have $\mathcal{P}(A) \cap \mathcal{P}(B) \not\subseteq \mathcal{P}(A \cap B)$. Therefore, the statement is \textbf{false}.
  \end{proof}

  \item $\mathcal{P}(A \cup B) \subseteq \mathcal{P}(A) \cup \mathcal{P}(B)$.
  \begin{proof}[Disproof]
    Let $A=\{1\}, B=\{2\}$ be subsets of a universal set $\mathcal{U}$ for the sake of counterexample. Then $\mathcal{P}(A \cup B)=\mathcal{P}(\{1,2\})=\{\emptyset,\{1\},\{2\},\{1,2\}\}$ and $\mathcal{P}(A) \cup \mathcal{P}(B)=\{\emptyset,\{1\}\} \cup \{\emptyset,\{2\}\}=\{\emptyset,\{1\},\{2\}\}$. Since $\{1,2\} \in \mathcal{P}(A \cup B)$ but $\{1,2\} \notin \mathcal{P}(A) \cup \mathcal{P}(B)$, we have $\mathcal{P}(A \cup B) \not\subseteq \mathcal{P}(A) \cup \mathcal{P}(B)$. Therefore, the statement is \textbf{false}.
  \end{proof}

  \item $\mathcal{P}(A) \cup \mathcal{P}(B) \subseteq \mathcal{P}(A \cup B)$.
  \begin{proof}
    Let $X \in \mathcal{P}(A) \cup \mathcal{P}(B)$ be arbitrary. Then $X \in \mathcal{P}(A)$ or $X \in \mathcal{P}(B)$ by definition of union. If $X \in \mathcal{P}(A)$, then $X \subseteq A \subseteq A \cup B$, so $X \in \mathcal{P}(A \cup B)$. If $X \in \mathcal{P}(B)$, then $X \subseteq B \subseteq A \cup B$, so $X \in \mathcal{P}(A \cup B)$. In both cases, $X \in \mathcal{P}(A \cup B)$. Hence, $\mathcal{P}(A) \cup \mathcal{P}(B) \subseteq \mathcal{P}(A \cup B)$.
  \end{proof}
  \end{enumerate}



  
\end{enumerate}
