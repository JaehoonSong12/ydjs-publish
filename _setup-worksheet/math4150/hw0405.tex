  Let $n \in \setZ$ with $n>0$. Prove that
  $$
  \frac{\sigma(n!)}{n!} \geq \sum_{i=1}^n \frac{1}{i}
  $$
  \begin{define}{The sum-of-divisors function $\sigma$}[def:sigma][red]
  For $n\in\setZ_{>0}$, the sum-of-divisors function $\sigma(n)$ is defined as the sum of all positive divisors of $n$:
  \[
  \sigma(n):=\sum_{d\mid n} d.
  \]
  For powers of prime numbers, the sum of divisors formula is defined as follows.
  \[
  \sigma(n) =\sum_{d\mid p^k} d = 1 + p + p^2 + \cdots + p^k = \frac{p^{k+1}-1}{p-1} \text{ for some prime } p \text{ and } k\geq 0.
  \]
  \end{define}
  \begin{subprove}
    For any integer $i$ where $1 \le i \le n$, it is trivial that the number $i$ is a factor 
    of $n!$ by definition as follows.
    \[
    n! = 1 \cdot 2 \cdot 3 \cdots n = i \cdot (1 \cdot 2 \cdots (i-1) \cdot (i+1) \cdots n).
    \]

    Thus, for each integer $i$ where $1 \le i \le n$, the expression $\dfrac{n!}{i}$ is a 
    positive divisor of $n!$ by definition. Let $D$ be the set of such divisors and a 
    subset of the universal set, $\setUNIV = \sBUILD{x}{x\text{ is a positive divisor of }n!}$, as follows.
    \[
      D = \left\{ \frac{n!}{1}, \frac{n!}{2}, \frac{n!}{3}, \ldots, \frac{n!}{n} \right\}
    \]
    Thus, by \nameref{def:sigma}, it is trivial that
    \[
      \sum_{d \in D} d = \sum_{i=1}^n \frac{n!}{i} \leq \sum_{d \in \setUNIV} d = \sum_{d\mid n!, d>0} d =\sigma(n!)\text{ where }D \subseteq \setUNIV.
    \]
    Therefore, by the property of inequality (division by positive integer), it has been proven that
    \[
    \frac{\sigma(n!)}{n!} 
    \geq \frac{1}{n!} \left( \sum_{i=1}^n \frac{n!}{i} \right) 
    = \sum_{i=1}^n \left( \frac{1}{n!} \cdot \frac{n!}{i} \right) 
    = \sum_{i=1}^n \frac{1}{i}
    \]
    for $n \in \setZ$ with $n>0$.
  \end{subprove}

  \[
  \mu(n)=
  \begin{cases}
  1,& n=1,\\[4pt]
  0,&\text{if } p^2\mid n \text{ for some prime } p,\\[4pt]
  (-1)^r,&\text{if } n=p_1p_2\cdots p_r \text{ is a product of } r \text{ distinct primes.}
  \end{cases}
  \]

  \[
    \sum_{d \mid n, d>0} \mu(d) = 0 \text{ for } n>1 \text{ and } \sum_{d \mid 1, d>0} \mu(d) = 1.

  Let $f$ and $g$ be arithmetic functions. Then, 
  $f(n)=\sum_{d\mid n, d>0} g(d)$ for all $n\ge 1$ if and only if
  \[
  g(n)=\sum_{d\mid n, d>0}\mu(d)f(n/d)
  \]
  for all $n\ge 1$.
  \end{define}

    Let \(n=\prod_{i=1}^r p_i^{a_i}\) be the prime factorization of \(n\ge1\).
    Then, by \nameref{def:vonmangoldt},
    \[
      \sum_{d\mid n, d>0}\Lambda(d)
      =\sum_{i=1}^r\sum_{k=1}^{a_i}\Lambda(p_i^k)
      =\sum_{i=1}^r\sum_{k=1}^{a_i}\ln p_i
      =\sum_{i=1}^r a_i\ln p_i
      =\ln\!\bigg(\prod_{i=1}^r p_i^{a_i}\bigg)
      =\ln n.
    \]
    Let $f(n) = \ln n$ and $g(n) = \Lambda(n)$ for all \(n\ge1\). Then, by \nameref{thm:mobiusinv},
    \[
      \Lambda(n)=\sum_{d\mid n, d>0}\mu(d)\ln\!\bigg(\frac{n}{d}\bigg).
    \]
    Since \(\ln(n/d)=\ln n-\ln d\), we have
    \[
      \Lambda(n)
      =\ln n\sum_{d\mid n, d>0}\mu(d)-\sum_{d\mid n, d>0}\mu(d)\ln d.
    \]
    By \nameref{def:mobius}, \(\sum_{d\mid n, d>0}\mu(d)=0\) for \(n>1\) and 
    \(\ln(1)=0\). Therefore, for all \(n\ge1\),
    \[
      \Lambda(n)=-\sum_{d\mid n, d>0}\mu(d)\ln d.
    \]
  \end{subprove}

  \item Exercise Set 3.6, \#69. This exercise presents an interpretation of the Möbius Inversion Formula from the point of view of convolutions.
  \begin{define}{Convolution of Arithmetic Functions}[def:convolution]
  Let $n \in \setZ$ with $n>0$, and let $f$ and $g$ be arithmetic functions. The convolution of $f$ and $g$, denoted $f \star g$, is given by
  \[
  (f \star g)(n):=\sum_{d \mid n, d>0} f(d) g\left(\frac{n}{d}\right)
  \]
  \end{define}
  \begin{enumerate}
    \item \textbf{Commutativity:} Let $f$ and $g$ be arithmetic functions. Prove that $f \star g=g \star f$.
    \label{prob:3.6.69-a}
    \begin{subprove}
      By \nameref{def:convolution}, for any $n \in \setZ_{>0}$,
      \[
      (f \star g)(n) = \sum_{d \mid n} f(d) g(n/d).
      \]
      Since $d$ is a divisor of $n$, by definition, there exists a positive integer $d^\prime$ 
      such that $n = d^\prime\cdot d$, that is $d^\prime = n/d$. Therefore,
      \[
      (f \star g)(n) = \sum_{d' \mid n} f(n/d') g(d') = \sum_{d' \mid n} g(d') f(n/d') = (g \star f)(n).
      \]
      Therefore, $f \star g = g \star f$ for all arithmetic functions $f$ and $g$.
    \end{subprove}

    \item \textbf{Associativity:} Let $f, g$, and $h$ be arithmetic functions. 
    Prove that $(f \star g) \star h= f \star(g \star h)$.
    \label{prob:3.6.69-b}
    \begin{subprove}
      By \nameref{def:convolution}, for any $n \in \setZ_{>0}$,
      \[
      ((f\star g)\star h)(n)
      = \sum_{d\mid n} (f\star g)(d)\,h\!\left(\frac{n}{d}\right)
      = \sum_{d\mid n}\left(\sum_{c\mid d} f(c)\,g\!\left(\frac{d}{c}\right)\right)
        h\!\left(\frac{n}{d}\right).
      \]
      Since \(c\) is a divisor of \(d\) and \(d\) is a divisor of \(n\), it follows that \(c\) is a divisor of \(n\),
      and there exists a positive integer \(x, y\) such that \(n = d\cdot x = c \cdot y \cdot x\).
      Thus, 
      \[
      \sum_{c\mid n} \sum_{y\mid (n/c)} f(c)\,g(y)\,h\!\left(\frac{n}{c y}\right)
      = \sum_{c\mid n} f(c)\left(\sum_{y\mid (n/c)} g(y)\,h\!\left(\frac{(n/c)}{y}\right)\right)
      = \sum_{c\mid n} f(c)(g \star h)\left(\frac{n}{c}\right)
      = (f \star (g \star h))(n).
      \]
      Therefore, by both equalities above, 
      \((f \star g) \star h = f \star (g \star h)\) for all arithmetic functions \(f\), \(g\), and \(h\).
    \end{subprove}

    \newpage

    \item \textbf{Identity:} For $n \in \setZ$ with $n>0$, let
    \[
    \delta(n)= \begin{cases}1 & \text { if } n=1 \\ 0 & \text { otherwise }\end{cases}
    \]
    Let $f$ be an arithmetic function. Prove that $f \star \delta=\delta \star f=f$.
    \label{prob:3.6.69-c}
    \begin{subprove}
      By \nameref{def:convolution}, for any $n \in \setZ_{>0}$,
      \[
      (f \star \delta)(n) = \sum_{d \mid n} f(d) \delta(n/d) = f(n) \cdot 1 = f(n).
      \]
      since \(\delta(n/d) = 0\) for all \(n/d > 1\) and \(\delta(1) = 1\).
      By commutativity proved from problem \parenref{prob:3.6.69-a},
      \[
      \delta \star f = f \star \delta = f.
      \]
    \end{subprove}

    \item \textbf{Inverses:} For $n \in \setZ$ with $n>0$, let $\mathbf{1}(n)=1$. 
    In other words, $\mathbf{1}$ is the arithmetic function mapping every positive 
    integer to $1$. Prove that $\mu \star \mathbf{1}=\mathbf{1} \star \mu=\delta$.
    \label{prob:3.6.69-d}
    \begin{subprove}
      By \nameref{def:convolution} and \nameref{def:mobius}, for any $n \in \setZ_{>0}$,
      \[
      (\mu \star \mathbf{1})(n) = \sum_{d \mid n} \mu(d) \mathbf{1}(n/d) 
      = \sum_{d \mid n} \mu(d)
      = \delta(n)
      =\begin{cases}
        1 & \text { if } n=1 \\ 
        0 & \text { if } n>1
      \end{cases}.
      \]
      Therefore, $(\mu \star \mathbf{1})(n) = \delta(n)$ for all $n > 0$.
      By commutativity proved from problem \parenref{prob:3.6.69-a},
      \[
      \mathbf{1} \star \mu = \mu \star \mathbf{1} = \delta.
      \]
    \end{subprove}

    \item \textbf{Möbius Inversion Formula:} Prove the Möbius Inversion Formula using convolutions.\newline
    [Hint: Theorem 3.5 can be restated as $f=g \star \mathbf{1}$ if and only if $g=\mu \star f=f \star \mu$.]
    \begin{subprove}
      ($\Rightarrow$) Suppose $f = g \star \mathbf{1}$ for the sake of direct proof. 
      By problem \parenref{prob:3.6.69-a}, \parenref{prob:3.6.69-b}, \parenref{prob:3.6.69-c}, and \parenref{prob:3.6.69-d},
      \[
      f \star \mu = \mu \star f = \mu \star (g \star \mathbf{1}) = (\mu \star g) \star \mathbf{1} = g \star (\mu \star \mathbf{1}) = g \star \delta = g.
      \]
      ($\Leftarrow$) Suppose $g = \mu \star f$ for the sake of direct proof. Similarly,
      \[
      g \star \mathbf{1} = \mathbf{1} \star g = (\mathbf{1} \star \mu) \star f = \delta \star f = f.
      \]
      Therefore, by ($\Rightarrow$) and ($\Leftarrow$), $f = g \star \mathbf{1}$ if and only if $g = \mu \star f = f \star \mu$, which is \nameref{thm:mobiusinv}.
    \end{subprove}
  \end{enumerate}

\end{enumerate}

\begin{define}{Theorem 4.4 (Euler's Criterion)}[thm:4.4][red]
  Let $p$ be an odd prime and let $a\in\mathbb{Z}$ with $p\nmid a$. Then
  \[
  \left(\frac{a}{p}\right) \equiv a^{\frac{p-1}{2}} \pmod p,
  \]
  and, since both sides take values in $\{\pm1\}$, we may (and usually do) interpret this congruence as the equality
  \[
  \left(\frac{a}{p}\right) = a^{\frac{p-1}{2}} \pmod p \in\{1,-1\}.
  \]
  \begin{subprove}
    If $\left(\frac{a}{p}\right)=1$, there exists $x_0$ with $x_0^2\equiv a\pmod p$. Then
    \[
    a^{\frac{p-1}{2}} \equiv (x_0^2)^{\frac{p-1}{2}} \equiv x_0^{\,p-1}\equiv 1\pmod p
    \]
    by Fermat's little theorem.
    
    If $\left(\frac{a}{p}\right)=-1$, the nonzero residues $\{1,2,\dots,p-1\}$ partition into $(p-1)/2$ pairs with product $a$: if $i^2\equiv a\pmod p$ for some $i$, then $a$ is a residue, contradiction. Hence $(p-1)!\equiv a^{\frac{p-1}{2}} \pmod p$. By Wilson's theorem, $(p-1)!\equiv -1\pmod p$, so $a^{\frac{p-1}{2}}\equiv -1\pmod p$.
  \end{subprove}
\end{define}
\begin{define}{Lemma 4.7 (Gauss's Lemma)}[lem:4.7][red]
  Let $p$ be an odd prime and let $a\in\mathbb{Z}$ with $p\nmid a$. For $k=1,2,\dots,\frac{p-1}{2}$ let $r_k$ be the least positive residue of $ka$ modulo $p$ (i.e., $r_k\in\{1,\dots,p-1\}$ and $r_k\equiv ka\pmod p$). Let $n$ be the number of these residues $r_k$ that are greater than $p/2$. Then
  \[
  \left(\frac{a}{p}\right)=(-1)^n.
  \]
  \begin{subprove}
  Let $p$ be an odd prime and $a\in\mathbb{Z}$ with $p\nmid a$. For $k=1,2,\dots,\frac{p-1}{2}$, let $r_k$ be the least positive residue of $ka$ modulo $p$. Partition into $\{r_1,\dots,r_n\}$ (those $r_k>p/2$) and $\{s_1,\dots,s_m\}$ (those $s_k<p/2$), so $n+m=(p-1)/2$.

  Consider $p-r_1,\dots,p-r_n,s_1,\dots,s_m$. These lie in $\{1,\dots,(p-1)/2\}$ and are pairwise distinct modulo $p$: distinct $r_k,s_k$ by construction; if $p-r_i\equiv s_j\pmod p$, then $r_i+s_j\equiv 0\pmod p$ gives $(k_i+k_j)a\equiv 0\pmod p$, forcing $k_i+k_j=p$ (contradiction).

  Therefore, these form a permutation of $\{1,\dots,(p-1)/2\}$:
  \[
  \prod_{i=1}^n (p-r_i)\cdot\prod_{j=1}^m s_j \equiv \left(\frac{p-1}{2}\right)! \pmod p.
  \]
  Since $p-r_i \equiv -r_i\pmod p$ and $\{r_1,\dots,r_n,s_1,\dots,s_m\}$ equals the least positive residues of $a,2a,\dots,\frac{p-1}{2}a$:
  \[
  (-1)^n a^{\frac{p-1}{2}}\left(\frac{p-1}{2}\right)! \equiv \left(\frac{p-1}{2}\right)! \pmod p.
  \]
  Canceling the factorial yields $a^{\frac{p-1}{2}}\equiv (-1)^n\pmod p$. By \nameref{thm:4.4}, $\left(\frac{a}{p}\right)=(-1)^n$.
  \end{subprove}
\end{define}

  \begin{define}{Proposition 4.3}[prop:4.3][red]
  Let $p$ be an odd prime. There are exactly $\dfrac{p-1}{2}$ distinct
  (incongruent) quadratic residues modulo $p$ among the nonzero residue
  classes, and exactly $\dfrac{p-1}{2}$ quadratic non-residues among the
  nonzero classes.
  \begin{subprove}
  The multiplicative group $(\mathbb{Z}/p\mathbb{Z})^\times$ has order $p-1$. For each $x\in(\mathbb{Z}/p\mathbb{Z})^\times$, the square $x^2$ is a nonzero quadratic residue. Since $x^2 = (-x)^2$, each nonzero quadratic residue is obtained from exactly two distinct elements. Hence, $(p-1)/2$ distinct nonzero quadratic residues exist. The remaining $(p-1)/2$ nonzero classes are quadratic non-residues.
  \end{subprove}
  \end{define}
  
  \begin{subprove}
    Let $k$ be an integer with $1\leq k\leq \frac{p-1}{2}$. Since $k\in\{1,2,\ldots,p-1\}$ and $p$ does not divide $k$, the value $k^2$ is a quadratic residue modulo $p$.
    
    It is shown that if $k_1,k_2\in\{1,2,\ldots,\frac{p-1}{2}\}$ with $k_1\neq k_2$, then $k_1^2\not\equiv k_2^2\pmod{p}$. For if $k_1^2\equiv k_2^2\pmod{p}$, then $p\mid(k_1-k_2)(k_1+k_2)$. Since $0<k_1+k_2<p$, it is concluded that $p\mid(k_1-k_2)$. But $|k_1-k_2|\leq\frac{p-1}{2}<p$, hence $k_1=k_2$, a contradiction.

  \end{subprove}

  \[
  \left(\frac{-1}{p}\right)=(-1)^{\frac{p-1}{2}}=

  \[
  \left(\frac{-1}{p}\right)\equiv (-1)^{\frac{p-1}{2}}\pmod p.

  \begin{subprove}
    Let $S = \sum_{a=1}^{(p-1)/2}\left(\frac{a}{p}\right)$.

    \[
      \left(\frac{a}{p}\right) = \left(\frac{-1 \cdot -a}{p}\right) = \left(\frac{-1}{p}\right)\left(\frac{-a}{p}\right) = (1) \cdot \left(\frac{p-a}{p}\right) = \left(\frac{p-a}{p}\right).

    \begin{enumerate}

        \item $S_1 = S_2$
    \end{enumerate}

  \end{subprove}

  \item Exercise Set 4.3, \#34. Let $p$ and $q$ be odd prime numbers with 
  $p=q+4 a$ for some $a \in \mathbb{Z}$. Prove that
  $$
  \left(\frac{a}{p}\right)=\left(\frac{a}{q}\right).
  $$
  \begin{define}{Theorem 4.9 (Quadratic Reciprocity)}[thm:4.9][red]
  Let $p$ and $q$ be distinct odd primes. Then
  \[
  \left(\frac{p}{q}\right)\left(\frac{q}{p}\right)=(-1)^{\frac{p-1}{2}\cdot\frac{q-1}{2}}.
  \]
  Equivalently,
  \[
  \left(\frac{p}{q}\right)=\left(\frac{q}{p}\right)\quad\text{unless }p\equiv q\equiv 3\pmod 4,
  \]
  in which case
  \[
  \left(\frac{p}{q}\right)=-\left(\frac{q}{p}\right).
  \]
  \end{define}
  \begin{subprove}
    Let the prime factorization of $a$ be $a = \sigma \cdot 2^k \cdot r_1 \cdot r_2 \cdots r_m$, where $\sigma = \pm 1$ is the sign of $a$, $k \ge 0$, and each $r_i$ is an odd prime.

    By the multiplicative property of the Legendre symbol, it is found that
    \[
      \left(\frac{a}{p}\right) = \left(\frac{\sigma}{p}\right) \left(\frac{2}{p}\right)^k \prod_{i=1}^m \left(\frac{r_i}{p}\right)
      \quad \text{and} \quad
      \left(\frac{a}{q}\right) = \left(\frac{\sigma}{q}\right) \left(\frac{2}{q}\right)^k \prod_{i=1}^m \left(\frac{r_i}{q}\right).
    \]
    The proof is complete if it can be shown that each corresponding factor in these two products is equal.
    \begin{enumerate}
      \item 
      If $a$ is negative, $\sigma = -1$. It must be shown that $\left(\frac{-1}{p}\right) = \left(\frac{-1}{q}\right)$.
      It is given that $p = q + 4a$. This implies $p \equiv q \pmod 4$.
      By \nameref{thm:4.6}, the value of $\left(\frac{-1}{p}\right)$ depends only on the prime's residue modulo 4. Since $p \equiv q \pmod 4$, they must have the same residue (either both $1$ or both $3$).
      Therefore, $\left(\frac{-1}{p}\right) = \left(\frac{-1}{q}\right)$. (If $\sigma=1$, the factor is $1=1$).
      \item  
      If $a$ is even, $k > 0$. It must be shown that $\left(\frac{2}{p}\right) = \left(\frac{2}{q}\right)$.
      Since $a$ is even, $a = 2m$ for some integer $m$.
      The given equation is $p = q + 4a = q + 4(2m) = q + 8m$.
      This implies $p \equiv q \pmod 8$.
      The value of $\left(\frac{2}{p}\right)$ depends only on the prime's residue modulo 8 (it is $1$ if $p \equiv 1, 7 \pmod 8$ and $-1$ if $p \equiv 3, 5 \pmod 8$).
      Since $p \equiv q \pmod 8$, they must have the same residue.
      Therefore, $\left(\frac{2}{p}\right) = \left(\frac{2}{q}\right)$. (If $k=0$, the factor is $1=1$).
    \end{enumerate}
    It must be shown that $\left(\frac{r_i}{p}\right) = \left(\frac{r_i}{q}\right)$ for each odd prime factor $r_i$ of $a$.
    By \nameref{thm:4.9}, the symbols can be flipped:
    \[
      \left(\frac{r_i}{p}\right) = \left(\frac{p}{r_i}\right) (-1)^{\frac{p-1}{2} \cdot \frac{r_i-1}{2}}
    \]
    \[
      \left(\frac{r_i}{q}\right) = \left(\frac{q}{r_i}\right) (-1)^{\frac{q-1}{2} \cdot \frac{r_i-1}{2}}
    \]
    It must be shown that the expressions on the right-hand side are equal.
    \begin{itemize}
        \item 
        It is given that $p = q + 4a$. Since $r_i$ is a factor of $a$, it is found that $a \equiv 0 \pmod{r_i}$.
        Thus, $p = q + 4(0) \pmod{r_i}$, which means $p \equiv q \pmod{r_i}$.
        By the periodicity of the Legendre symbol in its top argument, $\left(\frac{p}{r_i}\right) = \left(\frac{q}{r_i}\right)$.
        \item 
        By part (1), it is already shown that $p \equiv q \pmod 4$.
        Therefore, $\frac{p-1}{2}$ and $\frac{q-1}{2}$ have the same parity.
        Therefore, the exponents $\frac{p-1}{2} \cdot \frac{r_i-1}{2}$ and $\frac{q-1}{2} \cdot \frac{r_i-1}{2}$ have the same parity.
        This implies $(-1)^{\frac{p-1}{2} \cdot \frac{r_i-1}{2}} = (-1)^{\frac{q-1}{2} \cdot \frac{r_i-1}{2}}$.
    \end{itemize}
    Since both parts of the right-hand side are equal for $p$ and $q$, it is found that $\left(\frac{r_i}{p}\right) = \left(\frac{r_i}{q}\right)$.
    It has been shown that $\left(\frac{\sigma}{p}\right) = \left(\frac{\sigma}{q}\right)$, $\left(\frac{2}{p}\right)^k = \left(\frac{2}{q}\right)^k$, and $\left(\frac{r_i}{p}\right) = \left(\frac{r_i}{q}\right)$ for all $i$.
    Since all corresponding factors are equal, the products must be equal.
    Therefore, $\left(\frac{a}{p}\right)=\left(\frac{a}{q}\right)$.
  \end{subprove}
