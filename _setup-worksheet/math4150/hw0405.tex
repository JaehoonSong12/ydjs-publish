\begin{define}{Arithmetic Function}[def:arithmeticfunction][red]
  An arithmetic function is a function $f:\setZ_{>0}\to\setC$ that maps 
  positive integers to complex numbers (or real numbers or integers).
  \begin{itemize}
    \item An arithmetic function $f$ is called \textbf{multiplicative} if
    \[
    f(mn)=f(m)f(n) \quad \text{whenever} \quad \gcd(m,n)=1.
    \]
    \item An arithmetic function $f$ is called \textbf{completely multiplicative} if
    \[
    f(mn)=f(m)f(n) \quad \text{for all} \quad m,n\in\setZ_{>0}.
    \]
  \end{itemize}
\end{define}

\begin{enumerate}
  \item Let $f$ be a multiplicative arithmetic function. Show that $F(n)=\sum_{d \mid n} f(d)$ is multiplicative.
  \begin{subprove}
    Let $f$ be multiplicative by \nameref{def:arithmeticfunction}. 
    For $m, n \in \setZ_{>0}$ with $\gcd(m,n) = 1$, it is shown that $F(mn) = F(m)F(n)$.
    Since $\gcd(m,n) = 1$, each divisor $d$ of $mn$ is uniquely expressed as 
    $d = d_1 d_2$ where $d_1 \mid m$, $d_2 \mid n$, and $\gcd(d_1, d_2) = 1$.
    Therefore,
    \[
    F(mn) = \sum_{d \mid mn} f(d) = \sum_{\substack{d_1 \mid m \\ d_2 \mid n}} f(d_1 d_2).
    \]
    Since $f$ is multiplicative and $\gcd(d_1, d_2) = 1$, it follows that $f(d_1 d_2) = f(d_1) f(d_2)$. Therefore
    \[
    F(mn) = \sum_{\substack{d_1 \mid m \\ d_2 \mid n}} f(d_1) f(d_2) = \left(\sum_{d_1 \mid m} f(d_1)\right) \left(\sum_{d_2 \mid n} f(d_2)\right) = F(m) F(n).
    \]
    Hence, $F$ is multiplicative.
  \end{subprove}
  
  \item For integers $k \geq 1$, and multiplicative function $f$, show that the function $g(n):=f\left(n^{k}\right)$ is multiplicative.
  \begin{subprove}
    Let $f$ be multiplicative by \nameref{def:arithmeticfunction}. 
    For $m, n \in \setZ_{>0}$ with $\gcd(m, n) = 1$, it is shown that $g(mn) = g(m)g(n)$.
    Since $\gcd(m, n) = 1$, it follows that $\gcd(m^k, n^k) = 1$ (raising to a power preserves coprimality).
    Therefore,
    \[
    g(mn) = f((mn)^k) = f(m^k n^k).
    \]
    Since $\gcd(m^k, n^k) = 1$ and $f$ is multiplicative, it follows that $f(m^k n^k) = f(m^k) f(n^k) = g(m) g(n)$.
    Hence, $g$ is multiplicative.
  \end{subprove}

  \item Prove that the von Mangoldt function $\Lambda$ is not multiplicative.
  \begin{define}{Von Mangoldt Function $\Lambda$}[def:vonmangoldt][red]
    For $n\in\setZ_{>0}$, the von Mangoldt function $\Lambda:\setZ_{>0}\to\setR$ is defined by
    \[
    \Lambda(n):= \begin{cases}\ln p & \text{if } n=p^{a} \text{ for some } p \text{ prime and } a \in \setZ_{>0} \\ 0 & \text{otherwise}\end{cases}
    \]
    \textbf{Note}: $\Lambda(n)$ is a \textbf{Prime-Power Detector}.
  \end{define}
  \begin{subprove}
    By \nameref{def:arithmeticfunction}, a function $f$ is multiplicative if $f(mn) = f(m)f(n)$ whenever $\gcd(m,n) = 1$.
    A counterexample is provided to show that $\Lambda$ does not satisfy this property.
    Let $m = 2$ and $n = 3$. Then $\gcd(2,3) = 1$.
    Since $2 = 2^1$ and $3 = 3^1$ are prime powers, it follows that $\Lambda(2) = \ln 2$ and $\Lambda(3) = \ln 3$.
    Since $6 = 2 \cdot 3$ is not a prime power, it follows that $\Lambda(6) = 0$.
    Therefore,
    \[
    \Lambda(2) \cdot \Lambda(3) = \ln 2 \cdot \ln 3 \neq 0 = \Lambda(6).
    \]
    Since $\Lambda(6) \neq \Lambda(2)\Lambda(3)$ and $\gcd(2,3) = 1$, it is shown that $\Lambda$ is not multiplicative.
  \end{subprove}
  \item Define the Möbius function $\mu(n)$. Show that $\mu$ is multiplicative, but not completely multiplicative.
  \begin{define}{Möbius function $\mu$}[def:mobius][red]
    For $n\in\setZ_{>0}$, the Möbius function $\mu:\setZ_{>0}\to\{-1,0,1\}$ is defined by
    \[
    \mu(n)=
    \begin{cases}
    1,& n=1,\\[4pt]
    0,&\text{if } p^2\mid n \text{ for some prime } p,\\[4pt]
    (-1)^r,&\text{if } n=p_1p_2\cdots p_r \text{ is a product of } r \text{ distinct primes.}
    \end{cases}
    \]
    The Möbius function has the following key property:
    \[
    \sum_{d \mid n, d>0} \mu(d) = \begin{cases}
    1, & \text{if } n=1,\\
    0, & \text{if } n>1.
    \end{cases}
    \]
    \textbf{Note}: $\mu(n)$ is a \textbf{Square-Free Detector}.
  \end{define}
  \begin{subprove}
    The Möbius function $\mu$ is defined by \nameref{def:mobius}. 
    It is shown that $\mu$ is multiplicative but not completely multiplicative by \nameref{def:arithmeticfunction}.
    
    For $m, n \in \setZ_{>0}$ with $\gcd(m,n) = 1$, it is shown that $\mu(mn) = \mu(m)\mu(n)$.
    If $m = 1$ or $n = 1$, then without loss of generality, $m = 1$, so $\mu(mn) = \mu(n) = \mu(1)\mu(n) = \mu(m)\mu(n)$.
    If $p^2 \mid m$ for some prime $p$, then $p^2 \mid mn$ (since $\gcd(m,n) = 1$), so $\mu(mn) = 0 = \mu(m) = \mu(m)\mu(n)$.
    If both $m$ and $n$ are square-free, then $m = p_1 \cdots p_r$ and $n = q_1 \cdots q_s$ where all primes are distinct (since $\gcd(m,n) = 1$).
    Therefore, $mn = p_1 \cdots p_r q_1 \cdots q_s$ is a product of $r+s$ distinct primes, so $\mu(mn) = (-1)^{r+s} = (-1)^r (-1)^s = \mu(m)\mu(n)$.
    Hence, $\mu$ is multiplicative.
    
    A counterexample is provided to show that $\mu$ is not completely multiplicative.
    Let $m = 2$ and $n = 2$. Since $2$ is a single prime, it follows that $\mu(2) = -1$ and $\mu(2) \cdot \mu(2) = 1$.
    Since $2^2 \mid 4$, it follows that $\mu(4) = 0$.
    Therefore,
    \[
    \mu(4) = 0 \neq 1 = \mu(2)\mu(2).
    \]
    Since $\mu(4) \neq \mu(2)\mu(2)$, it is shown that $\mu$ is not completely multiplicative.
  \end{subprove}
  \item Evaluate $\sum_{d \mid n, d>0} \mu(d)$ for all integers $n \geq 1$.
  \begin{subanswer}
    The Möbius function $\mu$ is defined by \nameref{def:mobius}. 
    Since $\mu$ is multiplicative (as shown above) by \nameref{def:arithmeticfunction}, it follows that $F(n) = \sum_{d \mid n, d>0} \mu(d)$ is multiplicative.
    For $n = 1$, it follows that $\sum_{d \mid 1, d>0} \mu(d) = \mu(1) = 1$.
    For any prime power $p^k$ with $k \geq 1$, it follows that
    \[
    F(p^k) = \sum_{d \mid p^k} \mu(d) = \mu(1) + \mu(p) = 1 + (-1) = 0.
    \]
    Since any $n > 1$ has at least one prime factor, it follows that $F(n) = 0$ for all $n > 1$.
    Therefore,
    \[
    \sum_{d \mid n, d>0} \mu(d) = \begin{cases}
    1, & \text{if } n = 1,\\
    0, & \text{if } n > 1.
    \end{cases}
    \]
  \end{subanswer}

  \begin{define}{Möbius Inversion Formula*}[thm:mobiusinv][red]
    Let $f$ and $g$ be arithmetic functions. Then
    \[
    f(n)=\sum_{d\mid n, d>0} g(d) \quad \text{for all } n\ge 1
    \]
    if and only if
    \[
    g(n)=\sum_{d\mid n, d>0}\mu(d)f(n/d) \quad \text{for all } n\ge 1.
    \]
    \begin{subprove}
      ($\Rightarrow$) Assume $f(n)=\sum_{d\mid n, d>0} g(d)$ for all $n\ge 1$. Then
      \begin{align*}
        \sum_{d\mid n, d>0}\mu(d)f(n/d) &= \sum_{d\mid n, d>0}\mu(d)\sum_{e\mid (n/d), e>0}g(e)\\
        &= \sum_{d\mid n, d>0}\sum_{e\mid (n/d), e>0}\mu(d)g(e).
      \end{align*}
      Let $de = k$, so $k\mid n$ and $d\mid k$. Then this becomes
      \[
      \sum_{k\mid n, k>0}g(k)\sum_{d\mid k, d>0}\mu(d).
      \]
      By the key property of the Möbius function, $\sum_{d\mid k, d>0}\mu(d) = 1$ if $k=1$ and $0$ if $k>1$. Therefore, the sum equals $g(n)$.
      
      ($\Leftarrow$) Assume $g(n)=\sum_{d\mid n, d>0}\mu(d)f(n/d)$ for all $n\ge 1$. Define $F(n) = \sum_{d\mid n, d>0}f(d)$ for all $n\ge 1$.
      
      Then
      \begin{align*}
        \sum_{d\mid n, d>0}g(d) &= \sum_{d\mid n, d>0}\sum_{e\mid d, e>0}\mu(e)f(d/e)\\
        &= \sum_{e\mid n, e>0}\mu(e)\sum_{d\mid n, e\mid d, d>0}f(d/e).
      \end{align*}
      Let $k = d/e$, so $k\mid (n/e)$ and $ke = d\mid n$. Then this becomes
      \[
      \sum_{e\mid n, e>0}\mu(e)\sum_{k\mid (n/e), k>0}f(k) = \sum_{e\mid n, e>0}\mu(e)F(n/e),
      \]
      where $F(n/e) = \sum_{k\mid (n/e), k>0}f(k)$.
      
      By the forward direction we already proved, since $F(n) = \sum_{d\mid n, d>0}f(d)$ for all $n\ge 1$, we have $f(n) = \sum_{d\mid n, d>0}\mu(d)F(n/d)$ for all $n\ge 1$. Therefore:
      \[
      \sum_{e\mid n, e>0}\mu(e)F(n/e) = f(n).
      \]
      Therefore, $\sum_{d\mid n, d>0}g(d) = f(n)$.
    \end{subprove}
  \end{define}

  \item Show that $\sum_{d \mid n} \phi(d)=n$ for all integers $n \geq 1$.
  \begin{define}{Euler's Totient Function $\phi$}[def:euler_totient][red]
    For $n \in \mathbb{Z}_{>0}$, Euler's totient function $\phi(n)$ is defined as the number of positive integers less than or equal to $n$ that are relatively prime to $n$:
    \[
    \phi(n) := \#\{\,k \in \mathbb{Z}_{>0} : 1 \le k \le n \text{ and } \gcd(k, n) = 1\,\}.
    \]
    If $n$ has the prime factorization $n = p_1^{e_1} p_2^{e_2} \cdots p_r^{e_r}$, then Euler's totient function is given by
    \[
    \phi(n) = n \prod_{i=1}^{r} \left(1 - \frac{1}{p_i}\right).
    \]
  \end{define}  
  \begin{subprove}
    The Euler totient function $\phi$ is defined by \nameref{def:euler_totient}. 
    For each positive integer $k$ with $1 \leq k \leq n$, let $d = \gcd(k, n)$. Then $d \mid n$ and $k = d \cdot k'$ where $\gcd(k', n/d) = 1$ and $1 \leq k' \leq n/d$.
    Conversely, for each divisor $d$ of $n$ and each positive integer $k'$ with $1 \leq k' \leq n/d$ and $\gcd(k', n/d) = 1$, the integer $k = dk'$ satisfies $1 \leq k \leq n$ and $\gcd(k, n) = d$.
    Therefore, the integers $1, 2, \ldots, n$ are partitioned into disjoint sets, where each set corresponds to a divisor $d$ of $n$ and contains exactly $\phi(n/d)$ integers $k$ with $\gcd(k, n) = d$.
    Since there are $n$ total integers, it follows that
    \[
    n = \sum_{d \mid n} \phi(n/d) = \sum_{d \mid n} \phi(d),
    \]
    where the last equality follows by reindexing (since $d \mapsto n/d$ is a bijection on the divisors of $n$).
    
    Alternatively, this is verified using \nameref{thm:mobiusinv}. 
    Let $f(n) = n$ and $g(d) = \phi(d)$. Then $f(n) = \sum_{d \mid n} g(d) = \sum_{d \mid n} \phi(d)$.
    By \nameref{thm:mobiusinv}, it follows that
    \[
    g(n) = \sum_{d \mid n} \mu(d) f(n/d) = \sum_{d \mid n} \mu(d) \cdot \frac{n}{d} = n \sum_{d \mid n} \frac{\mu(d)}{d}.
    \]
    Since $\sum_{d \mid n} \frac{\mu(d)}{d} = \frac{\phi(n)}{n}$ for all $n \geq 1$, it follows that $g(n) = n \cdot \frac{\phi(n)}{n} = \phi(n)$, confirming that $f(n) = \sum_{d \mid n} \phi(d) = n$.
    Hence, $\sum_{d \mid n} \phi(d) = n$ for all integers $n \geq 1$.
  \end{subprove}

  \item Evaluate the number divisor function $v(450), v(105), v\left(p^{k}\right)$, 
  for prime $p$ and non-negative integer $k$.
  \begin{define}{Divisor-Count Function $v(n)$}[def:vfunction][red]
    For $n\in\setZ_{>0}$, the divisor-count function $v(n)$ is defined as the number of positive divisors of $n$:
    \[
    v(n):=\#\{d\in\setZ_{>0}:d\mid n\}.
    \]
    For powers of prime numbers, the divisor-count function is given by:
    \[
    v(p^k)=k+1 \text{ for prime } p \text{ and } k\geq 0.
    \]
    In particular, for a prime $p$, we have $v(p)=2$.
  \end{define}
  \begin{subanswer}
    The divisor-count function $v$ is defined by \nameref{def:vfunction}. 
    Since $v(n) = \sum_{d \mid n, d>0} 1$ and the constant function $1$ is multiplicative, it follows that $v$ is multiplicative by \nameref{def:arithmeticfunction}.
    For a prime power $p^k$, it follows that $v(p^k) = k+1$ (the positive divisors of $p^k$ are $1, p, p^2, \ldots, p^k$).
    For $450 = 2 \cdot 3^2 \cdot 5^2$, since $v$ is multiplicative and $\gcd(2, 3^2, 5^2) = 1$, it follows that
    \[
    v(450) = v(2) \cdot v(3^2) \cdot v(5^2) = v(2^1) \cdot v(3^2) \cdot v(5^2) = (1+1) \cdot (2+1) \cdot (2+1) = 2 \cdot 3 \cdot 3 = 18.
    \]
    For $105 = 3 \cdot 5 \cdot 7$, since $v$ is multiplicative and $\gcd(3, 5, 7) = 1$, it follows that
    \[
    v(105) = v(3) \cdot v(5) \cdot v(7) = v(3^1) \cdot v(5^1) \cdot v(7^1) = (1+1) \cdot (1+1) \cdot (1+1) = 2 \cdot 2 \cdot 2 = 8.
    \]
    Therefore,
    \begin{align*}
    v(450) &= 18,\\
    v(105) &= 8,\\
    v(p^k) &= k+1 \quad \text{for prime } p \text{ and non-negative integer } k.
    \end{align*}
  \end{subanswer}

  \item Show that $\sum_{d \mid n, d>0} \frac{1}{d}=\frac{\sigma(n)}{n}$ for all integers $n \geq 1$.
  \begin{define}{Sum-of-Divisors Function $\sigma(n)$}[def:sigma][red]
    For $n\in\setZ_{>0}$, the sum-of-divisors function $\sigma(n)$ is defined as the sum of all positive divisors of $n$:
    \[
    \sigma(n):=\sum_{d\mid n, d>0} d.
    \]
    For powers of prime numbers, the sum of divisors formula is given by:
    \[
    \sigma(p^k)=\sum_{d\mid p^k, d>0} d = 1 + p + p^2 + \cdots + p^k = \frac{p^{k+1}-1}{p-1} \text{ for prime } p \text{ and } k\geq 0.
    \]
    In particular, for a prime $p$, we have $\sigma(p)=1+p=p+1$.
  \end{define}
  \begin{subprove}
    The sum-of-divisors function $\sigma$ is defined by \nameref{def:sigma}. 
    It is shown that $\sum_{d \mid n, d>0} \frac{1}{d} = \frac{\sigma(n)}{n}$ for all $n \in \setZ_{>0}$.
    By the definition of $\sigma$, it follows that
    \[
    \sigma(n) = \sum_{d \mid n, d>0} d.
    \]
    Dividing both sides by $n$, it follows that
    \[
    \frac{\sigma(n)}{n} = \frac{1}{n} \sum_{d \mid n, d>0} d = \sum_{d \mid n, d>0} \frac{d}{n}.
    \]
    Since the map $d \mapsto n/d$ is a bijection on the divisors of $n$, it follows that
    \[
    \frac{\sigma(n)}{n} = \sum_{d \mid n, d>0} \frac{d}{n} = \sum_{d \mid n, d>0} \frac{1}{n/d} = \sum_{e \mid n, e>0} \frac{1}{e},
    \]
    where reindexing is performed by letting $e = n/d$.
    Therefore,
    \[
    \sum_{d \mid n, d>0} \frac{1}{d} = \frac{\sigma(n)}{n}
    \]
    for all $n \in \setZ_{>0}$.
  \end{subprove}

  \item Evaluate $\sum_{d \mid n, d>0} \mu(d) v(n / d)$, where $v$ is the 
  number of divisors function and $n=4^{3} \cdot 7^{2}$.
  \begin{define}{Convolution of Arithmetic Functions}[def:convolution][red]
    For $n\in\setZ_{>0}$ and arithmetic functions $f$ and $g$, the convolution of $f$ and $g$, denoted $f \star g$, is defined by
    \[
    (f \star g)(n):=\sum_{d \mid n, d>0} f(d) g\left(\frac{n}{d}\right).
    \]
    The convolution operation is commutative: $f \star g = g \star f$ for all arithmetic functions $f$ and $g$.
  \end{define}
  \begin{subanswer}
    The convolution is defined by \nameref{def:convolution}, so it follows that $\sum_{d \mid n, d>0} \mu(d) v(n/d) = (\mu \star v)(n)$.
    The divisor-count function $v$ is defined by \nameref{def:vfunction}, so it follows that $v(n) = \sum_{d \mid n, d>0} 1$.
    Let $f(n) = v(n)$ and $g(d) = 1$ for all $d \geq 1$. Then $f(n) = \sum_{d \mid n, d>0} g(d) = \sum_{d \mid n, d>0} 1 = v(n)$.
    By \nameref{thm:mobiusinv}, it follows that
    \[
    g(n) = \sum_{d \mid n, d>0} \mu(d) f(n/d) = \sum_{d \mid n, d>0} \mu(d) v(n/d).
    \]
    Since $g(n) = 1$ for all $n \geq 1$, it follows that $\sum_{d \mid n, d>0} \mu(d) v(n/d) = 1$ for all $n \in \setZ_{>0}$.
    Therefore, for $n = 4^3 \cdot 7^2 = 64 \cdot 49 = 3136$, it follows that
    \[
    \sum_{d \mid 3136, d>0} \mu(d) v(3136/d) = 1.
    \]
  \end{subanswer}

  \item If $a$ is a quadratic residue modulo two distinct odd primes $p$ and $q$, 
  then the equation $x^{2} \equiv a \pmod{pq}$ is solvable.
  \begin{define}{Quadratic Residue}[def:quadres][red]
    For $a\in\setZ$ and $n\in\setZ_{>0}$, $a$ is a \textbf{quadratic residue modulo $n$} if there exists $x\in\setZ$ such that
    \[
    x^2 \equiv a \pmod{n}.
    \]
    If no such $x$ exists, then $a$ is a \textbf{quadratic non-residue modulo $n$}.
    For an odd prime $p$, there are exactly $\frac{p-1}{2}$ non-zero quadratic residues modulo $p$ among the integers $1, 2, \ldots, p-1$.
  \end{define}
  \begin{subprove}
    A quadratic residue is defined by \nameref{def:quadres}. 
    Since $a$ is a quadratic residue modulo $p$, there exists $x_1 \in \setZ$ such that $x_1^2 \equiv a \pmod{p}$.
    Since $a$ is a quadratic residue modulo $q$, there exists $x_2 \in \setZ$ such that $x_2^2 \equiv a \pmod{q}$.
    Since $p$ and $q$ are distinct odd primes, it follows that $\gcd(p, q) = 1$. 
    By the Chinese Remainder Theorem, there exists a unique solution $x$ modulo $pq$ to the system
    \[
    \begin{cases}
    x \equiv x_1 \pmod{p},\\
    x \equiv x_2 \pmod{q}.
    \end{cases}
    \]
    Since $x \equiv x_1 \pmod{p}$, it follows that $x^2 \equiv x_1^2 \equiv a \pmod{p}$.
    Since $x \equiv x_2 \pmod{q}$, it follows that $x^2 \equiv x_2^2 \equiv a \pmod{q}$.
    Since $x^2 \equiv a \pmod{p}$ and $x^2 \equiv a \pmod{q}$, and $\gcd(p, q) = 1$, by the Chinese Remainder Theorem, it follows that $x^2 \equiv a \pmod{pq}$.
    Hence, the equation $x^2 \equiv a \pmod{pq}$ is solvable.
  \end{subprove}
  
  \item Find all incongruent quadratic residues modulo 13. Do the same for the modulus 17, and the modulus 23.
  \begin{subanswer}
    A quadratic residue is defined by \nameref{def:quadres}. 
    For an odd prime $p$, the quadratic residues modulo $p$ are the squares of the integers $1, 2, \ldots, \frac{p-1}{2}$ modulo $p$.
    For $p = 13$, since $\frac{13-1}{2} = 6$, it follows that
    \begin{align*}
      1^2 &\equiv 1, \quad 2^2 \equiv 4, \quad 3^2 \equiv 9, \quad 4^2 \equiv 3, \quad 5^2 \equiv 12, \quad 6^2 \equiv 10 \pmod{13}.

  \begin{define}{Theorem 4.4 (Euler's Criterion)*}[thm:4.4][red]
    Let $p$ be an odd prime and let $a\in\setZ$ with $p\nmid a$. Then
    \[
    \left(\frac{a}{p}\right) \equiv a^{\frac{p-1}{2}} \pmod p,
    \]
    and, since both sides take values in $\{\pm1\}$, we may (and usually do) interpret this congruence as the equality
    \[
    \left(\frac{a}{p}\right) = a^{\frac{p-1}{2}} \pmod p \in\{1,-1\}.
    \]
    \begin{subprove}
      If $\left(\frac{a}{p}\right)=1$, there exists $x_0$ with $x_0^2\equiv a\pmod p$. Then
      \[
      a^{\frac{p-1}{2}} \equiv (x_0^2)^{\frac{p-1}{2}} \equiv x_0^{\,p-1}\equiv 1\pmod p
      \]
      by Fermat's little theorem.
      If $\left(\frac{a}{p}\right)=-1$, the nonzero residues $\{1,2,\dots,p-1\}$ partition into $(p-1)/2$ pairs with product $a$: if $i^2\equiv a\pmod p$ for some $i$, then $a$ is a residue, contradiction. Hence $(p-1)!\equiv a^{\frac{p-1}{2}} \pmod p$. By Wilson's theorem, $(p-1)!\equiv -1\pmod p$, so $a^{\frac{p-1}{2}}\equiv -1\pmod p$.
    \end{subprove}
  \end{define}

  \begin{define}{Lemma 4.7 (Gauss's Lemma)*}[lem:4.7][red]
    Let $p$ be an odd prime and let $a\in\setZ$ with $p\nmid a$. For $k=1,2,\dots,\frac{p-1}{2}$ let $r_k$ be the least positive residue of $ka$ modulo $p$ (i.e., $r_k\in\{1,\dots,p-1\}$ and $r_k\equiv ka\pmod p$). Let $n$ be the number of these residues $r_k$ that are greater than $p/2$. Then
    \[
    \left(\frac{a}{p}\right)=(-1)^n.
    \]
    \begin{subprove}
    Let $p$ be an odd prime and $a\in\setZ$ with $p\nmid a$. For $k=1,2,\dots,\frac{p-1}{2}$, let $r_k$ be the least positive residue of $ka$ modulo $p$. Partition into $\{r_1,\dots,r_n\}$ (those $r_k>p/2$) and $\{s_1,\dots,s_m\}$ (those $s_k<p/2$), so $n+m=(p-1)/2$.
  
    Consider $p-r_1,\dots,p-r_n,s_1,\dots,s_m$. These lie in $\{1,\dots,(p-1)/2\}$ and are pairwise distinct modulo $p$: distinct $r_k,s_k$ by construction; if $p-r_i\equiv s_j\pmod p$, then $r_i+s_j\equiv 0\pmod p$ gives $(k_i+k_j)a\equiv 0\pmod p$, forcing $k_i+k_j=p$ (contradiction).
  
    Therefore, these form a permutation of $\{1,\dots,(p-1)/2\}$:
    \[
    \prod_{i=1}^n (p-r_i)\cdot\prod_{j=1}^m s_j \equiv \left(\frac{p-1}{2}\right)! \pmod p.
    \]
    Since $p-r_i \equiv -r_i\pmod p$ and $\{r_1,\dots,r_n,s_1,\dots,s_m\}$ equals the least positive residues of $a,2a,\dots,\frac{p-1}{2}a$:
    \[
    (-1)^n a^{\frac{p-1}{2}}\left(\frac{p-1}{2}\right)! \equiv \left(\frac{p-1}{2}\right)! \pmod p.
    \]
    Canceling the factorial yields $a^{\frac{p-1}{2}}\equiv (-1)^n\pmod p$. By \nameref{thm:4.4}, $\left(\frac{a}{p}\right)=(-1)^n$.
    \end{subprove}
  \end{define}

  \begin{define}{Theorem 4.9 (Quadratic Reciprocity)*}[thm:4.9][red]
    Let $p$ and $q$ be distinct odd primes. Then
    \[
    \left(\frac{p}{q}\right)\left(\frac{q}{p}\right)=(-1)^{\frac{p-1}{2}\cdot\frac{q-1}{2}}.
    \]
    Equivalently,
    \[
    \left(\frac{p}{q}\right)=\left(\frac{q}{p}\right)\quad\text{unless }p\equiv q\equiv 3\pmod 4,
    \]
    in which case
    \[
    \left(\frac{p}{q}\right)=-\left(\frac{q}{p}\right).
    \]
    \begin{subprove}
      The Legendre symbol is defined by \nameref{def:legendre}.
      For distinct odd primes $p$ and $q$, let $N_1=\sum_{j=1}^{(q-1)/2}\lfloor jp/q\rfloor$ and $N_2=\sum_{j=1}^{(p-1)/2}\lfloor jq/p\rfloor$.
      By \nameref{lem:4.7}, it follows that the number of least positive residues of $jp \pmod{q}$ for $j=1,2,\ldots,(q-1)/2$ that are greater than $q/2$ equals $N_1$, so $\left(\frac{p}{q}\right)=(-1)^{N_1}$.
      Similarly, it follows that $\left(\frac{q}{p}\right)=(-1)^{N_2}$.
      A lattice-point count on $[0,(p-1)/2]\times[0,(q-1)/2]$ divided by $y=(q/p)x$ yields $N_1+N_2=(p-1)/2\cdot(q-1)/2$.
      Therefore, $\left(\frac{p}{q}\right)\left(\frac{q}{p}\right)=(-1)^{N_1+N_2}=(-1)^{\frac{p-1}{2}\cdot\frac{q-1}{2}}$.
    \end{subprove}
  \end{define}

  Let $n \in \setZ$ with $n>0$. Von Mangoldt's function, denoted $\Lambda(n)$, is given by
  \[
  \Lambda(n):= \begin{cases}\ln p & \text { if } n=p^a \text { for some } p \text { prime and } a \in \setZ \\ 0 & \text { otherwise }\end{cases}
  \]
  \end{define}
  Prove that
  \[
  \Lambda(n)=-\sum_{d \mid n, d>0} \mu(d) \ln d
  \]
  \begin{define}{The Möbius function $\mu$}[def:mobius][red]
  The Möbius function $\mu:\setZ_{>0}\to\{-1,0,1\}$ is defined by
  \[
  \mu(n)=
  \begin{cases}
  1,& n=1,\\[4pt]
  0,&\text{if } p^2\mid n \text{ for some prime } p,\\[4pt]
  (-1)^r,&\text{if } n=p_1p_2\cdots p_r \text{ is a product of } r \text{ distinct primes.}
  \end{cases}
  \]
  where 
  \[
    \sum_{d \mid n, d>0} \mu(d) = 0 \text{ for } n>1 \text{ and } \sum_{d \mid 1, d>0} \mu(d) = 1.
  \]
  \end{define}
  \begin{define}{Möbius Inversion Formula}[thm:mobiusinv][red]
  Let $f$ and $g$ be arithmetic functions. Then, 
  $f(n)=\sum_{d\mid n, d>0} g(d)$ for all $n\ge 1$ if and only if
  \[
  g(n)=\sum_{d\mid n, d>0}\mu(d)f(n/d)
  \]
  for all $n\ge 1$.
  \end{define}
  \begin{subprove}
    Let \(n=\prod_{i=1}^r p_i^{a_i}\) be the prime factorization of \(n\ge1\).
    Then, by \nameref{def:vonmangoldt},
    \[
      \sum_{d\mid n, d>0}\Lambda(d)
      =\sum_{i=1}^r\sum_{k=1}^{a_i}\Lambda(p_i^k)
      =\sum_{i=1}^r\sum_{k=1}^{a_i}\ln p_i
      =\sum_{i=1}^r a_i\ln p_i
      =\ln\!\bigg(\prod_{i=1}^r p_i^{a_i}\bigg)
      =\ln n.
    \]
    Let $f(n) = \ln n$ and $g(n) = \Lambda(n)$ for all \(n\ge1\). Then, by \nameref{thm:mobiusinv},
    \[
      \Lambda(n)=\sum_{d\mid n, d>0}\mu(d)\ln\!\bigg(\frac{n}{d}\bigg).
    \]
    Since \(\ln(n/d)=\ln n-\ln d\), we have
    \[
      \Lambda(n)
      =\ln n\sum_{d\mid n, d>0}\mu(d)-\sum_{d\mid n, d>0}\mu(d)\ln d.
    \]
    By \nameref{def:mobius}, \(\sum_{d\mid n, d>0}\mu(d)=0\) for \(n>1\) and 
    \(\ln(1)=0\). Therefore, for all \(n\ge1\),
    \[
      \Lambda(n)=-\sum_{d\mid n, d>0}\mu(d)\ln d.
    \]
  \end{subprove}

  \item Exercise Set 3.6, \#69. This exercise presents an interpretation of the Möbius Inversion Formula from the point of view of convolutions.
  \begin{define}{Convolution of Arithmetic Functions}[def:convolution]
  Let $n \in \setZ$ with $n>0$, and let $f$ and $g$ be arithmetic functions. The convolution of $f$ and $g$, denoted $f \star g$, is given by
  \[
  (f \star g)(n):=\sum_{d \mid n, d>0} f(d) g\left(\frac{n}{d}\right)
  \]
  \end{define}
  \begin{enumerate}
    \item \textbf{Commutativity:} Let $f$ and $g$ be arithmetic functions. Prove that $f \star g=g \star f$.
    \label{prob:3.6.69-a}
    \begin{subprove}
      By \nameref{def:convolution}, for any $n \in \setZ_{>0}$,
      \[
      (f \star g)(n) = \sum_{d \mid n} f(d) g(n/d).
      \]
      Since $d$ is a divisor of $n$, by definition, there exists a positive integer $d^\prime$ 
      such that $n = d^\prime\cdot d$, that is $d^\prime = n/d$. Therefore,
      \[
      (f \star g)(n) = \sum_{d' \mid n} f(n/d') g(d') = \sum_{d' \mid n} g(d') f(n/d') = (g \star f)(n).
      \]
      Therefore, $f \star g = g \star f$ for all arithmetic functions $f$ and $g$.
    \end{subprove}

    \item \textbf{Associativity:} Let $f, g$, and $h$ be arithmetic functions. 
    Prove that $(f \star g) \star h= f \star(g \star h)$.
    \label{prob:3.6.69-b}
    \begin{subprove}
      By \nameref{def:convolution}, for any $n \in \setZ_{>0}$,
      \[
      ((f\star g)\star h)(n)
      = \sum_{d\mid n} (f\star g)(d)\,h\!\left(\frac{n}{d}\right)
      = \sum_{d\mid n}\left(\sum_{c\mid d} f(c)\,g\!\left(\frac{d}{c}\right)\right)
        h\!\left(\frac{n}{d}\right).
      \]
      Since \(c\) is a divisor of \(d\) and \(d\) is a divisor of \(n\), it follows that \(c\) is a divisor of \(n\),
      and there exists a positive integer \(x, y\) such that \(n = d\cdot x = c \cdot y \cdot x\).
      Thus, 
      \[
      \sum_{c\mid n} \sum_{y\mid (n/c)} f(c)\,g(y)\,h\!\left(\frac{n}{c y}\right)
      = \sum_{c\mid n} f(c)\left(\sum_{y\mid (n/c)} g(y)\,h\!\left(\frac{(n/c)}{y}\right)\right)
      = \sum_{c\mid n} f(c)(g \star h)\left(\frac{n}{c}\right)
      = (f \star (g \star h))(n).
      \]
      Therefore, by both equalities above, 
      \((f \star g) \star h = f \star (g \star h)\) for all arithmetic functions \(f\), \(g\), and \(h\).
    \end{subprove}

    \newpage

    \item \textbf{Identity:} For $n \in \setZ$ with $n>0$, let
    \[
    \delta(n)= \begin{cases}1 & \text { if } n=1 \\ 0 & \text { otherwise }\end{cases}
    \]
    Let $f$ be an arithmetic function. Prove that $f \star \delta=\delta \star f=f$.
    \label{prob:3.6.69-c}
    \begin{subprove}
      By \nameref{def:convolution}, for any $n \in \setZ_{>0}$,
      \[
      (f \star \delta)(n) = \sum_{d \mid n} f(d) \delta(n/d) = f(n) \cdot 1 = f(n).
      \]
      since \(\delta(n/d) = 0\) for all \(n/d > 1\) and \(\delta(1) = 1\).
      By commutativity proved from problem \parenref{prob:3.6.69-a},
      \[
      \delta \star f = f \star \delta = f.
      \]
    \end{subprove}

    \item \textbf{Inverses:} For $n \in \setZ$ with $n>0$, let $\mathbf{1}(n)=1$. 
    In other words, $\mathbf{1}$ is the arithmetic function mapping every positive 
    integer to $1$. Prove that $\mu \star \mathbf{1}=\mathbf{1} \star \mu=\delta$.
    \label{prob:3.6.69-d}
    \begin{subprove}
      By \nameref{def:convolution} and \nameref{def:mobius}, for any $n \in \setZ_{>0}$,
      \[
      (\mu \star \mathbf{1})(n) = \sum_{d \mid n} \mu(d) \mathbf{1}(n/d) 
      = \sum_{d \mid n} \mu(d)
      = \delta(n)
      =\begin{cases}
        1 & \text { if } n=1 \\ 
        0 & \text { if } n>1
      \end{cases}.
      \]
      Therefore, $(\mu \star \mathbf{1})(n) = \delta(n)$ for all $n > 0$.
      By commutativity proved from problem \parenref{prob:3.6.69-a},
      \[
      \mathbf{1} \star \mu = \mu \star \mathbf{1} = \delta.
      \]
    \end{subprove}

    \item \textbf{Möbius Inversion Formula:} Prove the Möbius Inversion Formula using convolutions.\newline
    [Hint: Theorem 3.5 can be restated as $f=g \star \mathbf{1}$ if and only if $g=\mu \star f=f \star \mu$.]
    \begin{subprove}
      ($\Rightarrow$) Suppose $f = g \star \mathbf{1}$ for the sake of direct proof. 
      By problem \parenref{prob:3.6.69-a}, \parenref{prob:3.6.69-b}, \parenref{prob:3.6.69-c}, and \parenref{prob:3.6.69-d},
      \[
      f \star \mu = \mu \star f = \mu \star (g \star \mathbf{1}) = (\mu \star g) \star \mathbf{1} = g \star (\mu \star \mathbf{1}) = g \star \delta = g.
      \]
      ($\Leftarrow$) Suppose $g = \mu \star f$ for the sake of direct proof. Similarly,
      \[
      g \star \mathbf{1} = \mathbf{1} \star g = (\mathbf{1} \star \mu) \star f = \delta \star f = f.
      \]
      Therefore, by ($\Rightarrow$) and ($\Leftarrow$), $f = g \star \mathbf{1}$ if and only if $g = \mu \star f = f \star \mu$, which is \nameref{thm:mobiusinv}.
    \end{subprove}
  \end{enumerate}

\end{enumerate}
