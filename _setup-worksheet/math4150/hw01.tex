  \item Exercise Set 1.1, \#6.
  \begin{enumerate}
    \item Let $a,b, c\in \Z$ with $c\neq 0$. Show that $a\divides b$ if and only if $ac \divides bc$.
    \begin{subprove}
      ($\Rightarrow$) Assume $a\mid b$. 
      \begin{define}{Divisibility}
      Let $a,b\in\Z$. We say \emph{$a$ divides $b$}, and write $a\divides b$, if there exists an 
      integer $c\in\Z$ such that
      \[
      b = a c.
      \]
      If no such integer $c$ exists we write $a\nmid b$.
      \end{define}
      By definition of divisibility there exists $k\in\mathbb{Z}$ such that
      \[
        b = a k
      \]
      that is,
      \[
        bc = (a k)c = (ac)k.
      \]
      Since $k\in\mathbb{Z}$, by definition of divisibility, $ac\mid bc$. 
      Hence, $a\divides b$ only if $ac \divides bc$\vspace*{10px}\\
      ($\Leftarrow$) Assume $ac\mid bc$. Then there exists $t\in\mathbb{Z}$ with
      \[
        bc = (ac)t = at\cdot c
      \]
      that is, by division,
      \[
        b = a t
      \]
      since $c\neq 0$.
      Thus $a\mid b$. Hence, $a\divides b$ if $ac \divides bc$.\vspace*{10px}\\
      Therefore, by ($\Rightarrow$) and ($\Leftarrow$), $a\divides b$ if and only if $ac \divides bc$ for $c\neq 0$.
    \end{subprove}

    \item Provide a counterexample showing that the conclusion above does not hold if $c = 0$.
    \begin{subanswer}
      Take $a=2, b=1, c=0$. Then $ac = 0$ and $bc=0$, so $ac\mid bc$ is true by 
      multiplicative identity, $0=0\cdot k$ for every integer $k$.
      However,  $a\divides b$ is false, $a\nmid b$, since there is no integer $k$ such that $1=2k$. 
      Thus, the conclusion above does not hold if $c = 0$.
    \end{subanswer}
  \end{enumerate}

  \item Exercise Set 1.1, \#9. Let $a, m, n$ be positive integers, with $a > 1$. Show that $a^{m}-1 \divides a^{n}-1$ if and only if $m\divides n$.
  \begin{subprove}
    ($\Rightarrow$) Assume $a^{m}-1\mid a^{n}-1$. By the Division Algorithm there exist integers $q\ge 0$ and $r$ with $0\le r<m$ such that
    \[
      n = qm + r.
    \]
    Then, using elementary algebra,
    \[
      a^{n}-1 = a^{qm+r}-1 = a^{r}\bigl(a^{qm}-1\bigr) + \bigl(a^{r}-1\bigr).
    \]
    Since $a^{qm}-1 = (a^{m}-1)\bigl(1+a^{m}+\cdots+a^{m(q-1)}\bigr)$ when $q\ge 1$ (and is $0$ when $q=0$), it follows that the remainder upon division of $a^{n}-1$ by $a^{m}-1$ is $a^{r}-1$. By hypothesis $a^{m}-1\mid a^{n}-1$, hence we must have $a^{r}-1=0$. Thus $r=0$ because $a>1$. Therefore $n=qm$ and so $m\mid n$.\vspace*{10px}\\
    ($\Leftarrow$) Assume $m\divides n$. Then there exists $k\in\setZ_{>0}$ such that
    \[
      n = m k.
    \]
    where the integer $k\ge 2$, by the geometric series (trivial algebra),
    \[
      a^{n}-1 = a^{mk}-1 = \bigl(a^{m}\bigr)^{k}-1
        = (a^{m}-1)\bigl(a^{m(k-1)}+a^{m(k-2)}+\cdots + a^{m(1)}+a^{m(0)}\bigr).
    \]
    By the Closure Property of Integers, $\bigl(a^{m(k-1)}+a^{m(k-2)}+\cdots + a^{m(1)}+a^{m(0)}\bigr)\in\setZ$, 
    hence $a^{m}-1\mid a^{n}-1$. Thus, $a^{m}-1 \divides a^{n}-1$ if $m\divides n$.\vspace*{10px}\\
    Therefore, by ($\Rightarrow$) and ($\Leftarrow$), $a^{m}-1 \divides a^{n}-1$ if and only if $m\divides n$.
  \end{subprove}

  \item Exercise Set 1.1, \#10. Let $n\in \Z$.
  \begin{enumerate}
    \item Prove that $3\divides n^{3}-n$
    \begin{subprove}
      Factor
      \[
        n^{3}-n = n(n-1)(n+1).
      \]
      These are three consecutive integers; among any three consecutive integers, exactly one is a multiple of $3$. Hence $3\mid n(n-1)(n+1)$, i.e. $3\mid (n^{3}-n)$.
    \end{subprove}
    \item Prove that $5\divides n^{5}-n$
    \begin{subprove}
      We argue modulo $5$. Every integer $n$ is congruent to one of $0,1,2,3,4$ mod $5$. Compute
      \[
        0^{5}-0\equiv0,\quad 1^{5}-1\equiv0,\quad 2^{5}-2=30\equiv0,\quad 3^{5}-3=240\equiv0,\quad 4^{5}-4=1020\equiv0\pmod 5.
      \]
      Hence $n^{5}-n\equiv 0\pmod 5$ in all cases, so $5\mid(n^{5}-n)$.
    \end{subprove}
    \item Is it true that $4 \divides n^{4}-n$? Provide a proof or a counterexample.
    \begin{subanswer}
      It is \emph{not} true in general. Take $n=2$. Then $n^{4}-n=16-2=14$, and $4\nmid14$. (For completeness, if $n\equiv0,1\pmod 4$ then $n^{4}-n\equiv0$, while if $n\equiv2,3\pmod 4$ then $n^{4}-n\equiv2$.)
    \end{subanswer}
  \end{enumerate}

  \item Exercise Set 1.2, \#28. Let $a,n$ be positive integers, with $a > 1$. Show that 
  if $a^{n}+1$ is prime, then $a$ is even, and $n$ is a power of $2$.
  \begin{subprove}
      Suppose $a^{n}+1$ is prime. First, $a$ must be even. Indeed, if $a$ were odd then $a^{n}$ is odd, so $a^{n}+1$ is even and greater than $2$, hence composite. Thus $a$ is even.
      
      Next, write $n=2^{t}u$ with $u\ge 1$ odd. If $u=1$ we are done. If $u>1$, set $x:=a^{2^{t}}$. Then $x$ is an integer and
      \[
        a^{n}+1 = a^{2^{t}u}+1 = x^{u}+1.
      \]
      Since $u$ is odd, we have the factorization
      \[
        x^{u}+1 = (x+1)\bigl(x^{u-1}-x^{u-2}+\cdots - x + 1\bigr),
      \]
      so $a^{n}+1$ is composite. This contradicts primality. Therefore $u=1$ and $n$ is a power of $2$.
  \end{subprove}

  \item Let $n > 1$ be an integer. Prove that, if $n^{2} + 1$ is a prime number, then 
  $n^{2} + 1$ is expressible in the form $4k + 1$ with integer $k$.
  \begin{subprove}
      For any integer $n$, $n^{2}\equiv 0$ or $1\pmod 4$ according as $n$ is even or odd. If $n$ is odd then $n^{2}+1\equiv 2\pmod 4$, which is even and greater than $2$, hence not prime. Thus $n$ must be even, so $n^{2}\equiv0\pmod 4$ and consequently $n^{2}+1\equiv1\pmod 4$. Hence there exists $k\in\Z$ with $n^{2}+1=4k+1$.
  \end{subprove}

  \item Exercise Set 1.3, \#43.
  \begin{enumerate}
    \item Let $a,b,c \in \Z$ with $(a,b)=1$. If $a \divides c$ and $b\divides c$ prove that $ab\divides c$.
    \begin{subprove}
      Since $a\mid c$, write $c=a s$ with $s\in\Z$. Also $b\mid c$, so $b\mid as$. Because $(a,b)=1$, by Euclid's Lemma we have $b\mid s$. Thus $s=bt$ for some $t\in\Z$, and hence $c=as=a(bt)=ab\,t$. Therefore $ab\mid c$.
    \end{subprove}
    \item Provide a counterexample to show why the statement in (a) cannot hold if $(a,b)>1$.
    \begin{subanswer}
      Take $a=b=2$ and $c=2$. Then $a\mid c$ and $b\mid c$, but $ab=4\nmid 2$.
    \end{subanswer}
    \item Let $a_1, \ldots, a_n \in \Z$ be pairwise relatively prime numbers. Prove that if each $a_j \divides c$, then $a_1 \cdots a_n \divides c$.
    \begin{subprove}
      Proceed by induction on $n\ge 2$. The case $n=2$ is part (a). For the inductive step, assume the statement holds for $n-1$ pairwise relatively prime integers. Let $A:=a_1\cdots a_{n-1}$. By pairwise coprimality we have $(A,a_n)=1$, and by hypothesis $A\mid c$ and $a_n\mid c$. Applying part (a) to $A$ and $a_n$ yields $A a_n\mid c$, i.e. $a_1\cdots a_n\mid c$.
    \end{subprove}
  \end{enumerate}

  \item Exercise Set 1.3, \#48. Let $a,b\in \Z$. Show that $(a,b)=1$ if and only if $(a+b,ab) = 1$.
  \begin{subprove}
    ($\Rightarrow$) Assume $(a, b)=1$. Suppose, for the sake of contradiction, that 
    $d=(a+b, a b)$ where $d>1$ then there exists a prime $p$ divides $d$ by Well-Ordering 
    Principle Lemma.
    \begin{define}{Well-Ordering Principle Lemma}
    Every integer $n>1$ has at least one prime divisor.
    \end{define}
    Thus, by the transitive property of divisibility,
    $$
    p \divides(a+b) \quad \text { and } \quad p \divides a b
    $$
    since $p \divides d$, $d \divides \group{a+b}$, and $d \divides \group{a\cdot b}$.
    \begin{define}{Euclid's Lemma}
    Let $a,b,p\in\mathds{Z}$ with $p$ prime. If $p\divides ab$, then $p\divides a$ or $p\divides b$.
    \end{define}
    Then, by Euclid's Lemma, from $p \divides ab$ we have $p \divides a$ or $p \divides b$.  
    Without loss of generality and by the definition of divisibility, there exists $i\in\setZ$ such that 
    $$
    a = p i.
    $$
    Also, by assumption of $p\divides(a+b)$, there exists $j\in\setZ$ such that $a+b = pj$. Thus,
    $$
    b = pj - pi = p(j-i),
    $$
    so $p\divides b$ as well since it is trivial that $j-i\in\setZ$ by the Closure Property of Integers, 
    which means $p$ is a common prime divisor of $a$ and $b$.
    It contradicts the assumption $(a,b) = p \ne 1$.
    Therefore, $(a,b)=1$ only if $(a+b,ab) = 1$.\vspace*{10px}\\
    ($\Leftarrow$) Assume $(a+b, a b)=1$. Suppose, for the sake of contradiction, 
    that $(a, b)=g>1$. By the Well-Ordering Principle Lemma, there exists a prime 
    $p$ with $p \divides g$. 
    Since $g=(a, b)$ divides both $a$ and $b$, by transitivity of divisibility,
    $$
    p\divides a \text { and } p \divides b .
    $$
    Hence, trivially, $p \divides a b$ and $p \divides(a+b)$, which means $p$ is a common 
    prime divisor of $a b$ and $a+b$. It contradicts the assumption $(a+b, a b) = p \ne1$.
    Therefore, $(a,b)=1$ if $(a+b,ab) = 1$.\vspace*{10px}\\
    Thus, by ($\Rightarrow$) and ($\Leftarrow$), $(a,b)=1$ if and only if $(a+b,ab) = 1$.
  \end{subprove}

  \item Exercise Set 1.4, \#54(c). Use the Euclidean Algorithm for this exercise. Compute the 
  gcd $(441,1155)$, and express it as an integral linear combination of $441$ and $1155$.
  \begin{subanswer}
    By Euclidean algorithm,
    \[
      \begin{aligned}
        1155 &= 441\cdot 2 + 273,\\
        441 &= 273\cdot 1 + 168,\\
        273 &= 168\cdot 1 + 105,\\
        168 &= 105\cdot 1 + 63,\\
        105 &= 63\cdot 1 + 42,\\
        63  &= 42\cdot 1 + 21,\\
        42  &= 21\cdot 2 + 0.
      \end{aligned}
    \]
    Hence $\gcd(441,1155)=21$. Then, by back-substitution,
    \[
      21 = 21\cdot 441 - 8\cdot 1155.
    \]
    Thus $21 = (21)\,441 + (-8)\,1155$, which is a linear combination with Bézout coefficients.
  \end{subanswer}


  \item Exercise Set 1.4, \#56. Find two rational numbers with denominators $11$ and $13$, whose 
  sum is $\frac{7}{143}$.
  \begin{subanswer}
    Let $x,y\in\setZ$ be arbitrary, for the sake of definition of rational numbers, then construct 
    the following equation,
    $$\dfrac{x}{11}+\dfrac{y}{13}=\dfrac{7}{143}.$$
    Then, since $13\times11=143$, it is equivalent to 
    $$13x+11y=7.$$ 
    Take $x=9$, $y=-10$ since $117-110=7$. Hence, the two such rational numbers are as follows.
    \[
      \group{\frac{9}{11}}+\group{-\frac{10}{13}}=\frac{7}{143}.
    \]
  \end{subanswer}
\end{enumerate}
