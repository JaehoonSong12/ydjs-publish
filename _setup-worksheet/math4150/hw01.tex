  \begin{enumerate}
    \item Let $a,b, c\in \Z$ with $c\neq 0$. Show that $a\divides b$ if and only if $ac \divides bc$.
    \begin{subprove}
      ($\Rightarrow$) Assume $a\divides b$. 
      \begin{define}[redcolor]{Divisibility}
      Let $a,b\in\Z$. We say \emph{$a$ divides $b$}, and write $a\divides b$, if there exists an 
      integer $c\in\Z$ such that
      \[
      b = a c.
      \]
      If no such integer $c$ exists we write $a\nmid b$.
      \end{define}
      By definition of divisibility there exists $k\in\mathbb{Z}$ such that
      \[
        b = a k
      \]
      that is,
      \[
        bc = (a k)c = (ac)k.
      \]
      Since $k\in\mathbb{Z}$, by definition of divisibility, $ac\divides bc$. 
      Hence, $a\divides b$ only if $ac \divides bc$\vspace*{10px}\\
      ($\Leftarrow$) Assume $ac\divides bc$. Then there exists $t\in\mathbb{Z}$ with
      \[
        bc = (ac)t = at\cdot c
      \]
      that is, by division,
      \[
        b = a t
      \]
      since $c\neq 0$.
      Thus $a\divides b$. Hence, $a\divides b$ if $ac \divides bc$.\vspace*{10px}\\
      Therefore, by ($\Rightarrow$) and ($\Leftarrow$), $a\divides b$ if and only if $ac \divides bc$ for $c\neq 0$.
    \end{subprove}

    \item Provide a counterexample showing that the conclusion above does not hold if $c = 0$.
    \begin{subanswer}
      Take $a=2, b=1, c=0$. Then $ac = 0$ and $bc=0$, so $ac\divides bc$ is true by 
      multiplicative identity, $0=0\cdot k$ for every integer $k$.
      However,  $a\divides b$ is false, $a\nmid b$, since there is no integer $k$ such that $1=2k$. 
      Thus, the conclusion above does not hold if $c = 0$.
    \end{subanswer}
  \end{enumerate}

  \item Exercise Set 1.1, \#9. Let $a, m, n$ be positive integers, with $a > 1$. Show that $a^{m}-1 \divides a^{n}-1$ if and only if $m\divides n$.
  \begin{subprove}
    ($\Rightarrow$) Assume $a^{m}-1\divides a^{n}-1$. By the Division Algorithm,
    \begin{define}[redcolor]{Division Algorithm}
    Let $a,b\in\Z$ with $b>0$. Then there exist unique integers $q$ and $r$ such that
    \[
    a = b q + r \qquad\text{and}\qquad 0\le r < b.
    \]
    Here $q$ is called the \emph{quotient} and $r$ the \emph{remainder}.
    \end{define}
    there exist integers $q\ge 0$ and $r$ with $0\le r<m$ such that
    \[
      n = qm + r.
    \]
    Then, by the geometric series (trivial algebra),
    \[
      a^{n}-1 = a^{qm+r}-1 = a^{r}\bigl({a^{m}}^{q}-1\bigr) + \bigl(a^{r}-1\bigr)= a^{r}(a^{m}-1)\bigl(1+a^{m}+\cdots+a^{m(q-1)}\bigr) + \bigl(a^{r}-1\bigr).
    \]
    It follows that the remainder upon division of $a^{n}-1$ by $a^{m}-1$ is $a^{r}-1$ where $a^{r}-1 < a^{m}-1$. 
    By hypothesis 
    $a^{m}-1\divides a^{n}-1$, $a^{r}-1=0$. 
    Thus, $r=0$ since $a>1$. Moreover, $n=qm$, so $m\divides n$.
    Thus, $a^{m}-1 \divides a^{n}-1$ only if $m\divides n$.\vspace*{10px}\\
    ($\Leftarrow$) Assume $m\divides n$. Then there exists $k\in\setZ_{>0}$ such that
    \[
      n = m k.
    \]
    where the integer $k\ge 2$, by the geometric series (trivial algebra),
    \[
      a^{n}-1 = a^{mk}-1 = \bigl(a^{m}\bigr)^{k}-1
        = (a^{m}-1)\bigl(a^{m(k-1)}+a^{m(k-2)}+\cdots + a^{m(1)}+a^{m(0)}\bigr).
    \]
    By the Closure Property of Integers, $\bigl(a^{m(k-1)}+a^{m(k-2)}+\cdots + a^{m(1)}+a^{m(0)}\bigr)\in\setZ$, 
    hence $a^{m}-1\divides a^{n}-1$. 
    Thus, $a^{m}-1 \divides a^{n}-1$ if $m\divides n$.\vspace*{10px}\\
    Therefore, by ($\Rightarrow$) and ($\Leftarrow$), $a^{m}-1 \divides a^{n}-1$ if and only if $m\divides n$.
  \end{subprove}

  \item Exercise Set 1.3, \#43.
  \begin{enumerate}
    \item Let $a,b,c \in \Z$ with $(a,b)=1$. If $a \divides c$ and $b\divides c$ prove that $ab\divides c$.
    \begin{subprove}
      Since $a\divides c$, there exists $s\in\Z$ such that $c=a s$. Thus, $b\divides as$. Since $(a,b)=1$, $a \nmid b$, and
      \begin{define}[redcolor]{Euclid's Lemma}
      Let $a,b,p\in\mathds{Z}$ with $p$ prime. If $p\divides ab$, then $p\divides a$ or $p\divides b$.
      \end{define}
      by Euclid's Lemma, $b\divides s$. Thus, $s=bt$ for some $t\in\Z$, and hence $c=as=ab\,t$. Therefore, $ab\divides c$.
    \end{subprove}
    \item Provide a counterexample to show why the statement in (a) cannot hold if $(a,b)>1$.
    \begin{subanswer}
      Take $a=b=2$ and $c=2$. Then $a\divides c$ and $b\divides c$, but $ab=4\nmid 2$. 
      This is a counterexample. (false)
    \end{subanswer}
    \item Let $a_1, \ldots, a_n \in \Z$ be pairwise relatively prime numbers. Prove that if each $a_j \divides c$, then $a_1 \cdots a_n \divides c$.
    \begin{subprove}
      It is trivial to say \inlinequote{$a \divides c$, then $a \divides c$} since it is a 
      tautology, for the sake of inductive proof, base step is true. (base step)\vspace*{10px}\\
      Suppose, for the sake of inductive proof (inductive step), the statement holds for $n-1$ 
      pairwise relatively prime integers.
      Let $b=a_1\cdots a_{n-1}$ satisfying the $n-1$ relative primes and $c=b\cdot a_n = a_1\cdots a_{n}$ assuming $a_n$ is another 
      pairwise prime with $b$, for the sake of direct proof. 
      Then, by the definition of pairwise prime numbers, we have $(b,a_n)=1$, and by all the 
      assumptions, $b\divides c$ and $a_n\divides c$. This satisfies the condition of part (a). 
      Thus, by the part (a), $b a_n\divides c$. In other words, $a_1\cdots a_{n}\divides c$.\vspace*{10px}\\
      Thus, by (base step) and (inductive step), for all $n\in\setZ$, if each $a_j \divides c$, then $a_1 \cdots a_n \divides c$.
    \end{subprove}
  \end{enumerate}

  \item Exercise Set 1.3, \#48. Let $a,b\in \Z$. Show that $(a,b)=1$ if and only if $(a+b,ab) = 1$.
  \begin{subprove}
    ($\Rightarrow$) Assume $(a, b)=1$. Suppose, for the sake of contradiction, that 
    $d=(a+b, a b)$ where $d>1$ then there exists a prime $p$ divides $d$ by Well-Ordering 
    Principle Lemma.
    \begin{define}[redcolor]{Well-Ordering Principle Lemma}
    Every integer $n>1$ has at least one prime divisor.
    \end{define}
    Thus, by the transitive property of divisibility,
    $$
    p \divides(a+b) \quad \text { and } \quad p \divides a b
    $$
    since $p \divides d$, $d \divides \group{a+b}$, and $d \divides \group{a\cdot b}$.
    \begin{define}[redcolor]{Euclid's Lemma}
    Let $a,b,p\in\mathds{Z}$ with $p$ prime. If $p\divides ab$, then $p\divides a$ or $p\divides b$.
    \end{define}
    Then, by Euclid's Lemma, from $p \divides ab$ we have $p \divides a$ or $p \divides b$.  
    Without loss of generality and by the definition of divisibility, there exists $i\in\setZ$ such that 
    $$
    a = p i.
    $$
    Also, by assumption of $p\divides(a+b)$, there exists $j\in\setZ$ such that $a+b = pj$. Thus,
    $$
    b = pj - pi = p(j-i),
    $$
    so $p\divides b$ as well since it is trivial that $j-i\in\setZ$ by the Closure Property of Integers, 
    which means $p$ is a common prime divisor of $a$ and $b$.
    It contradicts the assumption $(a,b) = p \ne 1$.
    Therefore, $(a,b)=1$ only if $(a+b,ab) = 1$.\vspace*{10px}\\
    ($\Leftarrow$) Assume $(a+b, a b)=1$. Suppose, for the sake of contradiction, 
    that $(a, b)=g>1$. By the Well-Ordering Principle Lemma, there exists a prime 
    $p$ with $p \divides g$. 
    Since $g=(a, b)$ divides both $a$ and $b$, by transitivity of divisibility,
    $$
    p\divides a \text { and } p \divides b .
    $$
    Hence, trivially, $p \divides a b$ and $p \divides(a+b)$, which means $p$ is a common 
    prime divisor of $a b$ and $a+b$. It contradicts the assumption $(a+b, a b) = p \ne1$.
    Therefore, $(a,b)=1$ if $(a+b,ab) = 1$.\vspace*{10px}\\
    Thus, by ($\Rightarrow$) and ($\Leftarrow$), $(a,b)=1$ if and only if $(a+b,ab) = 1$.
  \end{subprove}

  \item Exercise Set 1.4, \#54(c). Use the Euclidean Algorithm for this exercise. Compute the 
  gcd $(441,1155)$, and express it as an integral linear combination of $441$ and $1155$.
  \begin{subanswer}
    By Euclidean algorithm,
    \[
      \begin{aligned}
        1155 &= 441\cdot 2 + 273,\\
        441 &= 273\cdot 1 + 168,\\
        273 &= 168\cdot 1 + 105,\\
        168 &= 105\cdot 1 + 63,\\
        105 &= 63\cdot 1 + 42,\\
        63  &= 42\cdot 1 + 21,\\
        42  &= 21\cdot 2 + 0.
      \end{aligned}
    \]
    Hence $\gcd(441,1155)=21$. Then, by back-substitution,
    \[
      21 = 21\cdot 441 - 8\cdot 1155.
    \]
    Thus $21 = (21)\,441 + (-8)\,1155$, which is a linear combination with Bézout coefficients.
  \end{subanswer}


  \item Exercise Set 1.4, \#56. Find two rational numbers with denominators $11$ and $13$, whose 
  sum is $\frac{7}{143}$.
  \begin{subanswer}
    Let $x,y\in\setZ$ be arbitrary, for the sake of definition of rational numbers, then construct 
    the following equation,
    $$\dfrac{x}{11}+\dfrac{y}{13}=\dfrac{7}{143}.$$
    Then, since $13\times11=143$, it is equivalent to 
    $$13x+11y=7.$$ 
    Take $x=9$, $y=-10$ since $117-110=7$. Hence, the two such rational numbers are as follows.
    \[
      \group{\frac{9}{11}}+\group{-\frac{10}{13}}=\frac{7}{143}.
    \]
  \end{subanswer}
