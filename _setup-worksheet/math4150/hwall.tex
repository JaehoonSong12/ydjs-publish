  \begin{enumerate}
    \item Let $a,b, c\in \Z$ with $c\neq 0$. Show that $a\divides b$ if and only if $ac \divides bc$.
    \begin{subprove}
      ($\Rightarrow$) Assume $a\divides b$. 
      \begin{define}{Divisibility}[def:divisibility][redcolor]
      Let $a,b\in\Z$. We say \emph{$a$ divides $b$}, and write $a\divides b$, if there exists an 
      integer $c\in\Z$ such that
      \[
      b = a c.
      \]
      If no such integer $c$ exists we write $a\nmid b$.
      \end{define}
      By definition of divisibility, there exists $k\in\setZ$ such that
      \[
        b = a k
      \]
      that is,
      \[
        bc = (a k)c = (ac)k.
      \]
      Since $k\in\setZ$, by definition of divisibility, $ac\divides bc$. 
      Hence, $a\divides b$ only if $ac \divides bc$\vspace*{10px}\\
      ($\Leftarrow$) Assume $ac\divides bc$. Then there exists $t\in\setZ$ with
      \[
        bc = (ac)t = at\cdot c
      \]
      that is, by division,
      \[
        b = a t
      \]
      since $c\neq 0$.
      Thus $a\divides b$. Hence, $a\divides b$ if $ac \divides bc$.\vspace*{10px}\\
      Therefore, by ($\Rightarrow$) and ($\Leftarrow$), $a\divides b$ if and only if $ac \divides bc$ for $c\neq 0$.
    \end{subprove}

    \item Provide a counterexample showing that the conclusion above does not hold if $c = 0$.
    \begin{subanswer}
      Take $a=2, b=1, c=0$. Then $ac = 0$ and $bc=0$, so $ac\divides bc$ is true by 
      multiplicative identity, $0=0\cdot k$ for every integer $k$.
      However,  $a\divides b$ is false, $a\nmid b$, since there is no integer $k$ such that $1=2k$. 
      Thus, the conclusion above does not hold if $c = 0$.
    \end{subanswer}
  \end{enumerate}

  \item Exercise Set 1.1, \#9. Let $a, m, n$ be positive integers, with $a > 1$. Show that $a^{m}-1 \divides a^{n}-1$ if and only if $m\divides n$.
  \begin{subprove}
    ($\Rightarrow$) Assume $a^{m}-1\divides a^{n}-1$. By the Division Algorithm,
    \begin{define}{Division Algorithm}[algo:division][redcolor]
    Let $a,b\in\Z$ with $b>0$. Then there exist unique integers $q$ and $r$ such that
    \[
    a = b q + r \qquad\text{and}\qquad 0\le r < b.
    \]
    Here $q$ is called the \emph{quotient} and $r$ the \emph{remainder}.
    \end{define}
    there exist integers $q\ge 0$ and $r$ with $0\le r<m$ such that
    \[
      n = qm + r.
    \]
    Then, by the geometric series (trivial algebra),
    \[
      a^{n}-1 = a^{qm+r}-1 = a^{r}\bigl({a^{m}}^{q}-1\bigr) + \bigl(a^{r}-1\bigr)= a^{r}(a^{m}-1)\bigl(1+a^{m}+\cdots+a^{m(q-1)}\bigr) + \bigl(a^{r}-1\bigr).
    \]
    It follows that the remainder upon division of $a^{n}-1$ by $a^{m}-1$ is $a^{r}-1$ where $a^{r}-1 < a^{m}-1$. 
    By hypothesis 
    $a^{m}-1\divides a^{n}-1$, $a^{r}-1=0$. 
    Thus, $r=0$ since $a>1$. Moreover, $n=qm$, so $m\divides n$.
    Thus, $a^{m}-1 \divides a^{n}-1$ only if $m\divides n$.\vspace*{10px}\\
    ($\Leftarrow$) Assume $m\divides n$. Then there exists $k\in\setZ_{>0}$ such that
    \[
      n = m k.
    \]
    where the integer $k\ge 2$, by the geometric series (trivial algebra),
    \[
      a^{n}-1 = a^{mk}-1 = \bigl(a^{m}\bigr)^{k}-1
        = (a^{m}-1)\bigl(a^{m(k-1)}+a^{m(k-2)}+\cdots + a^{m(1)}+a^{m(0)}\bigr).
    \]
    By the Closure Property of Integers, $\bigl(a^{m(k-1)}+a^{m(k-2)}+\cdots + a^{m(1)}+a^{m(0)}\bigr)\in\setZ$, 
    hence $a^{m}-1\divides a^{n}-1$. 
    Thus, $a^{m}-1 \divides a^{n}-1$ if $m\divides n$.\vspace*{10px}\\
    Therefore, by ($\Rightarrow$) and ($\Leftarrow$), $a^{m}-1 \divides a^{n}-1$ if and only if $m\divides n$.
  \end{subprove}

  \item Exercise Set 1.1, \#10. Let $n\in \Z$.
  \begin{enumerate}
    \item Prove that $3\divides n^{3}-n$.
    \begin{subprove}
      By algebra,
      \[
        n^{3}-n = (n-1)(n)(n+1).
      \]
      By the Division Algorithm with divisor $3$, there exist integers $q\in\setZ$ and $r$ with $0\le r<3$ such that
      \[
        n = 3q + r.
      \] 
      This only includes $n = 3q$, $n = 3q + 1$, and $n = 3q + 2$.
      Without loss of generality, 
      \[
        n^{3}-n = (n-1)(n)(n+1)=(3q-1)(3q)(3q+1)=3(3q-1)(q)(3q+1).
      \]
      It shows at least one of the three consecutive integers has $3$ as its factor.
      Thus, $3\divides n^{3}-n$.
    \end{subprove}
    \item Prove that $5\divides n^{5}-n$.
    \begin{subprove}
      By algebra,
      \[
      n^{5}-n = (n-1)(n)(n+1)(n^{2}+1).
      \]
      By the Division Algorithm with divisor $5$, there exist integers $q\in\setZ$ and $r$ with $0\le r<5$ such that
      \[
      n = 5q + r.
      \]
      Thus, for each $r\in\{0,1,2,3,4\}$, \\
      If $r=0$ then $5\mid n$, so $5\mid n^{5}-n$.\\
      If $r=1$ then $5\mid n-1$, so $5\mid n^{5}-n$.\\
      If $r=2$ then $n^{2}+1=(5q+2)^{2}+1=5(5q^{2}+4q+1)$, so $5\mid n^{2}+1$ and therefore $5\mid n^{5}-n$.\\
      If $r=3$ then $n^{2}+1=(5q+3)^{2}+1=5(5q^{2}+6q+2)$, so $5\mid n^{2}+1$ and therefore $5\mid n^{5}-n$.\\
      If $r=4$ then $n+1=5q+5=5(q+1)$, so $5\mid n+1$ and hence $5\mid n^{5}-n$.\\
      In every condition, the conclusion, $5\mid n^{5}-n$, has been consistent. Therefore, $5\mid n^{5}-n$.
    \end{subprove}
    \item Is it true that $4 \divides n^{4}-n$? Provide a proof or a counterexample.
    \begin{subanswer}
      Take $n=2$. $4 \nmid \group{n^{4}-n}$ since $n^{4}-n=16-2=14$ and $4\nmid14$. This is a counterexample. (false)
    \end{subanswer}
  \end{enumerate}

    Assume $a^{n}+1$ is prime, then $n$ is even by the claim above. Suppose, for

    Let $n=2^{t}u$ where $t\ge 0$ and a relative prime (or odd integer) $u>2$,

    \[
    a^{n}+1 = \group{a^{2^{t}}}^u+1

    so $a^{n}+1$ is composite, which contradicts the assumption, $a^{n}+1$ is prime. 
    Therefore, by contradiction, if $a^{n}+1$ is prime, then $a$ is even, and $n$ is a power of $2$.
  \end{subprove}

    \]
    Let $k=t^{2}\in\Z$. Then, $k$ is an integer by Closure Property of Integers. Thus, 
    $n^{2}+1$ has the form $4k+1$.
  \end{subprove}

  \item Exercise Set 1.3, \#43.
  \begin{enumerate}
    \item Let $a,b,c \in \Z$ with $(a,b)=1$. If $a \divides c$ and $b\divides c$ prove that $ab\divides c$.
    \begin{subprove}
      Since $a\divides c$, there exists $s\in\Z$ such that $c=a s$. Thus, $b\divides as$. Since $(a,b)=1$, $a \nmid b$, and
      \begin{define}{Euclid's Lemma}[lem:euclid-lemma][redcolor]
      Let $a,b,p\in\mathds{Z}$ with $p$ prime. If $p\divides ab$, then $p\divides a$ or $p\divides b$.
      \end{define}
      by Euclid's Lemma, $b\divides s$. Thus, $s=bt$ for some $t\in\Z$, and hence $c=as=ab\,t$. Therefore, $ab\divides c$.
    \end{subprove}
    \item Provide a counterexample to show why the statement in (a) cannot hold if $(a,b)>1$.
    \begin{subanswer}
      Take $a=b=2$ and $c=2$. Then $a\divides c$ and $b\divides c$, but $ab=4\nmid 2$. 
      This is a counterexample. (false)
    \end{subanswer}
    \item Let $a_1, \ldots, a_n \in \Z$ be pairwise relatively prime numbers. Prove that if each $a_j \divides c$, then $a_1 \cdots a_n \divides c$.
    \begin{subprove}
      It is trivial to say \inlinequote{$a \divides c$, then $a \divides c$} since it is a 
      tautology, for the sake of inductive proof, base step is true. (base step)\vspace*{10px}\\
      Suppose, for the sake of inductive proof (inductive step), the statement holds for $n-1$ 
      pairwise relatively prime integers.
      Let $b=a_1\cdots a_{n-1}$ satisfying the $n-1$ relative primes and $c=b\cdot a_n = a_1\cdots a_{n}$ assuming $a_n$ is another 
      pairwise prime with $b$, for the sake of direct proof. 
      Then, by the definition of pairwise prime numbers, we have $(b,a_n)=1$, and by all the 
      assumptions, $b\divides c$ and $a_n\divides c$. This satisfies the condition of part (a). 
      Thus, by the part (a), $b a_n\divides c$. In other words, $a_1\cdots a_{n}\divides c$.\vspace*{10px}\\
      Thus, by (base step) and (inductive step), for all $n\in\setZ$, if each $a_j \divides c$, then $a_1 \cdots a_n \divides c$.
    \end{subprove}
  \end{enumerate}

  \item Exercise Set 1.3, \#48. Let $a,b\in \Z$. Show that $(a,b)=1$ if and only if $(a+b,ab) = 1$.
  \begin{subprove}
    ($\Rightarrow$) Assume $(a, b)=1$. Suppose, for the sake of contradiction, that 
    $d=(a+b, a b)$ where $d>1$ then there exists a prime $p$ divides $d$ by Well-Ordering 
    Principle Lemma.
    \begin{define}{Well-Ordering Principle Lemma}[thm:well-ordering-princ][redcolor]
    Every integer $n>1$ has at least one prime divisor.
    \end{define}
    Thus, by the transitive property of divisibility,
    $$
    p \divides(a+b) \quad \text { and } \quad p \divides a b
    $$
    since $p \divides d$, $d \divides \group{a+b}$, and $d \divides \group{a\cdot b}$.
    Then, by \nameref{lem:euclid-lemma}, from $p \divides ab$ we have $p \divides a$ or $p \divides b$.  
    Without loss of generality and by the definition of divisibility, there exists $i\in\setZ$ such that 
    $$
    a = p i.
    $$
    Also, by assumption of $p\divides(a+b)$, there exists $j\in\setZ$ such that $a+b = pj$. Thus,
    $$
    b = pj - pi = p(j-i),
    $$
    so $p\divides b$ as well since it is trivial that $j-i\in\setZ$ by the Closure Property of Integers, 
    which means $p$ is a common prime divisor of $a$ and $b$.
    It contradicts the assumption $(a,b) = p \ne 1$.
    Therefore, $(a,b)=1$ only if $(a+b,ab) = 1$.\vspace*{10px}\\
    ($\Leftarrow$) Assume $(a+b, a b)=1$. Suppose, for the sake of contradiction, 
    that $(a, b)=g>1$. By the Well-Ordering Principle Lemma, there exists a prime 
    $p$ with $p \divides g$. 
    Since $g=(a, b)$ divides both $a$ and $b$, by transitivity of divisibility,
    $$
    p\divides a \text { and } p \divides b .
    $$
    Hence, trivially, $p \divides a b$ and $p \divides(a+b)$, which means $p$ is a common 
    prime divisor of $a b$ and $a+b$. It contradicts the assumption $(a+b, a b) = p \ne1$.
    Therefore, $(a,b)=1$ if $(a+b,ab) = 1$.\vspace*{10px}\\
    Thus, by ($\Rightarrow$) and ($\Leftarrow$), $(a,b)=1$ if and only if $(a+b,ab) = 1$.
  \end{subprove}

  \item Exercise Set 1.4, \#54(c). Use the Euclidean Algorithm for this exercise. Compute the 
  gcd $(441,1155)$, and express it as an integral linear combination of $441$ and $1155$.
  \begin{subanswer}
    By Euclidean algorithm,
    \[
      \begin{aligned}
        1155 &= 441\cdot 2 + 273,\\
        441 &= 273\cdot 1 + 168,\\
        273 &= 168\cdot 1 + 105,\\
        168 &= 105\cdot 1 + 63,\\
        105 &= 63\cdot 1 + 42,\\
        63  &= 42\cdot 1 + 21,\\
        42  &= 21\cdot 2 + 0.
      \end{aligned}
    \]
    Hence $\gcd(441,1155)=21$. Then, by back-substitution,
    \[
      21 = 21\cdot 441 - 8\cdot 1155.
    \]
    Thus $21 = (21)\,441 + (-8)\,1155$, which is a linear combination with Bézout coefficients.
  \end{subanswer}

  \[

  \[
    x\equiv x_0\pmod{M},\qquad x\equiv b_n\pmod{m_n}.

  Let $m$ be a positive integer with $m \neq 2$. If $\{r_1, r_2, \ldots, r_{\phi(m)}\}$ is a reduced residue system modulo $m$, prove that
  \[
  r_1+r_2+\cdots+r_{\phi(m)} \equiv 0 \pmod{m}
  \]
  \begin{define}{Reduced Residue System}[def:reducedRS][red]
  A set of integers $S = \{r_1, r_2, \ldots, r_{\phi(m)}\}$ is called a \underbar{reduced residue system} modulo $m$ if its elements satisfy:
  \begin{enumerate}
    \item For each $r_i \in S$, $\gcd(r_i, m) = 1$.
    \item For any distinct $r_i, r_j \in S$, $r_i \not\equiv r_j \pmod{m}$.
  \end{enumerate}
  \end{define}
  \begin{subprove}
    The case, $m=2$, has been excluded since 
    the reduced residue system is $\{1\}$ by \nameref{def:reducedRS} where the sum is $1 \not\equiv 0 \pmod m$.
    Thus, $m \neq 2$ will be assumed throughout the proof. Since $1 \equiv 0 \pmod 1$, $m=1$ is a trivial case.
    \begin{claim}[claim:additivePairing]
    For $m>2$, any reduced residue system $S$ modulo $m$ can be partitioned into $\phi(m)/2$ pairs of distinct elements $\{r, s\}$ such that
    \[
      r + s \equiv 0 \pmod m.
    \]
    \begin{subprove}
      Let $r$ be an arbitrary element of $S$. Its additive inverse, 
      $-r$, must also be relatively prime to $m$, since $\gcd(r,m)=1$ 
      if and only if $\gcd(-r,m)=1$. By the definition of a reduced residue system, 
      $-r$ must be congruent to a unique element $s \in S$.
      The element $s$ is distinct from $r$. If this were not the case, 
      $r \equiv -r \pmod m$ would hold, implying $2r \equiv 0 \pmod m$. 
      Since $\gcd(r, m) = 1$, this would require $m \mid 2$, 
      which contradicts the assumption that $m > 2$.

      Thus, each element $r \in S$ is uniquely paired with a distinct element $s \in S$ (its additive inverse). 
      In this way, the set $S$ is partitioned into $\phi(m)/2$ such pairs.
    \end{subprove}
    \end{claim}
    By \nameref{claim:additivePairing}, the set $\{r_1, \ldots, r_{\phi(m)}\}$ can be 
    partitioned into pairs, with each pair summing to a multiple of $m$. 
    The total sum of all elements is therefore the sum of these pairs.
    \[
      r_1+r_2+\cdots+r_{\phi(m)} \equiv \sum_{\text{pairs}} (r_i+r_j) \equiv \sum_{\phi(m)/2 \text{ pairs}} 0 \equiv 0 \pmod{m}.
    \]
    Therefore, if $\{r_1, r_2, \ldots, r_{\phi(m)}\}$ is a reduced residue system modulo $m$, then
    \[
    r_1+r_2+\cdots+r_{\phi(m)} \equiv 0 \pmod{m}
    \]
    where $m$ is a positive integer with $m \neq 2$.
  \end{subprove}

      \]
      For $n$ to be $3$-perfect, it is required that $\sigma(n) = 3n = 3 \cdot 2^k \cdot 3 \cdot p = 9 \cdot 2^k \cdot p$.
      Therefore, $(2^{k+1}-1) \cdot 4 \cdot (p+1) = 9 \cdot 2^k \cdot p$, which simplifies to
      \[
      (2^{k+1}-1)(p+1) = \frac{9 \cdot 2^k \cdot p}{4} = \frac{9 \cdot 2^{k-2} \cdot p}{1}.
      \]
      For $k=3$, this becomes $(16-1)(p+1) = 15(p+1) = 9 \cdot 2 \cdot p = 18p$, giving $15p + 15 = 18p$, so $15 = 3p$ and $p = 5$.
      Verification: $n = 2^3 \cdot 3 \cdot 5 = 120$, which was shown to be $3$-perfect in part (a).
      For other values of $k$, similar analysis shows no other odd primes $p$ satisfy the equation.
      Therefore, the only $3$-perfect number of the form $2^k \cdot 3 \cdot p$ is $120 = 2^3 \cdot 3 \cdot 5$.
    \end{subprove}

    \item Let $n \in \setZ$ with $n>1$ and let $p$ be a prime number not dividing $n$. Prove that if $n$ is $p$-perfect, then $p n$ is $(p+1)$-perfect.

  \begin{define}{Von Mangoldt's function}[def:vonmangoldt]
  Let $n \in \setZ$ with $n>0$. Von Mangoldt's function, denoted $\Lambda(n)$, is given by
  \[
  \Lambda(n):= \begin{cases}\ln p & \text { if } n=p^a \text { for some } p \text { prime and } a \in \setZ \\ 0 & \text { otherwise }\end{cases}
  \]
  \end{define}
  \begin{enumerate}
    \item Prove that
    \[
    \Lambda(n)=-\sum_{d \mid n, d>0} \mu(d) \ln d
    \]
    \begin{define}{The Möbius function $\mu$}[def:mobius][red]
    The Möbius function $\mu:\setZ_{>0}\to\{-1,0,1\}$ is defined by
    \[
    \mu(n)=
    \begin{cases}
    1,& n=1,\\[4pt]
    0,&\text{if } p^2\mid n \text{ for some prime } p,\\[4pt]
    (-1)^r,&\text{if } n=p_1p_2\cdots p_r \text{ is a product of } r \text{ distinct primes.}
    \end{cases}
    \]
    \end{define}
    \begin{define}{Möbius Inversion Formula}[thm:mobiusinv][red]
    Let $f$ and $g$ be arithmetic functions. Then $f(n)=\sum_{d\mid n} g(d)$ for all $n\ge 1$ if and only if
    $g(n)=\sum_{d\mid n}\mu(d)f(n/d)$ for all $n\ge 1$.
    \end{define}
    \begin{subprove}
      Let $f(n) = \ln n$ for all $n \geq 1$. Then $f(n) = \sum_{d \mid n} \Lambda(d)$ by the well-known identity.
      By \nameref{thm:mobiusinv}, it follows that
      \[
      \Lambda(n) = \sum_{d \mid n} \mu(d) f(n/d) = \sum_{d \mid n} \mu(d) \ln(n/d).
      \]
      Since $\ln(n/d) = \ln n - \ln d$, this becomes
      \[
      \Lambda(n) = \sum_{d \mid n} \mu(d) (\ln n - \ln d) = \ln n \sum_{d \mid n} \mu(d) - \sum_{d \mid n} \mu(d) \ln d.
      \]
      By the well-known identity $\sum_{d \mid n} \mu(d) = 0$ for $n > 1$ and $\sum_{d \mid 1} \mu(d) = 1$, it follows that
      \[
      \Lambda(n) = -\sum_{d \mid n} \mu(d) \ln d
      \]
      for all $n \geq 1$.
    \end{subprove}
  \end{enumerate}

  \begin{define}{Convolution of Arithmetic Functions}[def:convolution]

  \[
  (f \star g)(n):=\sum_{d \mid n, d>0} f(d) g\left(\frac{n}{d}\right)
  \]
  where $f$ is 
  \end{define}
  \begin{enumerate}
    \item \textbf{Commutativity:} Let $f$ and $g$ be arithmetic functions. Prove that $f \star g=g \star f$.
    \begin{subprove}
      By \nameref{def:convolution}, for any $n \in \setZ_{>0}$,
      \[
      (f \star g)(n) = \sum_{d \mid n} f(d) g(n/d).
      \]
      Substituting $d' = n/d$, as $d$ runs over all divisors of $n$, so does $d'$. Therefore,
      \[
      (f \star g)(n) = \sum_{d' \mid n} f(n/d') g(d') = \sum_{d' \mid n} g(d') f(n/d') = (g \star f)(n).
      \]
      Thus, $f \star g = g \star f$ for all arithmetic functions $f$ and $g$.
    \end{subprove}

    \item \textbf{Associativity:} Let $f, g$, and $h$ be arithmetic functions. Prove that $(f \star g) \star h= f \star(g \star h)$.
    \begin{subprove}
      By \nameref{def:convolution}, for any $n \in \setZ_{>0}$,
      \[
      ((f \star g) \star h)(n) = \sum_{d \mid n} (f \star g)(d) h(n/d) = \sum_{d \mid n} \left(\sum_{c \mid d} f(c) g(d/c)\right) h(n/d).
      \]
      Rearranging the sum over pairs $(c,d)$ where $c \mid d \mid n$, this becomes
      \[
      \sum_{c \mid n} \sum_{\substack{d\\ c \mid d \mid n}} f(c) g(d/c) h(n/d) = \sum_{c \mid n} f(c) \sum_{e \mid (n/c)} g(e) h(n/(ce)),
      \]
      where $e = d/c$. Since $\sum_{e \mid (n/c)} g(e) h((n/c)/e) = (g \star h)(n/c)$, it follows that
      \[
      ((f \star g) \star h)(n) = \sum_{c \mid n} f(c) (g \star h)(n/c) = (f \star (g \star h))(n).
      \]
      Therefore, $(f \star g) \star h = f \star (g \star h)$ for all arithmetic functions $f$, $g$, and $h$.
    \end{subprove}

    \item \textbf{Identity:} For $n \in \setZ$ with $n>0$, let

    \begin{subprove}
      By \nameref{def:convolution}, for any $n \in \setZ_{>0}$,
      \[
      (f \star \delta)(n) = \sum_{d \mid n} f(d) \delta(n/d).
      \]
      Since $\delta(n/d) = 1$ if and only if $n/d = 1$ (i.e., $d = n$), and $\delta(n/d) = 0$ otherwise, it follows that
      \[
      (f \star \delta)(n) = f(n) \delta(1) = f(n) \cdot 1 = f(n).
      \]
      By commutativity (part (a)), $\delta \star f = f \star \delta = f$. Therefore, $f \star \delta = \delta \star f = f$.
    \end{subprove}

      By commutativity (part (a)), $\mathbf{1} \star \mu = \mu \star \mathbf{1} = \delta$.
    \end{subprove}
