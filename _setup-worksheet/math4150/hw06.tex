\begin{define}{Quadratic Residue}[def:quadres][red]
For $a\in\setZ$ and $n\in\setZ_{>0}$, $a$ is a \textbf{quadratic residue modulo $n$} if there exists $x\in\setZ$ such that
\[
x^2 \equiv a \pmod{n}.
\]
If no such $x$ exists, then $a$ is a \textbf{quadratic non-residue modulo $n$}.
For an odd prime $p$, there are exactly $\frac{p-1}{2}$ non-zero quadratic residues modulo $p$ among the integers $1, 2, \ldots, p-1$.
\end{define}


\begin{define}{Legendre Symbol}[def:legendre][red]
  For an odd prime $p$ and $a\in\setZ$ with $p\nmid a$, the Legendre symbol $\left(\frac{a}{p}\right)$ is defined by
  \[
  \left(\frac{a}{p}\right):=
  \begin{cases}
  1, & \text{if } a \text{ is a quadratic residue modulo } p,\\[4pt]
  -1, & \text{if } a \text{ is a quadratic non-residue modulo } p.
  \end{cases}
  \]
  If $p\mid a$, then $\left(\frac{a}{p}\right)=0$.
\end{define}


\begin{define}{Order of an Integer Modulo $m$}[def:order][red]
  For $a\in\setZ$ and $m\in\setZ_{>0}$ with $\gcd(a,m)=1$, the \textbf{order of $a$ modulo $m$} is the smallest positive integer $k$ such that
  \[
  a^k \equiv 1 \pmod{m}.
  \]
\end{define}
\begin{define}{Primitive Root}[def:primroot][red]
  Let $m\in\setZ_{>0}$ and $\phi$ be Euler's totient function.
  An integer $g$ is a \textbf{primitive root modulo $m$} if it satisfies two properties:
  \begin{enumerate}
    \item $\gcd(g,m)=1$
    \item The order of $g$ modulo $m$ is equal to $\phi(m)$.
  \end{enumerate}
  This is equivalent to stating that $g$ is a \textbf{generator} of the multiplicative group of integers modulo $m$, $(\setZ/m\setZ)^\times$. That is, for every integer $a$ with $\gcd(a,m)=1$, there exists an integer $k$ such that
  \[
  g^k \equiv a \pmod m.
  \]
  \textbf{Note}: The order of $a$ modulo $m$ must be a divisor of $\phi(m)$ by Euler's theorem (Upper Bound).
  \end{define}
\begin{define}{Index (Discrete Logarithm)}[def:index][red]
  Let $m\in\setZ_{>0}$ and let $g$ be a primitive root modulo $m$.
  For any integer $a$ with $\gcd(a,m)=1$, the \textbf{index of $a$ with respect to the base $g$}, denoted $\operatorname{ind}_g(a)$, is the unique integer $k$ with $1 \leq k \leq \phi(m)$ such that
  \[
  g^k \equiv a \pmod{m}.
  \]
  The index is also called the \textbf{discrete logarithm} of $a$ to the base $g$ modulo $m$.
  \textbf{Note}: Since $g$ is a primitive root modulo $m$, every integer $a$ with $\gcd(a,m)=1$ has a unique index with respect to $g$.
\end{define}

  \item Exercise Set 5.1, \#9. Let $p$ be an odd prime number and let $r$ be an integer with $p \nmid r$. 
  Prove that $r$ is a primitive root modulo $p$ if and only if $r^{(p-1)/q} \not\equiv 1 \pmod{p}$ 
  for all prime divisors $q$ of $p-1$.
  \begin{subprove}
    The order of an integer modulo $m$ is defined by \nameref{def:order}.
    A primitive root is defined by \nameref{def:primroot}.
    Since $p$ is an odd prime, it follows that $\phi(p) = p-1$.
    For $r \in \setZ$ with $p \nmid r$, it follows that $\gcd(r, p) = 1$, so the order of $r$ modulo $p$ is well-defined.
    By Euler's theorem, the order of $r$ modulo $p$ divides $\phi(p) = p-1$.
    
    ($\Rightarrow$) Suppose $r$ is a primitive root modulo $p$ for the sake of direct proof.
    Then, by \nameref{def:primroot}, the order of $r$ modulo $p$ equals $\phi(p) = p-1$.
    Let $q$ be a prime divisor of $p-1$. It is shown that $r^{(p-1)/q} \not\equiv 1 \pmod{p}$.
    Suppose, for the sake of contradiction, that $r^{(p-1)/q} \equiv 1 \pmod{p}$.
    Then, by \nameref{def:order}, the order of $r$ modulo $p$ divides $(p-1)/q$.
    Since $q$ is a prime divisor of $p-1$, it follows that $(p-1)/q < p-1$.
    Therefore, the order of $r$ modulo $p$ is at most $(p-1)/q < p-1$, which contradicts the fact that the order of $r$ modulo $p$ equals $p-1$.
    Hence, $r^{(p-1)/q} \not\equiv 1 \pmod{p}$ for all prime divisors $q$ of $p-1$.
    
    ($\Leftarrow$) Suppose $r^{(p-1)/q} \not\equiv 1 \pmod{p}$ for all prime divisors $q$ of $p-1$ for the sake of direct proof.
    It is shown that $r$ is a primitive root modulo $p$, i.e., the order of $r$ modulo $p$ equals $p-1$.
    Let $d$ be the order of $r$ modulo $p$. By Euler's theorem, $d$ divides $p-1$.
    Suppose, for the sake of contradiction, that $d < p-1$.
    Then $p-1 = d \cdot k$ for some integer $k > 1$.
    Since $k > 1$, there exists a prime divisor $q$ of $k$, and hence $q$ is also a prime divisor of $p-1$.
    Since $q \mid k$ and $p-1 = d \cdot k$, it follows that $d \mid (p-1)/q$.
    Therefore, $(p-1)/q = d \cdot \ell$ for some positive integer $\ell$.
    Since $r^d \equiv 1 \pmod{p}$ by \nameref{def:order}, it follows that
    \[
    r^{(p-1)/q} = r^{d \cdot \ell} = (r^d)^\ell \equiv 1^\ell \equiv 1 \pmod{p},
    \]
    which contradicts the assumption that $r^{(p-1)/q} \not\equiv 1 \pmod{p}$ for all prime divisors $q$ of $p-1$.
    Therefore, $d = p-1$, so the order of $r$ modulo $p$ equals $p-1 = \phi(p)$.
    Hence, by \nameref{def:primroot}, $r$ is a primitive root modulo $p$.
    
    Therefore, by ($\Rightarrow$) and ($\Leftarrow$), $r$ is a primitive root modulo $p$ if and only if $r^{(p-1)/q} \not\equiv 1 \pmod{p}$ for all prime divisors $q$ of $p-1$.
  \end{subprove}

  \item Exercise Set 5.2, \#11(d). Find all incongruent integers having order 10 modulo 61. 
  [Hint: First find a primitive root modulo 61.]
  \begin{define}{Euler's Totient Function $\phi$}[def:euler_totient][red]
    For $n \in \mathbb{Z}_{>0}$, Euler's totient function $\phi(n)$ is defined as the number of positive integers less than or equal to $n$ that are relatively prime to $n$:
    \[
    \phi(n) := \#\{\,k \in \mathbb{Z}_{>0} : 1 \le k \le n \text{ and } \gcd(k, n) = 1\,\}.
    \]
    If $n$ has the prime factorization $n = p_1^{e_1} p_2^{e_2} \cdots p_r^{e_r}$, then Euler's totient function is given by
    \[
    \phi(n) = n \prod_{i=1}^{r} \left(1 - \frac{1}{p_i}\right).
    \]
  \end{define}  
  \begin{subanswer}
    The order of an integer modulo $m$ is defined by \nameref{def:order}.
    A primitive root is defined by \nameref{def:primroot}.
    Euler's totient function is defined by \nameref{def:euler_totient}.
    Since $61$ is prime, it follows that $\phi(61) = 60$.
    
    First, a primitive root modulo $61$ is found. Since $60 = 2^2 \cdot 3 \cdot 5$, the prime divisors of $60$ are $2$, $3$, and $5$.
    By the previous problem (Exercise Set 5.1, \#9), an integer $r$ with $61 \nmid r$ is a primitive root modulo $61$ if and only if $r^{60/2} = r^{30} \not\equiv 1 \pmod{61}$, $r^{60/3} = r^{20} \not\equiv 1 \pmod{61}$, and $r^{60/5} = r^{12} \not\equiv 1 \pmod{61}$.
    
    It is verified that $2$ is a primitive root modulo $61$:
    \begin{itemize}
      \item $2^{12} = 4096 = 61 \cdot 67 + 9$, so $2^{12} \equiv 9 \pmod{61}$.
      \item $2^{20} = 2^{12+8} = 2^{12} \cdot 2^8 = 2^{12} \cdot 256$. Since $256 = 61 \cdot 4 + 12$, it follows that $2^8 \equiv 12 \pmod{61}$.
      Therefore, $2^{20} \equiv 9 \cdot 12 = 108 = 61 \cdot 1 + 47 \equiv 47 \pmod{61}$.
      \item $2^{30} = 2^{20+10} = 2^{20} \cdot 2^{10}$. Since $2^{10} = 1024 = 61 \cdot 16 + 48$, it follows that $2^{10} \equiv 48 \pmod{61}$.
      Therefore, $2^{30} \equiv 47 \cdot 48 = 2256 = 61 \cdot 36 + 60 \equiv 60 \equiv -1 \pmod{61}$.
    \end{itemize}
    Since $2^{12} \equiv 9 \not\equiv 1$, $2^{20} \equiv 47 \not\equiv 1$, and $2^{30} \equiv 60 \equiv -1 \not\equiv 1 \pmod{61}$, it follows that $2$ is a primitive root modulo $61$.
    
    Let $g = 2$ be a primitive root modulo $61$. Then every integer $a$ with $\gcd(a, 61) = 1$ can be written as $a \equiv g^k \pmod{61}$ for some $k$ with $1 \leq k \leq 60$.
    The order of $g^k$ modulo $61$ is $\frac{60}{\gcd(k, 60)}$.
    
    For the order to be $10$, it is required that $\frac{60}{\gcd(k, 60)} = 10$, which implies $\gcd(k, 60) = 6$.
    The positive integers $k$ with $1 \leq k \leq 60$ and $\gcd(k, 60) = 6$ are those of the form $k = 6m$ where $\gcd(m, 10) = 1$ and $1 \leq 6m \leq 60$, i.e., $1 \leq m \leq 10$ with $\gcd(m, 10) = 1$.
    The integers $m$ with $1 \leq m \leq 10$ and $\gcd(m, 10) = 1$ are $m \in \{1, 3, 7, 9\}$.
    Therefore, $k \in \{6, 18, 42, 54\}$.
    
    It is verified that $\gcd(6, 60) = 6$, $\gcd(18, 60) = 6$, $\gcd(42, 60) = 6$, and $\gcd(54, 60) = 6$.
    Therefore, the integers with order $10$ modulo $61$ are $2^6, 2^{18}, 2^{42}, 2^{54} \pmod{61}$.
    
    Computing these values:
    \begin{align*}
      2^6 &= 64 \equiv 3 \pmod{61},\\
      2^{18} &= (2^6)^3 \equiv 3^3 = 27 \pmod{61},\\
      2^{42} &= 2^{60-18} \equiv 2^{-18} \equiv (2^{18})^{-1} \equiv 27^{-1} \pmod{61},\\
      2^{54} &= 2^{60-6} \equiv 2^{-6} \equiv (2^6)^{-1} \equiv 3^{-1} \pmod{61}.
    \end{align*}
    To find $27^{-1} \pmod{61}$, the extended Euclidean algorithm is used.
    Since $61 = 27 \cdot 2 + 7$, $27 = 7 \cdot 3 + 6$, and $7 = 6 \cdot 1 + 1$, working backwards:
    \[
    1 = 7 - 6 = 7 - (27 - 7 \cdot 3) = 7 \cdot 4 - 27 = (61 - 27 \cdot 2) \cdot 4 - 27 = 61 \cdot 4 - 27 \cdot 9.
    \]
    Therefore, $27 \cdot (-9) \equiv 1 \pmod{61}$, so $27^{-1} \equiv -9 \equiv 52 \pmod{61}$.
    Hence, $2^{42} \equiv 52 \pmod{61}$.
    
    To find $3^{-1} \pmod{61}$, since $61 = 3 \cdot 20 + 1$, it follows that $1 = 61 - 3 \cdot 20$, so $3 \cdot (-20) \equiv 1 \pmod{61}$.
    Therefore, $3^{-1} \equiv -20 \equiv 41 \pmod{61}$.
    Hence, $2^{54} \equiv 41 \pmod{61}$.
    
    Therefore, all incongruent integers having order $10$ modulo $61$ are $3, 27, 52, 41 \pmod{61}$.
  \end{subanswer}

  \item Exercise Set 5.2, \#12. Let $p$ be an odd prime number.
  \begin{enumerate}
    \item Prove that any primitive root $r$ modulo $p$ is a quadratic non-residue modulo $p$. 
    Deduce that $r^{(p-1)/2} \equiv -1 \pmod{p}$.
    \begin{define}{Theorem 4.4 (Euler's Criterion)}[thm:4.4][red]
      Let $p$ be an odd prime and let $a\in\setZ$ with $p\nmid a$. Then
      \[
      \left(\frac{a}{p}\right) \equiv a^{\frac{p-1}{2}} \pmod p,
      \]
      and, since both sides take values in $\{\pm1\}$, we may (and usually do) interpret this congruence as the equality
      \[
      \left(\frac{a}{p}\right) = a^{\frac{p-1}{2}} \pmod p \in\{1,-1\}.
      \]
      \begin{subprove}
        If $\left(\frac{a}{p}\right)=1$, there exists $x_0$ with $x_0^2\equiv a\pmod p$. Then
        \[
        a^{\frac{p-1}{2}} \equiv (x_0^2)^{\frac{p-1}{2}} \equiv x_0^{\,p-1}\equiv 1\pmod p
        \]
        by Fermat's little theorem.
        If $\left(\frac{a}{p}\right)=-1$, the nonzero residues $\{1,2,\dots,p-1\}$ partition into $(p-1)/2$ pairs with product $a$: if $i^2\equiv a\pmod p$ for some $i$, then $a$ is a residue, contradiction. Hence $(p-1)!\equiv a^{\frac{p-1}{2}} \pmod p$. By Wilson's theorem, $(p-1)!\equiv -1\pmod p$, so $a^{\frac{p-1}{2}}\equiv -1\pmod p$.
      \end{subprove}
    \end{define}
    \begin{subprove}
      The order of an integer modulo $m$ is defined by \nameref{def:order}.
      Euler's criterion is given by \nameref{thm:4.4}.
      
      Let $r$ be a primitive root modulo $p$. Then, by \nameref{def:primroot}, the order of $r$ modulo $p$ equals $\phi(p) = p-1$.
      
      It is shown that $r$ is a quadratic non-residue modulo $p$ by contradiction.
      Suppose, for the sake of contradiction, that $r$ is a quadratic residue modulo $p$.
      Then, by \nameref{def:quadres}, there exists $x \in \setZ$ such that $x^2 \equiv r \pmod{p}$.
      Since $r$ is a primitive root, it follows that $\gcd(r, p) = 1$, so $\gcd(x, p) = 1$ as well.
      Therefore,
      \[
      r^{(p-1)/2} \equiv (x^2)^{(p-1)/2} = x^{p-1} \equiv 1 \pmod{p}
      \]
      by Fermat's little theorem.
      However, since the order of $r$ modulo $p$ is $p-1$ and $(p-1)/2 < p-1$, it follows that $r^{(p-1)/2} \not\equiv 1 \pmod{p}$.
      This is a contradiction.
      Therefore, $r$ is not a quadratic residue modulo $p$, so $r$ is a quadratic non-residue modulo $p$.
      
      Since $r$ is a quadratic non-residue modulo $p$, by \nameref{def:legendre}, it follows that $\left(\frac{r}{p}\right) = -1$.
      By \nameref{thm:4.4} (Euler's criterion), it follows that
      \[
      r^{(p-1)/2} \equiv \left(\frac{r}{p}\right) = -1 \pmod{p}.
      \]
      Therefore, $r^{(p-1)/2} \equiv -1 \pmod{p}$.
    \end{subprove}
    
    \item Prove that there are exactly $\frac{p-1}{2}-\phi(p-1)$ incongruent quadratic non-residues 
    modulo $p$ that are not primitive roots modulo $p$.
    \begin{subprove}
      A primitive root is defined by \nameref{def:primroot}.
      A quadratic residue is defined by \nameref{def:quadres}.
      Euler's totient function is defined by \nameref{def:euler_totient}.
      
      Since $p$ is an odd prime, by \nameref{def:quadres}, there are exactly $\frac{p-1}{2}$ quadratic non-residues modulo $p$ among the integers $1, 2, \ldots, p-1$.
      
      Since $p$ is prime, primitive roots exist modulo $p$. The number of incongruent primitive roots modulo $p$ is $\phi(\phi(p)) = \phi(p-1)$.
      
      By part (a), every primitive root modulo $p$ is a quadratic non-residue modulo $p$.
      Therefore, all $\phi(p-1)$ primitive roots are among the $\frac{p-1}{2}$ quadratic non-residues.
      
      It is shown that no quadratic residue can be a primitive root.
      If $a$ is a quadratic residue modulo $p$, then by \nameref{thm:4.4}, $a^{(p-1)/2} \equiv 1 \pmod{p}$.
      Since $(p-1)/2 < p-1 = \phi(p)$, the order of $a$ modulo $p$ divides $(p-1)/2$ and is strictly less than $\phi(p)$.
      Therefore, by \nameref{def:primroot}, $a$ is not a primitive root modulo $p$.
      
      Since all primitive roots are quadratic non-residues and no quadratic residue is a primitive root, it follows that the number of quadratic non-residues that are not primitive roots is
      \[
      \frac{p-1}{2} - \phi(p-1).
      \]
      Therefore, there are exactly $\frac{p-1}{2}-\phi(p-1)$ incongruent quadratic non-residues modulo $p$ that are not primitive roots modulo $p$.
    \end{subprove}
  \end{enumerate}

      Since $2$ is a primitive root modulo $13$ and $2^{12} \not\equiv 1 \pmod{169}$, it follows that $2$ is a primitive root modulo $13^m$ for all $m \geq 1$.
      
      Therefore, $2$ is a primitive root modulo $13^m$ for every positive integer $m$.
    \end{subanswer}

        \item $3^8 = (3^4)^2 \equiv 13^2 = 169 = 17 \cdot 9 + 16 \equiv 16 \equiv -1 \pmod{17}$, so $\operatorname{ind}_3(16) = 8$.

        \item $3^{10} = 3^9 \cdot 3 \equiv 14 \cdot 3 = 42 = 17 \cdot 2 + 8 \equiv 8 \pmod{17}$, so $\operatorname{ind}_3(8) = 10$.
        \item $3^{11} = 3^{10} \cdot 3 \equiv 8 \cdot 3 = 24 = 17 \cdot 1 + 7 \equiv 7 \pmod{17}$, so $\operatorname{ind}_3(7) = 11$.
        \item $3^{12} = 3^{11} \cdot 3 \equiv 7 \cdot 3 = 21 = 17 \cdot 1 + 4 \equiv 4 \pmod{17}$, so $\operatorname{ind}_3(4) = 12$.
        \item $3^{13} = 3^{12} \cdot 3 \equiv 4 \cdot 3 = 12 \equiv 12 \pmod{17}$, so $\operatorname{ind}_3(12) = 13$.
        \item $3^{14} = 3^{13} \cdot 3 \equiv 12 \cdot 3 = 36 = 17 \cdot 2 + 2 \equiv 2 \pmod{17}$, so $\operatorname{ind}_3(2) = 14$.
      \end{itemize}
      
      Therefore, $\operatorname{ind}_3(7) = 11$ and $\operatorname{ind}_3(2) = 14$.
      The congruence becomes
      \[
      11 + 5\operatorname{ind}_3(x) \equiv 14 \pmod{16},
      \]
      which simplifies to
      \[
      5\operatorname{ind}_3(x) \equiv 3 \pmod{16}.
      \]
      
      Since $\gcd(5, 16) = 1$, the congruence has a unique solution modulo $16$.
      Since $5 \cdot 13 = 65 = 16 \cdot 4 + 1$, it follows that $5^{-1} \equiv 13 \pmod{16}$.
      Therefore,
      \[
      \operatorname{ind}_3(x) \equiv 3 \cdot 13 = 39 \equiv 7 \pmod{16}.
      \]
      
      Since $\operatorname{ind}_3(x) \equiv 7 \pmod{16}$, it follows that $x \equiv 3^7 \equiv 11 \pmod{17}$.
      Therefore, the unique solution is $x \equiv 11 \pmod{17}$.
    \end{subanswer}
  \end{enumerate}

  \item Exercise Set 5.4, \#35. Let $p$ be a prime number and let $r$ and $s$ be primitive roots modulo $p$. 
  Let $a \in \setZ$ with $p \nmid a$.
  \begin{enumerate}
    \item Prove that $\operatorname{ind}_s a \equiv (\operatorname{ind}_r a)(\operatorname{ind}_s r) \pmod{p-1}$. 
    (This corresponds to the change-of-base formula for logarithms.)
    \begin{subprove}
      A primitive root is defined by \nameref{def:primroot}.
      The index is defined by \nameref{def:index}.
      Since $r$ and $s$ are primitive roots modulo $p$, by \nameref{def:index}, there exist unique integers $\operatorname{ind}_r a$, $\operatorname{ind}_s a$, and $\operatorname{ind}_r s$ such that
      \[
      a \equiv r^{\operatorname{ind}_r a} \pmod{p}, \quad a \equiv s^{\operatorname{ind}_s a} \pmod{p}, \quad s \equiv r^{\operatorname{ind}_r s} \pmod{p}.
      \]
      
      Since $s \equiv r^{\operatorname{ind}_r s} \pmod{p}$, it follows that
      \[
      s^{\operatorname{ind}_s a} \equiv (r^{\operatorname{ind}_r s})^{\operatorname{ind}_s a} = r^{\operatorname{ind}_r s \cdot \operatorname{ind}_s a} \pmod{p}.
      \]
      Since $a \equiv s^{\operatorname{ind}_s a} \pmod{p}$ and $a \equiv r^{\operatorname{ind}_r a} \pmod{p}$, it follows that
      \[
      r^{\operatorname{ind}_r s \cdot \operatorname{ind}_s a} \equiv r^{\operatorname{ind}_r a} \pmod{p}.
      \]
      
      Since $r$ is a primitive root modulo $p$, the order of $r$ modulo $p$ is $\phi(p) = p-1$.
      Therefore, by \nameref{def:order}, it follows that
      \[
      \operatorname{ind}_r s \cdot \operatorname{ind}_s a \equiv \operatorname{ind}_r a \pmod{p-1}.
      \]
      
      Since $s$ is a primitive root modulo $p$, it follows that $\gcd(\operatorname{ind}_r s, p-1) = 1$ (because if $\gcd(\operatorname{ind}_r s, p-1) = d > 1$, then the order of $s$ modulo $p$ would divide $(p-1)/d < p-1$, contradicting that $s$ is a primitive root).
      Therefore, $\operatorname{ind}_r s$ has a multiplicative inverse modulo $p-1$.
      Multiplying both sides of the congruence by $(\operatorname{ind}_r s)^{-1} \pmod{p-1}$, it follows that
      \[
      \operatorname{ind}_s a \equiv \operatorname{ind}_r a \cdot (\operatorname{ind}_r s)^{-1} \pmod{p-1}.
      \]
      
      However, since $r \equiv s^{\operatorname{ind}_s r} \pmod{p}$ and $s \equiv r^{\operatorname{ind}_r s} \pmod{p}$, by the same argument it follows that $\operatorname{ind}_r s \cdot \operatorname{ind}_s r \equiv 1 \pmod{p-1}$.
      Therefore, $(\operatorname{ind}_r s)^{-1} \equiv \operatorname{ind}_s r \pmod{p-1}$.
      
      Substituting, it follows that
      \[
      \operatorname{ind}_s a \equiv \operatorname{ind}_r a \cdot \operatorname{ind}_s r \pmod{p-1}.
      \]
      Therefore, $\operatorname{ind}_s a \equiv (\operatorname{ind}_r a)(\operatorname{ind}_s r) \pmod{p-1}$.
    \end{subprove}
    
    \item Prove that $\operatorname{ind}_r(p-a) \equiv \operatorname{ind}_r a + \frac{p-1}{2} \pmod{p-1}$. 
    (This congruence yields all indices relative to $r$ after half of these indices are computed.)
    \begin{subprove}
      A primitive root is defined by \nameref{def:primroot}.
      The index is defined by \nameref{def:index}.
      The order of an integer modulo $m$ is defined by \nameref{def:order}.
      Since $r$ is a primitive root modulo $p$, by \nameref{def:index}, there exists a unique integer $\operatorname{ind}_r a$ such that
      \[
      a \equiv r^{\operatorname{ind}_r a} \pmod{p}.
      \]
      
      Since $p-a \equiv -a \pmod{p}$, it follows that
      \[
      p-a \equiv -a \equiv -r^{\operatorname{ind}_r a} \pmod{p}.
      \]
      
      Since $r$ is a primitive root modulo $p$, the order of $r$ modulo $p$ is $\phi(p) = p-1$.
      By part (a) of Exercise Set 5.2, \#12, since $r$ is a primitive root modulo $p$, it follows that $r^{(p-1)/2} \equiv -1 \pmod{p}$.
      Therefore,
      \[
      -r^{\operatorname{ind}_r a} \equiv r^{(p-1)/2} \cdot r^{\operatorname{ind}_r a} = r^{\operatorname{ind}_r a + (p-1)/2} \pmod{p}.
      \]
      
      Since $p-a \equiv r^{\operatorname{ind}_r a + (p-1)/2} \pmod{p}$ and $p-a \equiv r^{\operatorname{ind}_r(p-a)} \pmod{p}$, it follows that
      \[
      r^{\operatorname{ind}_r a + (p-1)/2} \equiv r^{\operatorname{ind}_r(p-a)} \pmod{p}.
      \]
      
      Since the order of $r$ modulo $p$ is $p-1$, by \nameref{def:order}, it follows that
      \[
      \operatorname{ind}_r a + \frac{p-1}{2} \equiv \operatorname{ind}_r(p-a) \pmod{p-1}.
      \]
      Therefore, $\operatorname{ind}_r(p-a) \equiv \operatorname{ind}_r a + \frac{p-1}{2} \pmod{p-1}$.
    \end{subprove}
  \end{enumerate}

  \item Exercise Set 5.4, \#36.
  \begin{enumerate}
    \item Let $p$ be an odd prime number. Prove that the congruence $x^4 \equiv -1 \pmod{p}$ is solvable 
    if and only if $p \equiv 1 \pmod{8}$.
    \begin{define}{Theorem 4.6 (Value of $(-1/p)$)}[thm:4.6][red]
    Let $p$ be an odd prime. Then
    \[
    \left(\frac{-1}{p}\right)=(-1)^{\frac{p-1}{2}}=
    \begin{cases}
    1, & \text{if } p\equiv 1\pmod 4,\\[4pt]
    -1,& \text{if } p\equiv 3\pmod 4.
    \end{cases}
    \]
    \begin{subprove}
    By \nameref{thm:4.4} with $a=-1$:
    \[
    \left(\frac{-1}{p}\right)\equiv (-1)^{\frac{p-1}{2}}\pmod p.
    \]
    Both sides lie in $\{\pm1\}$, hence equality holds. Writing $p=4k+1$ or $p=4k+3$ gives the stated cases.
    \end{subprove}
    \end{define}
    \begin{subprove}
      ($\Rightarrow$) Suppose $x^4 \equiv -1 \pmod{p}$ is solvable for the sake of direct proof.
      Then there exists $x \in \setZ$ with $p \nmid x$ such that $x^4 \equiv -1 \pmod{p}$.
      Since $x^4 \equiv -1 \pmod{p}$, it follows that $(x^2)^2 \equiv -1 \pmod{p}$.
      Therefore, $-1$ is a quadratic residue modulo $p$.
      By \nameref{thm:4.6}, it follows that $\left(\frac{-1}{p}\right) = 1$, which implies $p \equiv 1 \pmod{4}$.
      
      Since $x^4 \equiv -1 \pmod{p}$, it follows that $x^8 \equiv 1 \pmod{p}$.
      By \nameref{def:order}, the order of $x$ modulo $p$ divides $8$.
      Since $x^4 \equiv -1 \not\equiv 1 \pmod{p}$, the order of $x$ modulo $p$ does not divide $4$.
      Therefore, the order of $x$ modulo $p$ is exactly $8$.
      Since the order of $x$ modulo $p$ divides $\phi(p) = p-1$ by Euler's theorem, it follows that $8 \mid p-1$.
      Therefore, $p \equiv 1 \pmod{8}$.
      
      ($\Leftarrow$) Suppose $p \equiv 1 \pmod{8}$ for the sake of direct proof.
      It is shown that $x^4 \equiv -1 \pmod{p}$ is solvable.
      Since $p \equiv 1 \pmod{8}$, there exists a primitive root $g$ modulo $p$ by \nameref{def:primroot}.
      Since $8 \mid p-1$, let $h = g^{(p-1)/8}$.
      Then $h^8 = g^{p-1} \equiv 1 \pmod{p}$ by Fermat's little theorem.
      Since $g$ is a primitive root modulo $p$, the order of $g$ modulo $p$ is $\phi(p) = p-1$.
      Therefore, the order of $h = g^{(p-1)/8}$ modulo $p$ is exactly $8$ (since if the order were a proper divisor $d$ of $8$, then $g^{(p-1)/8 \cdot d} \equiv 1 \pmod{p}$, which would imply the order of $g$ divides $(p-1)/8 \cdot d < p-1$, a contradiction).
      Since the order of $h$ modulo $p$ is $8$ and $h^8 \equiv 1 \pmod{p}$, it follows that $h^4 \not\equiv 1 \pmod{p}$ (otherwise the order would divide $4$).
      Since $(h^4)^2 = h^8 \equiv 1 \pmod{p}$ and $h^4 \not\equiv 1 \pmod{p}$, it follows that $h^4 \equiv -1 \pmod{p}$.
      Therefore, $x = h$ is a solution to $x^4 \equiv -1 \pmod{p}$.
      
      Therefore, by ($\Rightarrow$) and ($\Leftarrow$), the congruence $x^4 \equiv -1 \pmod{p}$ is solvable if and only if $p \equiv 1 \pmod{8}$.
    \end{subprove}
    
    \item Prove that there are infinitely many primes expressible of the form $8n+1$ where $n$ is a positive integer.
    [Hint: Parallel the proof from Question 6 on Homework 5.]
    \begin{subprove}      
      Suppose that only finitely many primes of the form $8n+1$ exist for the sake of contradiction.
      Let these be $p_1, p_2, \ldots, p_r$. Consider $N = (2p_1p_2\cdots p_r)^4 + 1$.
      Since $N$ is odd and greater than $1$, it has an odd prime divisor $q$.
      Since $N \equiv 1 \pmod{8}$ (since $(2p_1\cdots p_r)^4$ is divisible by $16$ and $N = (2p_1\cdots p_r)^4 + 1 \equiv 1 \pmod{8}$), any odd divisor of $N$ satisfies $q \equiv 1, 3, 5, \text{ or } 7 \pmod{8}$.
      
      Since $(2p_1\cdots p_r)^4 \equiv -1 \pmod{q}$, it follows that $x^4 \equiv -1 \pmod{q}$ is solvable with $x = 2p_1\cdots p_r$.
      By part (a), it follows that $q \equiv 1 \pmod{8}$.
      
      However, $q$ does not divide any $p_i$ (since $N \equiv 1 \pmod{p_i}$), and $q$ is not among $p_1, \ldots, p_r$.
      This contradicts the assumption that only finitely many primes of the form $8n+1$ exist.\vspace*{10px}\\
      Therefore, by contradiction, infinitely many primes of the form $8n+1$ exist.
    \end{subprove}
  \end{enumerate}
\end{enumerate}
