\begin{define}{Quadratic residue modulo $m$}[def:quadres][red]
Let $a,m\in\mathbb{Z}$ with $m>0$ and $\gcd(a,m)=1$. An integer $a$ is said to be a
\emph{quadratic residue modulo $m$} if the congruence
\[
x^2 \equiv a \pmod{n}.
\]
For an odd prime $p$, there are exactly $\frac{p-1}{2}$ non-zero quadratic residues modulo $p$ among the integers $1, 2, \ldots, p-1$.
\end{define}


\begin{define}{Definition 4.1 (Legendre symbol)}[def:legendre][red]
Let $p$ be an odd prime and let $a\in\mathbb{Z}$. The \emph{Legendre symbol} $(\tfrac{a}{p})$ is defined by
\[
\left(\frac{a}{p}\right):=
\begin{cases}
0, & \text{if } p\mid a,\\[4pt]
1, & \text{if } p\nmid a\text{ and } a \text{ is a quadratic residue modulo } p,\\[4pt]
-1,& \text{if } p\nmid a\text{ and } a \text{ is a quadratic non-residue modulo } p.
\end{cases}
\]
\end{define}

  \item Exercise Set 5.1, \#9. Let $p$ be an odd prime number and let $r$ be an integer with $p \nmid r$. 
  Prove that $r$ is a primitive root modulo $p$ if and only if $r^{(p-1)/q} \not\equiv 1 \pmod{p}$ 
  for all prime divisors $q$ of $p-1$.
  \begin{define}{Order of an Integer Modulo $m$}[def:order][red]
  Let $m>0$ be an integer and let $a\in\mathbb{Z}$ with $\gcd(a,m)=1$. The \emph{order} of $a$ modulo $m$, denoted $\operatorname{ord}_m(a)$, is the least positive integer $n$ such that
  \[
  a^n\equiv 1 \pmod m.
  \]
  \end{define}
  \begin{define}{Primitive Root Modulo $m$}[def:primroot][red]
  Let $m>0$. An integer $r$ with $\gcd(r,m)=1$ is called a \emph{primitive root modulo $m$} if
  \[
  \operatorname{ord}_m(r)=\varphi(m).
  \]
  \end{define}
  \begin{define}{Euler's Theorem}[thm:euler][red]
  Let $m>0$ and let $a\in\mathbb{Z}$ with $\gcd(a,m)=1$. Then
  \[
  a^{\varphi(m)}\equiv 1 \pmod{m}.
  \]
  In particular, for a prime $p$ and $a\in\mathbb{Z}$ with $p\nmid a$, we have
  \[
  a^{p-1}\equiv 1 \pmod{p}.
  \]
  \end{define}
  \begin{subprove}
    The order of an integer modulo $m$ is defined by \nameref{def:order}.
    A primitive root modulo $m$ is defined by \nameref{def:primroot}.
    Euler's theorem is given by \nameref{thm:euler}.
    Since $p$ is an odd prime, it follows that $\varphi(p) = p-1$.
    For $r \in \mathbb{Z}$ with $p \nmid r$, it follows that $\gcd(r, p) = 1$, so the order of $r$ modulo $p$ is well-defined.
    By \nameref{thm:euler}, the order of $r$ modulo $p$ divides $\varphi(p) = p-1$.
    
    ($\Rightarrow$) Suppose $r$ is a primitive root modulo $p$ for the sake of direct proof.
    Then, by \nameref{def:primroot}, $\operatorname{ord}_p(r) = \varphi(p) = p-1$.
    Let $q$ be a prime divisor of $p-1$. It is shown that $r^{(p-1)/q} \not\equiv 1 \pmod{p}$.
    Suppose, for the sake of contradiction, that $r^{(p-1)/q} \equiv 1 \pmod{p}$.
    Then, by \nameref{def:order}, $\operatorname{ord}_p(r)$ divides $(p-1)/q$.
    Since $q$ is a prime divisor of $p-1$, it follows that $(p-1)/q < p-1$.
    Therefore, $\operatorname{ord}_p(r) \leq (p-1)/q < p-1$, which contradicts the fact that $\operatorname{ord}_p(r) = p-1$.
    Hence, $r^{(p-1)/q} \not\equiv 1 \pmod{p}$ for all prime divisors $q$ of $p-1$.
    
    ($\Leftarrow$) Suppose $r^{(p-1)/q} \not\equiv 1 \pmod{p}$ for all prime divisors $q$ of $p-1$ for the sake of direct proof.
    It is shown that $r$ is a primitive root modulo $p$, i.e., $\operatorname{ord}_p(r) = p-1$.
    Let $d = \operatorname{ord}_p(r)$. By \nameref{thm:euler}, $d$ divides $p-1$.
    Suppose, for the sake of contradiction, that $d < p-1$.
    Then $p-1 = d \cdot k$ for some integer $k > 1$.
    Since $k > 1$, there exists a prime divisor $q$ of $k$, and hence $q$ is also a prime divisor of $p-1$.
    Since $q \mid k$ and $p-1 = d \cdot k$, it follows that $d \mid (p-1)/q$.
    Therefore, $(p-1)/q = d \cdot \ell$ for some positive integer $\ell$.
    Since $r^d \equiv 1 \pmod{p}$ by \nameref{def:order}, it follows that
    \[
    r^{(p-1)/q} = r^{d \cdot \ell} = (r^d)^\ell \equiv 1^\ell \equiv 1 \pmod{p},
    \]
    which contradicts the assumption that $r^{(p-1)/q} \not\equiv 1 \pmod{p}$ for all prime divisors $q$ of $p-1$.
    Therefore, $d = p-1$, so $\operatorname{ord}_p(r) = p-1 = \varphi(p)$.
    Hence, by \nameref{def:primroot}, $r$ is a primitive root modulo $p$.
    
    Therefore, by ($\Rightarrow$) and ($\Leftarrow$), $r$ is a primitive root modulo $p$ if and only if $r^{(p-1)/q} \not\equiv 1 \pmod{p}$ for all prime divisors $q$ of $p-1$.
  \end{subprove}

  \item Exercise Set 5.2, \#11(d). Find all incongruent integers having order 10 modulo 61. 
  [Hint: First find a primitive root modulo 61.]
  \begin{define}{Euler Totient Function}[def:euler_totient][red]
    For $n\in\mathbb{Z}_{>0}$ the Euler totient function $\varphi(n)$ is the number of integers
    $1\le x\le n$ with $\gcd(x,n)=1$; equivalently,
    \[
    \varphi(n):=\#\{x\in\{1,2,\dots,n\}:\gcd(x,n)=1\}.
    \]
    If $n$ has the prime factorization $n = p_1^{e_1} p_2^{e_2} \cdots p_r^{e_r}$, then Euler's totient function is given by
    \[
    \varphi(n) = n \prod_{i=1}^{r} \left(1 - \frac{1}{p_i}\right).
    \]
  \end{define}  
  \begin{subanswer}
    The order of an integer modulo $m$ is defined by \nameref{def:order}.
    A primitive root modulo $m$ is defined by \nameref{def:primroot}.
    Euler's totient function is defined by \nameref{def:euler_totient}.
    Since $61$ is prime, it follows that $\varphi(61) = 60$.
    
    First, a primitive root modulo $61$ is found. Since $60 = 2^2 \cdot 3 \cdot 5$, the prime divisors of $60$ are $2$, $3$, and $5$.
    By the previous problem (Exercise Set 5.1, \#9), an integer $r$ with $61 \nmid r$ is a primitive root modulo $61$ if and only if $r^{60/2} = r^{30} \not\equiv 1 \pmod{61}$, $r^{60/3} = r^{20} \not\equiv 1 \pmod{61}$, and $r^{60/5} = r^{12} \not\equiv 1 \pmod{61}$.
    
    It is verified that $2$ is a primitive root modulo $61$:
    \begin{itemize}
      \item $2^{12} = 4096 = 61 \cdot 67 + 9$, so $2^{12} \equiv 9 \pmod{61}$.
      \item $2^{20} = 2^{12+8} = 2^{12} \cdot 2^8 = 2^{12} \cdot 256$. Since $256 = 61 \cdot 4 + 12$, it follows that $2^8 \equiv 12 \pmod{61}$.
      Therefore, $2^{20} \equiv 9 \cdot 12 = 108 = 61 \cdot 1 + 47 \equiv 47 \pmod{61}$.
      \item $2^{30} = 2^{20+10} = 2^{20} \cdot 2^{10}$. Since $2^{10} = 1024 = 61 \cdot 16 + 48$, it follows that $2^{10} \equiv 48 \pmod{61}$.
      Therefore, $2^{30} \equiv 47 \cdot 48 = 2256 = 61 \cdot 36 + 60 \equiv 60 \equiv -1 \pmod{61}$.
    \end{itemize}
    Since $2^{12} \equiv 9 \not\equiv 1$, $2^{20} \equiv 47 \not\equiv 1$, and $2^{30} \equiv 60 \equiv -1 \not\equiv 1 \pmod{61}$, it follows that $2$ is a primitive root modulo $61$.
    
    \begin{define}{Proposition 5.4}[prop:5.4][red]
      Let $m>0$ and $a\in\mathbb{Z}$ with $\gcd(a,m)=1$. For any positive integer $i$,
      \[
      \operatorname{ord}_m\!\bigl(a^i\bigr)=\frac{\operatorname{ord}_m(a)}{\gcd\bigl(\operatorname{ord}_m(a),\,i\bigr)}.
      \]
    \end{define}

    Let $g = 2$ be a primitive root modulo $61$. Then every integer $a$ with $\gcd(a, 61) = 1$ can be written as $a \equiv g^k \pmod{61}$ for some $k$ with $1 \leq k \leq 60$.
    By \nameref{prop:5.4}, the order of $g^k$ modulo $61$ is $\frac{\operatorname{ord}_{61}(g)}{\gcd(\operatorname{ord}_{61}(g), k)} = \frac{60}{\gcd(60, k)}$.
    
    For the order to be $10$, it is required that $\frac{60}{\gcd(k, 60)} = 10$, which implies $\gcd(k, 60) = 6$.
    The positive integers $k$ with $1 \leq k \leq 60$ and $\gcd(k, 60) = 6$ are those of the form $k = 6m$ where $\gcd(m, 10) = 1$ and $1 \leq 6m \leq 60$, i.e., $1 \leq m \leq 10$ with $\gcd(m, 10) = 1$.
    The integers $m$ with $1 \leq m \leq 10$ and $\gcd(m, 10) = 1$ are $m \in \{1, 3, 7, 9\}$.
    Therefore, $k \in \{6, 18, 42, 54\}$.
    
    It is verified that $\gcd(6, 60) = 6$, $\gcd(18, 60) = 6$, $\gcd(42, 60) = 6$, and $\gcd(54, 60) = 6$.
    Therefore, the integers with order $10$ modulo $61$ are $2^6, 2^{18}, 2^{42}, 2^{54} \pmod{61}$.
    
    Computing these values:
    \begin{align*}
      2^6 &= 64 \equiv 3 \pmod{61},\\
      2^{18} &= (2^6)^3 \equiv 3^3 = 27 \pmod{61},\\
      2^{42} &= 2^{60-18} \equiv 2^{-18} \equiv (2^{18})^{-1} \equiv 27^{-1} \pmod{61},\\
      2^{54} &= 2^{60-6} \equiv 2^{-6} \equiv (2^6)^{-1} \equiv 3^{-1} \pmod{61}.
    \end{align*}
    To find $27^{-1} \pmod{61}$, the extended Euclidean algorithm is used.
    Since $61 = 27 \cdot 2 + 7$, $27 = 7 \cdot 3 + 6$, and $7 = 6 \cdot 1 + 1$, working backwards:
    \[
    1 = 7 - 6 = 7 - (27 - 7 \cdot 3) = 7 \cdot 4 - 27 = (61 - 27 \cdot 2) \cdot 4 - 27 = 61 \cdot 4 - 27 \cdot 9.
    \]
    Therefore, $27 \cdot (-9) \equiv 1 \pmod{61}$, so $27^{-1} \equiv -9 \equiv 52 \pmod{61}$.
    Hence, $2^{42} \equiv 52 \pmod{61}$.
    
    To find $3^{-1} \pmod{61}$, since $61 = 3 \cdot 20 + 1$, it follows that $1 = 61 - 3 \cdot 20$, so $3 \cdot (-20) \equiv 1 \pmod{61}$.
    Therefore, $3^{-1} \equiv -20 \equiv 41 \pmod{61}$.
    Hence, $2^{54} \equiv 41 \pmod{61}$.
    
    Therefore, all incongruent integers having order $10$ modulo $61$ are $3, 27, 52, 41 \pmod{61}$.
  \end{subanswer}

  \item Exercise Set 5.2, \#12. Let $p$ be an odd prime number.
  \begin{enumerate}
    \item Prove that any primitive root $r$ modulo $p$ is a quadratic non-residue modulo $p$. 
    Deduce that $r^{(p-1)/2} \equiv -1 \pmod{p}$.
    \begin{define}{Theorem 4.4 (Euler's Criterion)}[thm:4.4][red]
      Let $p$ be an odd prime and let $a\in\mathbb{Z}$ with $p\nmid a$. Then
      \[
      \left(\frac{a}{p}\right) \equiv a^{\frac{p-1}{2}} \pmod p,
      \]
      and, since both sides take values in $\{\pm1\}$, this congruence may be (and usually is) interpreted as the equality
      \[
      \left(\frac{a}{p}\right) = a^{\frac{p-1}{2}} \pmod p \in\{1,-1\}.
      \]
    \end{define}
    \begin{define}{Fermat's Little Theorem}[thm:fermat][red]
      Let $p$ be a prime and let $a\in\mathbb{Z}$ with $p\nmid a$. Then
      \[
      a^{p-1}\equiv 1 \pmod{p}.
      \]
    \end{define}
    \begin{subprove}
      A primitive root modulo $m$ is defined by \nameref{def:primroot}.
      A quadratic residue modulo $m$ is defined by \nameref{def:quadres}.
      The Legendre symbol is defined by \nameref{def:legendre}.
      The order of an integer modulo $m$ is defined by \nameref{def:order}.
      Euler's criterion is given by \nameref{thm:4.4}.
      Fermat's little theorem is given by \nameref{thm:fermat}.
      
      Let $r$ be a primitive root modulo $p$. Then, by \nameref{def:primroot}, $\operatorname{ord}_p(r) = \varphi(p) = p-1$.
      
      It is shown that $r$ is a quadratic non-residue modulo $p$ by contradiction.
      Suppose, for the sake of contradiction, that $r$ is a quadratic residue modulo $p$.
      Then, by \nameref{def:quadres}, there exists $x \in \mathbb{Z}$ such that $x^2 \equiv r \pmod{p}$.
      Since $r$ is a primitive root, it follows that $\gcd(r, p) = 1$, so $\gcd(x, p) = 1$ as well.
      Therefore,
      \[
      r^{(p-1)/2} \equiv (x^2)^{(p-1)/2} = x^{p-1} \equiv 1 \pmod{p}
      \]
      by \nameref{thm:fermat}.
      However, since $\operatorname{ord}_p(r) = p-1$ and $(p-1)/2 < p-1$, it follows that $r^{(p-1)/2} \not\equiv 1 \pmod{p}$.
      This is a contradiction.
      Therefore, $r$ is not a quadratic residue modulo $p$, so $r$ is a quadratic non-residue modulo $p$.
      
      Since $r$ is a quadratic non-residue modulo $p$, by \nameref{def:legendre}, it follows that $\left(\frac{r}{p}\right) = -1$.
      By \nameref{thm:4.4} (Euler's criterion), it follows that
      \[
      r^{(p-1)/2} \equiv \left(\frac{r}{p}\right) = -1 \pmod{p}.
      \]
      Therefore, $r^{(p-1)/2} \equiv -1 \pmod{p}$.
    \end{subprove}
    
    \item Prove that there are exactly $\frac{p-1}{2}-\varphi(p-1)$ incongruent quadratic non-residues 
    modulo $p$ that are not primitive roots modulo $p$.
    \begin{subprove}
      A primitive root modulo $m$ is defined by \nameref{def:primroot}.
      A quadratic residue modulo $m$ is defined by \nameref{def:quadres}.
      Euler's totient function is defined by \nameref{def:euler_totient}.
      Euler's criterion is given by \nameref{thm:4.4}.
      
      Since $p$ is an odd prime, by \nameref{def:quadres}, there are exactly $\frac{p-1}{2}$ quadratic non-residues modulo $p$ among the integers $1, 2, \ldots, p-1$.
      
      Since $p$ is prime, primitive roots exist modulo $p$. The number of incongruent primitive roots modulo $p$ is $\varphi(\varphi(p)) = \varphi(p-1)$.
      
      By part (a), every primitive root modulo $p$ is a quadratic non-residue modulo $p$.
      Therefore, all $\varphi(p-1)$ primitive roots are among the $\frac{p-1}{2}$ quadratic non-residues.
      
      It is shown that no quadratic residue can be a primitive root.
      If $a$ is a quadratic residue modulo $p$, then by \nameref{thm:4.4}, $a^{(p-1)/2} \equiv 1 \pmod{p}$.
      Since $(p-1)/2 < p-1 = \varphi(p)$, the order of $a$ modulo $p$ divides $(p-1)/2$ and is strictly less than $\varphi(p)$.
      Therefore, by \nameref{def:primroot}, $a$ is not a primitive root modulo $p$.
      
      Since all primitive roots are quadratic non-residues and no quadratic residue is a primitive root, it follows that the number of quadratic non-residues that are not primitive roots is
      \[
      \frac{p-1}{2} - \varphi(p-1).
      \]
      Therefore, there are exactly $\frac{p-1}{2}-\varphi(p-1)$ incongruent quadratic non-residues modulo $p$ that are not primitive roots modulo $p$.
    \end{subprove}
  \end{enumerate}

      For an odd prime $p$, if $g$ is a primitive root modulo $p$ and $g^{p-1} \not\equiv 1 \pmod{p^2}$, then $g$ is a primitive root modulo $p^m$ for all $m \geq 1$.
      If $g^{p-1} \equiv 1 \pmod{p^2}$, then $g+p$ is a primitive root modulo $p^m$ for all $m \geq 1$.
    \end{define}
    \begin{subanswer}
      A primitive root modulo $m$ is defined by \nameref{def:primroot}.
      The order of an integer modulo $m$ is defined by \nameref{def:order}.
      Euler's totient function is defined by \nameref{def:euler_totient}.
      The result on primitive roots modulo prime powers is given by \nameref{thm:primroot_primepower}.
      
      For $p = 13$, it follows that $\varphi(13) = 12$. The prime divisors of $12$ are $2$ and $3$.
      By Exercise Set 5.1, \#9, an integer $r$ with $13 \nmid r$ is a primitive root modulo $13$ if and only if $r^{12/2} = r^6 \not\equiv 1 \pmod{13}$ and $r^{12/3} = r^4 \not\equiv 1 \pmod{13}$.
      
      It is verified that $2$ is a primitive root modulo $13$:
      \begin{itemize}
        \item $2^4 = 16 = 13 \cdot 1 + 3$, so $2^4 \equiv 3 \not\equiv 1 \pmod{13}$.
        \item $2^6 = 64 = 13 \cdot 4 + 12$, so $2^6 \equiv 12 \equiv -1 \not\equiv 1 \pmod{13}$.
      \end{itemize}
      Therefore, $2$ is a primitive root modulo $13$.
      
      To check if $2$ works modulo $13^m$ for all $m \geq 1$, it is verified that $2^{12} \not\equiv 1 \pmod{169}$.
      Since $2^6 \equiv -1 \pmod{13}$, it follows that $2^6 = 13k - 1$ for some integer $k$.
      Then $2^{12} = (2^6)^2 = (13k - 1)^2 = 169k^2 - 26k + 1 \equiv 1 - 26k \pmod{169}$.
      Since $2^6 = 64 = 13 \cdot 5 - 1$, it follows that $k = 5$.
      Therefore, $2^{12} \equiv 1 - 26 \cdot 5 = 1 - 130 = -129 \equiv 40 \pmod{169}$.
      More directly: $2^{12} = 4096 = 169 \cdot 24 + 40$, so $2^{12} \equiv 40 \not\equiv 1 \pmod{169}$.
      
      Since $2$ is a primitive root modulo $13$ and $2^{12} \not\equiv 1 \pmod{169}$, by \nameref{thm:primroot_primepower}, it follows that $2$ is a primitive root modulo $13^m$ for all $m \geq 1$.
      
      Therefore, $2$ is a primitive root modulo $13^m$ for every positive integer $m$.
    \end{subanswer}
  \end{enumerate}

  \item Exercise Set 5.4, \#30(a),(e). Use indices to find all incongruent solutions of each congruence below.
  \begin{define}{Index}[def:index][red]
  Let $r$ be a primitive root modulo $m$. For $a\in\mathbb{Z}$ with $\gcd(a,m)=1$ the \emph{index of $a$ with respect to $r$}, denoted $\operatorname{ind}_r(a)$, is the least positive integer $n$ such that
  \[
  r^n\equiv a\pmod m.
  \]
  \end{define}
  \begin{define}{Properties of indices}[prop:5.20][red]
  Let $r$ be a primitive root modulo $m$ and let $a,b\in\mathbb{Z}$ with $\gcd(a,m)=\gcd(b,m)=1$. Then
  \begin{enumerate}
    \item $\operatorname{ind}_r(1)=\varphi(m)$ (equivalently $\operatorname{ind}_r(1)\equiv 0\pmod{\varphi(m)}$).
    \item $\operatorname{ind}_r(r)=1$.
    \item $\operatorname{ind}_r(ab)\equiv \operatorname{ind}_r(a)+\operatorname{ind}_r(b)\pmod{\varphi(m)}$.
    \item For $n\in\mathbb{Z}_{>0}$: $\operatorname{ind}_r(a^n)\equiv n\operatorname{ind}_r(a)\pmod{\varphi(m)}$.
  \end{enumerate}
  \end{define}
  \begin{enumerate}
    \item $8x^7 \equiv 5 \pmod{13}$
    \begin{subanswer}
      A primitive root modulo $m$ is defined by \nameref{def:primroot}.
      The index is defined by \nameref{def:index}.
      The properties of indices are given by \nameref{prop:5.20}.
      From the previous problem (Exercise Set 5.3, \#24(c)), it is known that $2$ is a primitive root modulo $13$.
      Let $g = 2$ be a primitive root modulo $13$. Then $\varphi(13) = 12$.
      
      The congruence $8x^7 \equiv 5 \pmod{13}$ is equivalent to
      \[
      \operatorname{ind}_2(8) + 7\operatorname{ind}_2(x) \equiv \operatorname{ind}_2(5) \pmod{12}
      \]
      by \nameref{prop:5.20}.
      
      The indices are computed:
      \begin{itemize}
        \item $2^1 = 2 \equiv 2 \pmod{13}$, so $\operatorname{ind}_2(2) = 1$.
        \item $2^2 = 4 \equiv 4 \pmod{13}$, so $\operatorname{ind}_2(4) = 2$.
        \item $2^3 = 8 \equiv 8 \pmod{13}$, so $\operatorname{ind}_2(8) = 3$.
        \item $2^4 = 16 \equiv 3 \pmod{13}$, so $\operatorname{ind}_2(3) = 4$.
        \item $2^5 = 32 = 13 \cdot 2 + 6 \equiv 6 \pmod{13}$, so $\operatorname{ind}_2(6) = 5$.
        \item $2^6 = 64 = 13 \cdot 4 + 12 \equiv 12 \equiv -1 \pmod{13}$, so $\operatorname{ind}_2(12) = 6$.
        \item $2^7 = 2^6 \cdot 2 \equiv (-1) \cdot 2 = -2 \equiv 11 \pmod{13}$, so $\operatorname{ind}_2(11) = 7$.
        \item $2^8 = 2^7 \cdot 2 \equiv 11 \cdot 2 = 22 = 13 \cdot 1 + 9 \equiv 9 \pmod{13}$, so $\operatorname{ind}_2(9) = 8$.
        \item $2^9 = 2^8 \cdot 2 \equiv 9 \cdot 2 = 18 = 13 \cdot 1 + 5 \equiv 5 \pmod{13}$, so $\operatorname{ind}_2(5) = 9$.
      \end{itemize}
      
      Therefore, $\operatorname{ind}_2(8) = 3$ and $\operatorname{ind}_2(5) = 9$.

      \[
      3 + 7\operatorname{ind}_2(x) \equiv 9 \pmod{12},
      \]
      which simplifies to
      \[
      7\operatorname{ind}_2(x) \equiv 6 \pmod{12}.
      \]
      
      Since $\gcd(7, 12) = 1$, the congruence has a unique solution modulo $12$.
      Since $7 \cdot 7 = 49 = 12 \cdot 4 + 1$, it follows that $7^{-1} \equiv 7 \pmod{12}$.
      Therefore,
      \[
      \operatorname{ind}_2(x) \equiv 6 \cdot 7 = 42 \equiv 6 \pmod{12}.
      \]
      
      Since $\operatorname{ind}_2(x) \equiv 6 \pmod{12}$, it follows that $x \equiv 2^6 \equiv 12 \equiv -1 \pmod{13}$.
      Therefore, the unique solution is $x \equiv 12 \pmod{13}$, or equivalently, $x \equiv -1 \pmod{13}$.
    \end{subanswer}
    
    \item $7x^5 \equiv 2 \pmod{17}$
    \begin{subanswer}
      A primitive root modulo $m$ is defined by \nameref{def:primroot}.
      The index is defined by \nameref{def:index}.
      The properties of indices are given by \nameref{prop:5.20}.
      Euler's totient function is defined by \nameref{def:euler_totient}.
      Since $17$ is prime, it follows that $\varphi(17) = 16$. The prime divisors of $16$ are $2$ only.
      By Exercise Set 5.1, \#9, an integer $r$ with $17 \nmid r$ is a primitive root modulo $17$ if and only if $r^{16/2} = r^8 \not\equiv 1 \pmod{17}$.
      
      It is verified that $3$ is a primitive root modulo $17$:
      \begin{itemize}
        \item $3^2 = 9 \equiv 9 \pmod{17}$.
        \item $3^4 = (3^2)^2 = 9^2 = 81 = 17 \cdot 4 + 13 \equiv 13 \pmod{17}$.
        \item $3^8 = (3^4)^2 = 13^2 = 169 = 17 \cdot 9 + 16 \equiv 16 \equiv -1 \pmod{17}$.
      \end{itemize}
      Since $3^8 \equiv -1 \not\equiv 1 \pmod{17}$, it follows that $3$ is a primitive root modulo $17$.
      
      Let $g = 3$ be a primitive root modulo $17$. Then $\varphi(17) = 16$.
      The congruence $7x^5 \equiv 2 \pmod{17}$ is equivalent to
      \[
      \operatorname{ind}_3(7) + 5\operatorname{ind}_3(x) \equiv \operatorname{ind}_3(2) \pmod{16}
      \]
      by \nameref{prop:5.20}.
      
      The indices are computed:

      Since the order of $x$ modulo $p$ divides $\varphi(p) = p-1$ by \nameref{thm:euler}, it follows that $8 \mid p-1$.
      Therefore, $p \equiv 1 \pmod{8}$.
      
      ($\Leftarrow$) Suppose $p \equiv 1 \pmod{8}$ for the sake of direct proof.
      It is shown that $x^4 \equiv -1 \pmod{p}$ is solvable.
      Since $p \equiv 1 \pmod{8}$, there exists a primitive root $g$ modulo $p$ by \nameref{def:primroot}.
      Since $8 \mid p-1$, let $h = g^{(p-1)/8}$.
      Then $h^8 = g^{p-1} \equiv 1 \pmod{p}$ by \nameref{thm:fermat}.
      Since $g$ is a primitive root modulo $p$, by \nameref{def:primroot}, the order of $g$ modulo $p$ is $\varphi(p) = p-1$.
      Therefore, the order of $h = g^{(p-1)/8}$ modulo $p$ is exactly $8$ (since if the order were a proper divisor $d$ of $8$, then $g^{(p-1)/8 \cdot d} \equiv 1 \pmod{p}$, which would imply the order of $g$ divides $(p-1)/8 \cdot d < p-1$, a contradiction).

      Since $(h^4)^2 = h^8 \equiv 1 \pmod{p}$ and $h^4 \not\equiv 1 \pmod{p}$, it follows that $h^4 \equiv -1 \pmod{p}$.
      Therefore, $x = h$ is a solution to $x^4 \equiv -1 \pmod{p}$.
      
      Therefore, by ($\Rightarrow$) and ($\Leftarrow$), the congruence $x^4 \equiv -1 \pmod{p}$ is solvable if and only if $p \equiv 1 \pmod{8}$.
    \end{subprove}

    2z = a^2 + 3 b^2 \implies z = \frac{a^2 + 3 b^2}{2},
    \]
    and subtracting gives
    \[
    2x = 3 b^2 - a^2 \implies x = \frac{3 b^2 - a^2}{2}.
    \]
  \end{subanswer}




  \item Exercise Set 6.4, \#21. 
  Prove that the Diophantine equation $x^4-y^4=z^2$ has no solutions in non-zero integers $x, y, z$.
  [Hint: The proof of Theorem 6.4 (\inlinequote{Fermat Descent}) might give you 
  some ideas on how to get started.]
  \begin{subprove}
    Theorem 6.4 (Fermat Descent) is given by \nameref{thm:6.4}.
    
    Suppose there exist non-zero integers $x, y, z$ such that $x^4 - y^4 = z^2$ for the sake of contradiction.
    Without loss of generality, assume that $\gcd(x, y, z) = 1$ (if not, divide through by $\gcd(x, y, z)$).
    Therefore, $(x, y, z)$ is a primitive solution.
    
    Rearranging the equation gives
    \[
    x^4 - y^4 = z^2 \implies (x^2 - y^2)(x^2 + y^2) = z^2.
    \]
    
    Since $\gcd(x^2 - y^2, x^2 + y^2) = \gcd(2y^2, x^2 + y^2) = 1$ or $2$, and both factors are positive integers, it follows that both $x^2 - y^2$ and $x^2 + y^2$ are perfect squares.
    
    Let $x^2 - y^2 = a^2$ and $x^2 + y^2 = b^2$ for some integers $a, b$.
    Adding these equations gives
    \[
    2x^2 = a^2 + b^2 \implies x^2 = \frac{a^2 + b^2}{2},
    \]
    and subtracting gives
    \[
    2y^2 = b^2 - a^2 \implies y^2 = \frac{b^2 - a^2}{2}.
    \]
    
    However, this leads to a contradiction because it implies that both $x$ and $y$ are rational numbers unless $a$ and $b$ are both even or both odd.
    If both are even, then we can reduce the equation further, contradicting the assumption of primitiveness.
    If both are odd, then $b^2 - a^2$ is divisible by $4$, which contradicts the form of $y^2$.
    
    Therefore, there are no non-zero integer solutions for $x, y, z$.
    This contradicts the assumption that such integers exist.\vspace*{10px}\\
    Therefore, by contradiction, the Diophantine equation $x^4 - y^4 = z^2$ has no solutions in non-zero integers $x, y, z$.
  \end{subprove}



  \item Exercise Set 6.4, \#27.

  [Hint: Consider $x=n^3-3 n$ and $y=3 n^2-1$ for $n \in \mathbb{Z}$.]

    \[
    x = n^3 - 3n, \quad y = 3n^2 - 1.

    First, compute $x^2 + y^2$:
    \[
    x^2 = (n^3 - 3n)^2 = n^6 - 6n^4 + 9n^2,
    \]
    \[
    y^2 = (3n^2 - 1)^2 = 9n^4 - 6n^2 + 1.
    \]

    \[
    x^2 + y^2 = (n^6 - 6n^4 + 9n^2) + (9n^4 - 6n^2 + 1) = n^6 + 3n^4 + 3n^2 + 1.
    \]
    
    Next, compute $z^3$:
    \[
    z = n^2 - 1,
    \]
    \[
    z^3 = (n^2 - 1)^3 = n^6 - 3n^4 + 3n^2 - 1.
    \]

    \[
    x^2 + y^2 = n^6 + 3n^4 + 3n^2 + 1,
    \]
    \[
    z^3 = n^6 - 3n^4 + 3n^2 - 1.

  \end{subprove}
