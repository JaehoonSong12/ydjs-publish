\textbf{Remark} The following definitions (or results) are already proved in class or on previous ICA, Wbwk, Hmwk assignments, and contents of the textbook.
\begin{define}{Theorem 4.4 (Euler's Criterion)}[thm:4.4][red]
  Let $p$ be an odd prime and let $a\in\mathbb{Z}$ with $p\nmid a$. Then
  \[
  \left(\frac{a}{p}\right) \equiv a^{\frac{p-1}{2}} \pmod p,
  \]
  and, since both sides take values in $\{\pm1\}$, we may (and usually do) interpret this congruence as the equality
  \[
  \left(\frac{a}{p}\right) = a^{\frac{p-1}{2}} \pmod p \in\{1,-1\}.
  \]
  \begin{subprove}
    If $\left(\frac{a}{p}\right)=1$, there exists $x_0$ with $x_0^2\equiv a\pmod p$. Then
    \[
    a^{\frac{p-1}{2}} \equiv (x_0^2)^{\frac{p-1}{2}} \equiv x_0^{\,p-1}\equiv 1\pmod p
    \]
    by Fermat's little theorem.
    
    If $\left(\frac{a}{p}\right)=-1$, the nonzero residues $\{1,2,\dots,p-1\}$ partition into $(p-1)/2$ pairs with product $a$: if $i^2\equiv a\pmod p$ for some $i$, then $a$ is a residue, contradiction. Hence $(p-1)!\equiv a^{\frac{p-1}{2}} \pmod p$. By Wilson's theorem, $(p-1)!\equiv -1\pmod p$, so $a^{\frac{p-1}{2}}\equiv -1\pmod p$.
  \end{subprove}
\end{define}
\begin{define}{Lemma 4.7 (Gauss's Lemma)}[lem:4.7][red]
  Let $p$ be an odd prime and let $a\in\mathbb{Z}$ with $p\nmid a$. For $k=1,2,\dots,\frac{p-1}{2}$ let $r_k$ be the least positive residue of $ka$ modulo $p$ (i.e., $r_k\in\{1,\dots,p-1\}$ and $r_k\equiv ka\pmod p$). Let $n$ be the number of these residues $r_k$ that are greater than $p/2$. Then
  \[
  \left(\frac{a}{p}\right)=(-1)^n.
  \]
  \begin{subprove}
  Let $p$ be an odd prime and $a\in\mathbb{Z}$ with $p\nmid a$. For $k=1,2,\dots,\frac{p-1}{2}$, let $r_k$ be the least positive residue of $ka$ modulo $p$. Partition into $\{r_1,\dots,r_n\}$ (those $r_k>p/2$) and $\{s_1,\dots,s_m\}$ (those $s_k<p/2$), so $n+m=(p-1)/2$.

  Consider $p-r_1,\dots,p-r_n,s_1,\dots,s_m$. These lie in $\{1,\dots,(p-1)/2\}$ and are pairwise distinct modulo $p$: distinct $r_k,s_k$ by construction; if $p-r_i\equiv s_j\pmod p$, then $r_i+s_j\equiv 0\pmod p$ gives $(k_i+k_j)a\equiv 0\pmod p$, forcing $k_i+k_j=p$ (contradiction).

  Therefore, these form a permutation of $\{1,\dots,(p-1)/2\}$:
  \[
  \prod_{i=1}^n (p-r_i)\cdot\prod_{j=1}^m s_j \equiv \left(\frac{p-1}{2}\right)! \pmod p.
  \]
  Since $p-r_i \equiv -r_i\pmod p$ and $\{r_1,\dots,r_n,s_1,\dots,s_m\}$ equals the least positive residues of $a,2a,\dots,\frac{p-1}{2}a$:
  \[
  (-1)^n a^{\frac{p-1}{2}}\left(\frac{p-1}{2}\right)! \equiv \left(\frac{p-1}{2}\right)! \pmod p.
  \]
  Canceling the factorial yields $a^{\frac{p-1}{2}}\equiv (-1)^n\pmod p$. By \nameref{thm:4.4}, $\left(\frac{a}{p}\right)=(-1)^n$.
  \end{subprove}
\end{define}

  \item Exercise Set 4.1, \#5. Find all incongruent solutions of each quadratic congruence below.
  \begin{enumerate}
    \item $x^2 \equiv 23 \pmod{77}$. \\
    \quad[Hint: consider the congruences $x^2\equiv 23\pmod{7}$ and $x^2\equiv 23\pmod{11}$ and use the Chinese Remainder Theorem.]
    \begin{subanswer}
      Since $77=7\cdot 11$ with $\gcd(7,11)=1$, by the Chinese Remainder Theorem the congruence $x^2\equiv 23\pmod{77}$ is equivalent to the system $x^2\equiv 23\pmod{7}$ and $x^2\equiv 23\pmod{11}$.
      \begin{itemize}
        \item Modulo $7$, it is found that $23\equiv 2\pmod{7}$. The quadratic residues modulo $7$ are $1^2\equiv 1$, $2^2\equiv 4$, $3^2\equiv 2\pmod{7}$. Therefore, $x^2\equiv 2\pmod{7}$ has solutions $x\equiv \pm 3\equiv 3,4\pmod{7}$.
        \item Modulo $11$, it is found that $23\equiv 1\pmod{11}$. Since $1\equiv 1^2\pmod{11}$, the congruence $x^2\equiv 1\pmod{11}$ has solutions $x\equiv \pm 1\equiv 1,10\pmod{11}$.
      \end{itemize}
      By the Chinese Remainder Theorem, $2\cdot 2=4$ incongruent solutions modulo $77$ exist.
      \begin{align*}
        x &\equiv 3\pmod{7},\quad x\equiv 1\pmod{11} \THEN x\equiv 10\pmod{77},\\
        x &\equiv 3\pmod{7},\quad x\equiv 10\pmod{11} \THEN x\equiv 45\pmod{77},\\
        x &\equiv 4\pmod{7},\quad x\equiv 1\pmod{11} \THEN x\equiv 32\pmod{77},\\
        x &\equiv 4\pmod{7},\quad x\equiv 10\pmod{11} \THEN x\equiv 67\pmod{77}.
      \end{align*}
      Therefore, the solutions are $x\equiv 10,32,45,67\pmod{77}$.
    \end{subanswer}
    \item $x^2 \equiv 11 \pmod{39}$.
    \begin{subanswer}
      Since $39=3\cdot 13$ with $\gcd(3,13)=1$, the congruence $x^2\equiv 11\pmod{39}$ is equivalent to the system $x^2\equiv 11\pmod{3}$ and $x^2\equiv 11\pmod{13}$.
      \begin{itemize}
        \item Modulo $3$, it is found that $11\equiv 2\pmod{3}$. The quadratic residues modulo $3$ are $1^2\equiv 1\pmod{3}$ only, so $x^2\equiv 2\pmod{3}$ has no solutions.
      \end{itemize}
      Since one of the congruences has no solution, the original congruence $x^2\equiv 11\pmod{39}$ has no solutions.
    \end{subanswer}
    \item $x^2 \equiv 46 \pmod{105}$.
    \begin{subanswer}
      Since $105=3\cdot 5\cdot 7$, the congruence $x^2\equiv 46\pmod{105}$ is equivalent to the system $x^2\equiv 46\pmod{3}$, $x^2\equiv 46\pmod{5}$, and $x^2\equiv 46\pmod{7}$.
      \begin{itemize}
        \item Modulo $3$, $46\equiv 1\pmod{3}$, and $x^2\equiv 1\pmod{3}$ has solutions $x\equiv 1,2\pmod{3}$.
        \item Modulo $5$, $46\equiv 1\pmod{5}$, and $x^2\equiv 1\pmod{5}$ has solutions $x\equiv 1,4\pmod{5}$.

      \end{itemize}
      Combining with $p\equiv 1$ or $3\pmod{4}$ yields $\left(\frac{3}{p}\right) = 1$ if and only if $p\equiv 1,11\pmod{12}$, i.e., $p\equiv \pm 1\pmod{12}$.
    \end{subprove}
    \item $\left(\frac{-3}{p}\right)=1$ if and only if $p \equiv 1\pmod{6}$.
    \begin{subprove}
      By multiplicativity, $\left(\frac{-3}{p}\right) = \left(\frac{-1}{p}\right)\left(\frac{3}{p}\right)$.
      By \nameref{thm:4.6}, it is known that $\left(\frac{-1}{p}\right) = (-1)^{\frac{p-1}{2}}$.
      By the proof of part (b), it is known that $\left(\frac{3}{p}\right) = (-1)^{\frac{p-1}{2}}\left(\frac{p}{3}\right)$.
      Substitute these into the product
      \[
      \left(\frac{-3}{p}\right) = \left( (-1)^{\frac{p-1}{2}} \right) \cdot \left( (-1)^{\frac{p-1}{2}}\left(\frac{p}{3}\right) \right)
      \]
      \[
      \left(\frac{-3}{p}\right) = \left( (-1)^{\frac{p-1}{2}} \right)^2 \left(\frac{p}{3}\right)
      \]
      Since $\left( (-1)^{\frac{p-1}{2}} \right)^2 = (-1)^{p-1} = 1$ (as $p-1$ is even), the expression simplifies to
      \[
      \left(\frac{-3}{p}\right) = \left(\frac{p}{3}\right).
      \]
      The condition $\left(\frac{-3}{p}\right) = 1$ is equivalent to $\left(\frac{p}{3}\right) = 1$ for all primes $p>3$.
      The Legendre symbol $\left(\frac{p}{3}\right) = 1$ if and only if $p \equiv 1 \pmod 3$.
      Since $p$ is a prime greater than $3$, it cannot be $p \equiv 0 \pmod 3$. The only other possibility is $p \equiv 2 \pmod 3$ (which is $p \equiv 5 \pmod 6$).
      Thus, the condition $p \equiv 1 \pmod 3$ is identical to the condition $p \equiv 1 \pmod 6$ for all primes $p>3$.\vspace*{10px}\\
      Therefore, $\left(\frac{-3}{p}\right) = 1$ if and only if $p \equiv 1\pmod 6$.
    \end{subprove}
  \end{enumerate}
\end{enumerate}
