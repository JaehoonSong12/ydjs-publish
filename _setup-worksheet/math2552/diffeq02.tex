  \item \textbf{Repeated Eigenvalues --- Initial Value Problem} (10 points)\\
  Consider the initial value problem
  $$
  \overrightarrow{\boldsymbol{y}}^{\prime}=\left[\begin{array}{rr}
  -6 & 1 \\
  0 & -6
  \end{array}\right] \overrightarrow{\boldsymbol{y}}, \quad \overrightarrow{\boldsymbol{y}}(0)=\left[\begin{array}{l}
  -4 \\
  -2
  \end{array}\right]
  $$
  \begin{enumerate}[label=(\alph*)]
    \item Find the eigenvalue $\lambda$, an eigenvector $\overrightarrow{\boldsymbol{v}_1}$, and a generalized eigenvector $\overrightarrow{\boldsymbol{v}_2}$ for the coefficient matrix of this linear system.
    \begin{subanswer}
      The coefficient matrix is
      \[
        A=\begin{bmatrix}-6 & 1\\[4pt]0 & -6\end{bmatrix}.
      \]
      By the characteristic polynomial, $(\lambda+6)^2$, the repeated eigenvalue is defined as follows.
      \[
        \lambda=-6.
      \]
      Thus, \(\overrightarrow{v_1}\in\operatorname{null}(A+6I)\) where
      \[
        A+6I=\begin{bmatrix}0 & 1\\[4pt]0 & 0\end{bmatrix}
      \]
      Therefore,
      \[
        \overrightarrow{v_1}=\begin{bmatrix}1\\[4pt]0\end{bmatrix}.
      \]
      Moreover, a generalized eigenvector $\overrightarrow{v_2}$ satisfies 
      $(A+6I)\overrightarrow{v_2}=\overrightarrow{v_1}$. Thus,
      \[
        \overrightarrow{v_2}=\begin{bmatrix}0\\[4pt]1\end{bmatrix}.
      \]
    \end{subanswer}

    \item Find the most general real-valued solution to the linear system of differential equations. Use $t$ as the independent variable in your answers.
    \begin{subanswer}
      For a repeated eigenvalue with one eigenvector and one generalized eigenvector, the general real solution is
      \[
        \overrightarrow{y}(t)=e^{\lambda t}\big(C_1\overrightarrow{v_1}+C_2(t\overrightarrow{v_1}+\overrightarrow{v_2})\big).
      \]
      , that is, 
      \[
        \overrightarrow{y}(t)=e^{-6t}\Big(C_1\begin{bmatrix}1\\[4pt]0\end{bmatrix}+C_2\Big(t\begin{bmatrix}1\\[4pt]0\end{bmatrix}+\begin{bmatrix}0\\[4pt]1\end{bmatrix}\Big)\Big)
        =e^{-6t}\begin{bmatrix}C_1+C_2 t\\[6pt]C_2\end{bmatrix},
      \]
      where $C_1,C_2\in\mathbb{R}$ are arbitrary constants.
    \end{subanswer}

    \item Solve the original initial value problem.
    \begin{subanswer}
      Apply $\overrightarrow{y}(0)=\begin{bmatrix}-4\\[4pt]-2\end{bmatrix}$. From the general solution at $t=0$,
      \[
        \overrightarrow{y}(0)=\begin{bmatrix}C_1\\[4pt]C_2\end{bmatrix}=\begin{bmatrix}-4\\[4pt]-2\end{bmatrix},
      \]
      so $C_1=-4$ and $C_2=-2$. Therefore the IVP solution is
      \[
        \boxed{\displaystyle
          \overrightarrow{y}(t)=e^{-6t}\begin{bmatrix}-4-2t\\[6pt]-2\end{bmatrix}
          \quad\text{or}\quad
          y_1(t)=(-4-2t)e^{-6t},\; y_2(t)=-2e^{-6t}.
        }
      \]
    \end{subanswer}
  \end{enumerate}
