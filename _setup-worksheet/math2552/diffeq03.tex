\textbf{Definition.} For $f(t)$ defined for $t \ge 0$, piecewise continuous and of exponential order:
$$
F(s) = \mathcal{L}\{f(t)\} = \int_{0}^{\infty} e^{-st} f(t) \, dt.
$$
\textbf{Functional transforms.}
\renewcommand{\arraystretch}{1.8}%
\begin{center}
\begin{tabularx}{\textwidth}{|X|c|X|}
  \hline
  \textbf{$f(t)$} & \textbf{$F(s)=\mathcal{L}\{f(t)\}$} & \textbf{ROC} \\
  \hline
  $\displaystyle 1$ & $\displaystyle \frac{1}{s}$ & $\displaystyle s > 0$ \\
  $\displaystyle t^n$ ($n \in \mathbb{N}$) & $\displaystyle \frac{n!}{s^{n+1}}$ & $\displaystyle s > 0$ \\
  $\displaystyle e^{at}$ & $\displaystyle \frac{1}{s-a}$ & $\displaystyle s > a$ \\
  $\displaystyle \sin(at)$ & $\displaystyle \frac{a}{s^2 + a^2}$ & $\displaystyle s > 0$ \\
  $\displaystyle \cos(at)$ & $\displaystyle \frac{s}{s^2 + a^2}$ & $\displaystyle s > 0$ \\
  $\displaystyle \sinh(at)$ & $\displaystyle \frac{a}{s^2 - a^2}$ & $\displaystyle s > |a|$ \\
  $\displaystyle \cosh(at)$ & $\displaystyle \frac{s}{s^2 - a^2}$ & $\displaystyle s > |a|$ \\
  $\displaystyle t^n e^{at}$ & $\displaystyle \frac{n!}{(s-a)^{n+1}}$ & $\displaystyle s > a$ \\
  $\displaystyle e^{at} \sin(bt)$ & $\displaystyle \frac{b}{(s-a)^2 + b^2}$ & $\displaystyle s > a$ \\
  $\displaystyle e^{at} \cos(bt)$ & $\displaystyle \frac{s-a}{(s-a)^2 + b^2}$ & $\displaystyle s > a$ \\
  \hline
\end{tabularx}
\end{center}

\textbf{Operational properties.}
\begin{itemize}[noitemsep,topsep=2pt]
  \item \textbf{Linearity:}
  $$\mathcal{L}\{a f(t) + b g(t)\} = a F(s) + b G(s).$$
  \item \textbf{First shifting (frequency):}
  $$\mathcal{L}\{e^{at} f(t)\} = F(s-a).$$
  \item \textbf{Second shifting (time):} With unit step $u(t-a)$:
  $$\mathcal{L}\{f(t-a)u(t-a)\} = e^{-as} F(s); \qquad \mathcal{L}\{g(t)u(t-a)\} = e^{-as} \mathcal{L}\{g(t+a)\}.$$
  \item \textbf{Differentiation of transform:}
  $$\mathcal{L}\{t^n f(t)\} = (-1)^n \frac{d^n F}{ds^n}(s).$$
  \item \textbf{Integration of transform:} If $\displaystyle\lim_{t \to 0^+} \frac{f(t)}{t}$ exists:
  $$\mathcal{L}\left\{\frac{f(t)}{t}\right\} = \int_{s}^{\infty} F(\sigma) \, d\sigma.$$
\end{itemize}

\textbf{Transforms of derivatives (IVPs).}
$$
\mathcal{L}\{f'\} = s F(s) - f(0); \quad \mathcal{L}\{f''\} = \mathcal{L}\{f\} \cdot \group{s^2} - f(0) \cdot \group{s} - f'(0); \quad \mathcal{L}\{f^{(n)}\} = s^n F(s) - s^{n-1} f(0) - \cdots - f^{(n-1)}(0).
$$
\textbf{Integral:} $\mathcal{L}\bigl\{\displaystyle\int_0^t f(\tau)\,d\tau\bigr\} = \dfrac{F(s)}{s}$.

\textbf{Convolution \& special.} $(f * g)(t) = \displaystyle\int_0^t f(\tau)g(t-\tau)\,d\tau$; $\mathcal{L}\{f * g\} = F(s)G(s)$. Dirac delta: $\mathcal{L}\{\delta(t - t_0)\} = e^{-s t_0}$ ($t_0 > 0$). Periodic $f$ (period $T$): $\mathcal{L}\{f\} = \dfrac{1}{1 - e^{-sT}} \displaystyle\int_0^T e^{-st} f(t)\,dt$.
