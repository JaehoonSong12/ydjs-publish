The following sections provide guidance on expressing logical statements 
in natural language and methods for proving implications.

\begin{define}{Implication in Natural Language}\phantomsection
  \label{def:implication-natural-language} 
  % \hyperref[def:implication-natural-language]{YOUR_CONTEXT}
  Instead of using logical symbols like $p \to q$ in proofs, use natural language:
  \begin{enumerate}
    \item If $p$, then $q$
    \item $q$ if $p$
    \item $q$ only if $p$
    \item $p$ then $q$
    \item $q$ whenever $p$
    \item $q$ is necessary for $p$
    \item $p$ is sufficient for $q$
  \end{enumerate}
\end{define}

\begin{define}{Biconditional in Natural Language}\phantomsection
  \label{def:biconditional-natural-language} 
  % \hyperref[def:biconditional-natural-language]{YOUR_CONTEXT}
  Instead of $p \leftrightarrow q$, use:
  \begin{enumerate}
    \item $p$ if and only if $q$
    \item $p$ is necessary and sufficient for $q$
  \end{enumerate}
\end{define}

When proving implications, there are three standard methods available:

\begin{define}{Methods to Prove Implications}\phantomsection
  \label{def:methods-prove-implications} 
  % \hyperref[def:methods-prove-implications]{YOUR_CONTEXT}
  To prove an implication $p \to q$, you can use any of these three methods:
  \begin{enumerate}
    \item \textbf{Direct Proof:} Assume $p$, prove $q$.
    \item \textbf{Contrapositive:} Assume $\sNOT{q}$, prove $\sNOT{p}$.
    \item \textbf{Contradiction:} Assume $p \wedge \sNOT{q}$, prove $\FALSE$ (contradiction).
  \end{enumerate}
  
  For predicates $P(x)$ and $Q(x)$ with domain $D$, to prove $P(x) \to Q(x)$ for all $x \in D$:
  \begin{enumerate}
    \item \textbf{Direct Proof:} Let $x \in D$. Assume $P(x)$ is true, prove $Q(x)$ is true.
    \item \textbf{Contrapositive:} Let $x \in D$. Assume $Q(x)$ is false, prove $P(x)$ is false.
    \item \textbf{Contradiction:} Let $x \in D$. Assume $P(x)$ is true and $Q(x)$ is false, prove a contradiction.
  \end{enumerate}
\end{define}

The following definitions extend the understanding of quantified statements and their negations.

\begin{define}{Prove or Disprove by Counterexample}\phantomsection
  \label{def:counterexample} 
  % \hyperref[def:counterexample]{YOUR_CONTEXT}
  Let $P(x)$ be a predicate with domain of discourse $\opD$. The 
  universally quantified proposition, $\forall x \in \opD, P(x)$
  is false if and only if the existentially quantified proposition 
  $\exists x \in \opD, \sNOT{P(x)}$, is true.
\end{define}

The following claims demonstrate methods for proving existential statements and disproving universal statements.

\begin{claim}{Even Implies Odd: If $n^2+3n$ is even, then $n$ is odd.}\phantomsection
  \label{claim:even-implies-odd} 
  % \hyperref[claim:even-implies-odd]{YOUR_CONTEXT}
  \begin{subdisprove}[bluecolor]
    Let $n\in \setZ$. 
    Consider $n=2$. Then $n^2+3n=2^2+3(2)=4+6=10$ is even, 
    but $n=2$ is also even, not odd. Since we found a case where the hypothesis 
    is true but the conclusion is false, the statement is disproven.
  \end{subdisprove}
\end{claim}

\begin{claim}{Integer with Equal Cube and Square: There exists an integer whose cube equals its square.}\phantomsection\label{claim:integer-cube-square} % \hyperref[claim:integer-cube-square]{YOUR_CONTEXT}
  \begin{subprove}[bluecolor]
    Consider the integer $1$. We have $1^2=1$ and $1^3=1$, so $1^2=1^3=1$.
    Therefore, there exists an integer (namely $1$) whose cube equals its square.
  \end{subprove}
\end{claim}

\begin{claim}{Proof by Example: Sum of Three Odd Integers}\phantomsection
  \label{claim:sum-three-odd-integers} 
  % \hyperref[claim:sum-three-odd-integers]{YOUR_CONTEXT}
  There exist odd integers $a$, $b$, and $c$ such that $a+b+c=1$.
  \begin{subprove}[bluecolor]
    Consider $(a,b,c)=(5,-3,-1)$. Each of these numbers is odd 
    since $5=2\cdot 2 + 1$, $-3=2\cdot (-2) + 1$, and $-1=2\cdot (-1) + 1$. 
    Thus, $5+(-3)+(-1)=1$. Therefore, there exist odd integers whose sum is $1$.
  \end{subprove}
\end{claim}
