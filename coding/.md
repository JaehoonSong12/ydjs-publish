C Implementation (staircase.c)This version uses manual memory management (malloc, memset) and pointer arithmetic for a standard C approach.#include <stdio.h>
#include <stdlib.h> // Required for malloc, free
#include <string.h> // Required for memset

/**
 * Prints a right-aligned staircase of '#' characters of height n.
 * * This function manually manages memory to construct the string buffer
 * for each row. It allocates memory once, but rewrites the buffer contents
 * on every iteration using pointer arithmetic.
 *
 * @param n The height and width of the staircase.
 */
void staircase(int n) {
    // Allocate memory on the heap for the string line.
    // Size is n (for characters) + 1 (for the Null Terminator '\0').
    char* rowStr = (char*)malloc((n + 1) * sizeof(char));

    // Safety check: if malloc fails (returns NULL), exit to prevent crash.
    if (rowStr == NULL) return; 

    // Set the null terminator at the very end of the buffer.
    // This marks the end of the string for printf.
    rowStr[n] = '\0';

    for (int row = 1; row <= n; row++) {
        int spaces = n - row; // Number of spaces needed for alignment
        int hashes = row;     // Number of hash marks to print

        // 1. Fill the beginning of the buffer with spaces.
        if (spaces > 0) {
            memset(rowStr, ' ', spaces);
        }
        
        // 2. Fill the rest of the buffer with hashes.
        // Pointer Arithmetic: 'rowStr + spaces' calculates the memory address 
        // where the spaces end, so we begin writing hashes exactly there.
        memset(rowStr + spaces, '#', hashes);

        // Print the constructed string row.
        printf("%s\n", rowStr);
    }

    // Free the allocated memory to prevent memory leaks.
    free(rowStr);
}

int main() {
    // Test the function with size 6
    staircase(6);
    return 0;
}
2. Java Implementation (StaircasePrinter.java)This version uses String.repeat (Java 11+) for a clean, idiomatic implementation.public class StaircasePrinter {

    /**
     * Prints a right-aligned staircase of '#' symbols to the standard output.
     * <p>
     * This method utilizes the String.repeat() method (Java 11+) for efficient 
     * character repetition without nested loops.
     * </p>
     *
     * @param n the height of the staircase; must be greater than 0.
     */
    public static void staircase(int n) {
        for (int row = 1; row <= n; row++) {
            // Calculate the number of spaces required for right alignment
            int spacesCount = n - row;
            
            // Construct the string components
            // " ".repeat(count) creates a string of 'count' spaces efficiently
            String spaces = " ".repeat(spacesCount);
            String hashes = "#".repeat(row);
            
            // Combine and print the row
            System.out.println(spaces + hashes);
        }
    }

    /**
     * Main method to execute the staircase function.
     * @param args Command line arguments (not used).
     */
    public static void main(String[] args) {
        // Test the function with size 6
        staircase(6);
    }
}
3. Python Implementation (staircase.py)This version uses Python's string multiplication features.import sys

def staircase(n):
    """
    Prints a right-aligned staircase of '#' characters.

    This function utilizes Python's string multiplication feature 
    to generate the rows without explicit inner loops.

    Args:
        n (int): The size (height and width) of the staircase.
    """
    # Loop from 1 to n (inclusive)
    for row in range(1, n + 1):
        # Calculate padding spaces
        # Python allows string multiplication: " " * 3 results in "   "
        spaces_count = n - row
        
        # Create the parts of the string
        spaces = " " * spaces_count
        hashes = "#" * row
        
        # Concatenate and print using f-string
        print(f"{spaces}{hashes}")

if __name__ == "__main__":
    # Test the function with size 6
    staircase(6)
