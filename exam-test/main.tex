% Side-by-side request/response with \texttt{lstlisting} (CS-A: \texttt{rest.tex}). One tabular
% row (\texttt{m} columns); \verb|\httpmidarrows| sits between listings. Middle strip: \textbf{REQ}
% above, two short arrows, \textbf{RESP} below (compact gap so code columns get most of
% \verb|\linewidth|). Do not pass \texttt{lstlisting} through a macro argument.
\tikzset{%
  http@arrow/.style={%
    -{Stealth[scale=0.88]}, semithick, shorten >=0.5pt, shorten <=0.5pt, draw=black!55},
}
\newenvironment{httpexchange}{%
  \par\medskip
  \begingroup
  \hfuzz=30pt\relax
  \begin{center}%
  \noindent
  \begin{tabular}{@{} m{0.4\linewidth} >{\centering\arraybackslash}m{1.28cm} m{0.4\linewidth} @{}}%
}{%
  \end{tabular}%
  \end{center}%
  \endgroup
  \medskip
}
\newcommand{\httpmidarrows}{%
  &
  \begin{minipage}[c]{1.28cm}%
    \centering
    \normalfont\footnotesize REQ\par\vspace{1pt}%
    \begin{tikzpicture}[x=1cm, y=0.75cm]
      \draw[http@arrow] (0,0.1) -- (0.92,0.1);
      \draw[http@arrow] (0.92,-0.1) -- (0,-0.1);
    \end{tikzpicture}\par\vspace{1pt}%
    \normalfont\footnotesize RESP%
  \end{minipage}%
  &
}

% Inline blanks using \fillin (no \answerline question-number prefix).
% Space between printed answer and rule: exam default \answerclearance is tight; widen with:
\newlength{\fillinrulegap}
\setlength{\fillinrulegap}{0.55ex}
\makeatletter
\newcommand{\exam@fillinspaced}[2]{%
  \begingroup
  \setlength{\answerclearance}{\fillinrulegap}%
  \fillin[#1][#2]%
  \endgroup
}%
\newcommand{\labeledfillin}[2]{%
  \par\nobreak\vskip\answerskip
  \noindent#1:\ \exam@fillinspaced{#2}{2.3in}%
}%
% Optional arg = blank width (default 2.2in); mandatory arg = answer key for \printanswers.
\newcommand{\rightfillin}[2][2.2in]{%
  \par\nobreak\vskip\answerskip
  \noindent\hfill\exam@fillinspaced{#2}{#1}%
}%
% Visual-only underlined blank in running text (e.g. MCQ stems). Does not print an answer.
% For blanks that should show the key when \printanswers is on, use \fillin[Answer][width]
% or \exam@fillinspaced{Answer}{width} (see above). The exam class has no global option
% to turn every question into fill-in; use \fillin / \makeemptybox / \answerline per item.
\newcommand{\blank}[1][6em]{\underline{\hspace{#1}}}%
\makeatother

% Per-module cover (own title, name box, instructions). Use before each \begin{questions} block.
% Args: #1 = course/series, #2 = module title, #3 = one-line blurb (empty {} if none).
\newcommand{\exammodulecover}[3]{%
  \clearpage
  \thispagestyle{empty}%
  \vspace*{0.35in}%
  \begin{center} 
    {\Large \textbf{#1}\par}
    \vspace{1.1em}
    {\LARGE \textbf{#2}\par}
    \vspace{0.65em}
    {\large \textbf{\today}\par}
    \if\relax\detokenize{#3}\relax\else
      \vspace{0.9em}
      {\normalsize #3\par}%
    \fi
  \end{center}
  \vspace{1.1cm}
  \begin{center}
    \fbox{%
      \begin{minipage}{0.92\linewidth}
        \vspace{2mm}
        \textbf{Name:}\ \hrulefill \\
        \textbf{Student ID:}\ \hrulefill \quad
        \textbf{Section:}\ \hrulefill
        \vspace{2mm}
      \end{minipage}%
    }%
  \end{center}
  \vspace{4mm}
  \begin{center}
    \textbf{Instructions}
  \end{center}
  \begin{itemize}
    \item Verify that your copy has \textbf{all \numpages\ pages}.
    \item Show all work where space is provided; partial credit may be awarded.
    \item Do not use calculators unless your instructor says otherwise.
    \item Total possible points: \textbf{\numpoints}.
  \end{itemize}
  \clearpage
}
