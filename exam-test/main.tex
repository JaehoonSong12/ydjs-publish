% Inline blanks using \fillin (no \answerline question-number prefix).
% Space between printed answer and rule: exam default \answerclearance is tight; widen with:
\newlength{\fillinrulegap}
\setlength{\fillinrulegap}{0.55ex}
\makeatletter
\newcommand{\exam@fillinspaced}[2]{%
  \begingroup
  \setlength{\answerclearance}{\fillinrulegap}%
  \fillin[#1][#2]%
  \endgroup
}%
\newcommand{\labeledfillin}[2]{%
  \par\nobreak\vskip\answerskip
  \noindent#1:\ \exam@fillinspaced{#2}{2.3in}%
}%
% Optional arg = blank width (default 2.2in); mandatory arg = answer key for \printanswers.
\newcommand{\rightfillin}[2][2.2in]{%
  \par\nobreak\vskip\answerskip
  \noindent\hfill\exam@fillinspaced{#2}{#1}%
}%
\makeatother

% Per-module cover (own title, name box, instructions). Use before each \begin{questions} block.
% Args: #1 = course/series, #2 = module title, #3 = one-line blurb (empty {} if none).
\newcommand{\exammodulecover}[3]{%
  \clearpage
  \thispagestyle{empty}%
  \vspace*{0.35in}%
  \begin{center}
    {\Large \textbf{#1}\par}
    \vspace{1.1em}
    {\LARGE \textbf{#2}\par}
    \vspace{0.65em}
    {\large \textbf{\today}\par}
    \if\relax\detokenize{#3}\relax\else
      \vspace{0.9em}
      {\normalsize #3\par}%
    \fi
  \end{center}
  \vspace{1.1cm}
  \begin{center}
    \fbox{%
      \begin{minipage}{0.92\linewidth}
        \vspace{2mm}
        \textbf{Name:}\ \hrulefill \\
        \textbf{Student ID:}\ \hrulefill \quad
        \textbf{Section:}\ \hrulefill
        \vspace{2mm}
      \end{minipage}%
    }%
  \end{center}
  \vspace{4mm}
  \begin{center}
    \textbf{Instructions}
  \end{center}
  \begin{itemize}
    \item Verify that your copy has \textbf{all \numpages\ pages}.
    \item Show all work where space is provided; partial credit may be awarded.
    \item Do not use calculators unless your instructor says otherwise.
    \item Total possible points: \textbf{\numpoints}.
  \end{itemize}
  \clearpage
}
