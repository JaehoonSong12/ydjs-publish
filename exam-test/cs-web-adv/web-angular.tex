\question[4]
TRUE or FALSE: In an Angular single-page application (SPA), the browser requests a brand-new HTML document from the server on every route change.
\begin{choices}
  \choice TRUE
  \CorrectChoice FALSE
\end{choices}
\begin{solution}
  An SPA loads \texttt{index.html} once. Angular's client-side \textbf{Router} swaps components inside the existing page and updates the URL without a full server round-trip for a new HTML document, which keeps navigation fast.
\begin{lstlisting}[style=htmlinline]
<!-- index.html loaded once; <app-root> hosts routed views -->
<app-root></app-root>
\end{lstlisting}
\begin{lstlisting}[style=ts]
// Router config maps paths to components, not separate .html pages
{ path: 'labs', component: LabsComponent }
\end{lstlisting}
\end{solution}

\question[4]
Which command creates a new Angular project named \texttt{my-app}?
\begin{choices}
  \choice \lstinline[style=bashinline]|ng serve my-app|
  \CorrectChoice \lstinline[style=bashinline]|ng new my-app|
  \choice \lstinline[style=bashinline]|ng generate my-app|
  \choice \lstinline[style=bashinline]|npm install my-app|
\end{choices}
\begin{solution}
  \lstinline[style=bashinline]|ng new| scaffolds a new workspace. \lstinline[style=bashinline]|ng serve| runs the dev server for an \emph{existing} project; \lstinline[style=bashinline]|ng generate| (or \lstinline[style=bashinline]|ng g|) adds artifacts inside a project.
\begin{lstlisting}[style=bash]
ng new my-app
cd my-app
ng serve
\end{lstlisting}
\end{solution}

  \choice FALSE
\end{choices}
\begin{solution}

\question[4]
An Angular \textbf{component} is best described as:
\begin{choices}
  \choice A npm package that replaces TypeScript
  \CorrectChoice A building block made of a template, selector, optional styles, and a TypeScript class
  \choice A SQL table used only on the server
  \choice The same thing as a Webpack loader
\end{choices}
\begin{solution}
  Each component typically includes an HTML \textbf{template} (view), a CSS \textbf{selector} (where it is inserted), optional \textbf{styles}, and a TypeScript \textbf{class} (data and behavior).
\begin{lstlisting}[style=ts]
@Component({
  selector: 'app-hero',
  templateUrl: './hero.component.html',
  styleUrls: ['./hero.component.css'],
})
export class HeroComponent {
  name = 'Ada';
}
\end{lstlisting}
\end{solution}

\question[4]
TRUE or FALSE: In Angular templates, property binding uses the syntax \lstinline[style=htmlinline]|(property)="value"|.
\begin{choices}
  \choice TRUE
  \CorrectChoice FALSE
\end{choices}
\begin{solution}
  \textbf{Property binding} flows data from the component class into the template with square brackets: \lstinline[style=htmlinline]|[property]="value"|. Parentheses \lstinline[style=htmlinline]|(event)="handler()"| are \textbf{event binding} (template to class). Two-way binding combines both: \lstinline[style=htmlinline]|[(ngModel)]="value"|.
\begin{lstlisting}[style=htmlinline]
<img [src]="photoUrl" alt="Profile">
<button (click)="save()">Save</button>
<input [(ngModel)]="title">
\end{lstlisting}
\end{solution}

\question[4]
\textbf{Event binding} is used to:
\begin{choices}
  \choice Display \lstinline[style=htmlinline]|{{ title }}| in the template
  \CorrectChoice Listen for user actions (clicks, keystrokes, etc.) and run component logic
  \choice Set the \lstinline[style=htmlinline]|src| attribute from the model
  \choice Import \lstinline[style=tsinline]|FormsModule|
\end{choices}
\begin{solution}
  Event binding flows template to model. Parentheses on the event name wire the DOM event to a handler.
\begin{lstlisting}[style=htmlinline]
<button (click)="onSave()">Save</button>
\end{lstlisting}
\begin{lstlisting}[style=ts]
onSave(): void {
  console.log('Saved');
}
\end{lstlisting}
\end{solution}

\question[4]
In Angular templates, \textbf{pipes} are used to:
\begin{choices}
  \choice Compile TypeScript into JavaScript at build time
  \CorrectChoice Transform displayed data from one form into another (dates, currency, case, etc.)
  \choice Replace HTTP calls to the backend
  \choice Define root module metadata only
\end{choices}
\begin{solution}
  Pipes format values \emph{in the view} without changing the underlying model. Examples: \lstinline[style=htmlinline]|date|, \lstinline[style=htmlinline]|currency|, \lstinline[style=htmlinline]|uppercase|, \lstinline[style=htmlinline]|async|.
\begin{lstlisting}[style=htmlinline]
<p>{{ today | date:'MM-dd-yyyy' }}</p>
<p>{{ price | currency:'USD' }}</p>
\end{lstlisting}
\end{solution}

\question[4]
Which is the correct \textbf{pipe syntax} in a template?
\begin{choices}
  \CorrectChoice \lstinline[style=htmlinline]!{{ text | pipeName }}!
  \choice \lstinline[style=htmlinline]![(text | pipeName)]!
  \choice \lstinline[style=htmlinline]|[[text : pipeName]]|
  \choice \lstinline[style=htmlinline]|((text : pipeName))|

\question[4]
To pass data from a parent component down to a child component, which decorator is used on the child? \emph{Select all that apply.}
\begin{checkboxes}
  \choice \lstinline[style=tsinline]|@Component()|
  \choice \lstinline[style=tsinline]|@Injectable()|
  \choice \lstinline[style=tsinline]|@Output()|
  \CorrectChoice \lstinline[style=tsinline]|@Input()|
\end{checkboxes}
\begin{solution}
  \lstinline[style=tsinline]|@Input()| marks a child property that the parent sets via property binding. \lstinline[style=tsinline]|@Output()| emits events upward with an \lstinline[style=tsinline]|EventEmitter|.
\begin{lstlisting}[style=ts]
@Component({ selector: 'app-child', template: `<p>{{ label }}</p>` })
export class ChildComponent {
  @Input() label = '';
}

// Parent: <app-child [label]="parentTitle"></app-child>
\end{lstlisting}
\end{solution}

\question[4]
In a parent--child relationship, data usually flows \textbf{down} with \lstinline[style=tsinline]|@Input| and flows \textbf{up} with:
\begin{choices}
  \choice \lstinline[style=htmlinline]|*ngFor| only
  \CorrectChoice \lstinline[style=tsinline]|@Output| and an \lstinline[style=tsinline]|EventEmitter|
  \choice \lstinline[style=htmlinline]|[(ngModel)]| on the parent selector only
  \choice \lstinline[style=tsinline]|bootstrap| in \lstinline[style=tsinline]|@NgModule|
\end{choices}
\begin{solution}
  The child raises a custom event; the parent listens with event binding on the child selector.
\begin{lstlisting}[style=ts]
@Output() picked = new EventEmitter<string>();
this.picked.emit('item-1');
\end{lstlisting}
\begin{lstlisting}[style=htmlinline]
<app-item (picked)="onPicked($event)"></app-item>
\end{lstlisting}
\end{solution}

\question[4]
TRUE or FALSE: The modern \lstinline[style=htmlinline]|@for| control flow requires a \lstinline[style=htmlinline]|track| expression for optimal rendering.
\begin{choices}
  \CorrectChoice TRUE
  \choice FALSE
\end{choices}
\begin{solution}
  In Angular's built-in control flow, \lstinline[style=htmlinline]|track| is \textbf{required}---omitting it is a compile error. The track expression lets change detection identify added, removed, or reordered items instead of re-rendering the entire list.
\begin{lstlisting}[style=htmlinline]
@for (item of items; track item.id) {
  <li>{{ item.name }}</li>
} @empty {
  <li>No items yet.</li>
}
\end{lstlisting}
\end{solution}

\question[4]
An RxJS \textbf{Observable} is best described as:
\begin{choices}
  \choice A single static JSON file in \lstinline[style=tsinline]|assets/|
  \CorrectChoice A lazy stream that can emit zero or more values over time and may complete or error
  \choice A CSS encapsulation mode
  \choice The same as a route \lstinline[style=tsinline]|path| string
\end{choices}
\begin{solution}
  Observers \lstinline[style=tsinline]|subscribe| with handlers for \textbf{next}, \textbf{error}, and \textbf{complete}. \lstinline[style=tsinline]|HttpClient| and many Angular APIs return Observables.
\begin{lstlisting}[style=ts]
this.http.get<Hero[]>('/api/heroes').subscribe({

  error: err => console.error(err),
});
\end{lstlisting}
\end{solution}

\question[4]
Which lifecycle hook is designed for cleanup tasks such as unsubscribing from Observables? \emph{Select all that apply.}
\begin{checkboxes}
  \CorrectChoice \lstinline[style=tsinline]|ngOnDestroy|
  \choice \lstinline[style=tsinline]|ngOnInit|
  \choice \lstinline[style=tsinline]|ngOnChanges|
  \choice \lstinline[style=tsinline]|ngAfterViewInit|
\end{checkboxes}
\begin{solution}
  \lstinline[style=tsinline]|ngOnDestroy| runs immediately before Angular destroys a component or directive. Unsubscribe from long-lived Observables, clear timers, and detach manual listeners here to avoid memory leaks. The \lstinline[style=htmlinline]|async| pipe also unsubscribes when its binding is destroyed.
\begin{lstlisting}[style=ts]
export class UserListComponent implements OnInit, OnDestroy {
  private sub?: Subscription;

  ngOnInit() {
    this.sub = this.users$.subscribe();
  }

  ngOnDestroy() {
    this.sub?.unsubscribe();
  }
}
\end{lstlisting}
\end{solution}
