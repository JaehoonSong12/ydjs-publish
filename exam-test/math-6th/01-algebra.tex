\question[5]
Simplify: $3x + 7 - x + 2$.
\begin{choices}
  \CorrectChoice $2x + 9$
  \choice $4x + 9$
  \choice $2x + 5$
  \choice $4x + 5$
\end{choices}

\question[5]
Which expression is equivalent to $2(x+3)$?
\begin{choices}
  \choice $2x+3$
  \CorrectChoice $2x+6$
  \choice $x+5$
  \choice $x+6$
\end{choices}
\begin{solution}
  Distribute: $2\cdot x + 2\cdot 3 = 2x+6$.
\end{solution}

\question[5]
If $3x = 5 + 7$, then $x = \fillin[4]$.
\begin{solution}
  $3x=12$, so $x=4$.
\end{solution}

\question[5]
A rectangle has length $L$ and width $W$. Which expression gives its perimeter?
\begin{choices}
  \choice $LW$
  \choice $L+W$
  \CorrectChoice $2L+2W$
  \choice $L^2+W^2$
\end{choices}
\begin{solution}
  Opposite sides are equal: $L+W+L+W=2L+2W$.
\end{solution}

\question[5]
Solve the followings:
\begin{parts}
  \part
  Compute: $-2 + 7 - 4$.
  \begin{choices}
    \choice $-13$
    \choice $9$
    \CorrectChoice $1$
    \choice $3$
  \end{choices}
  \part
  Simplify: $x + x + x$.
  \begin{choices}
    \choice $x^3$
    \choice $3+x$
    \CorrectChoice $3x$
    \choice $x^2$
  \end{choices}
\end{parts}
\begin{solution}
  (a) $-2+7=5$, then $5-4=1$. \quad (b) $x+x+x=3x$.
\end{solution}

\question[5]
Solve for $x$: $5x + 5 = 20$. Write your answer in the box.
\begin{solutionorbox}[0.75in]
  Subtract $5$: $5x=15$, then $x=3$.
\end{solutionorbox}


\question[5]
Simplify: $4(2a)$.
\begin{choices}
  \choice $6a$
  \choice $8+a$
  \CorrectChoice $8a$
  \choice $24a$
\end{choices}
\begin{solution}
  $4\cdot 2\cdot a = 8a$.
\end{solution}
