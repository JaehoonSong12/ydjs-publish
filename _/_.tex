\section*{Lecture 5: Dirichlet's Theorem and applications of the FTA}

\begin{define}{Theorem 1.21 (Dirichlet's theorem on primes in arithmetic progressions)}[thm:1.21]
Let $a,b\in\mathbb{Z}$ with $b>0$ and $\gcd(a,b)=1$. Then the arithmetic progression
\[
a,\ a+b,\ a+2b,\ a+3b,\ \ldots
\]
contains infinitely many prime numbers.
\end{define}

\begin{define}{Remark (on Theorem \nameref{thm:1.21})}[rem:1.21]
Setting $a=1$ and $b=1$ recovers the elementary statement that there are infinitely many primes.
(We remark that the full proof of Dirichlet's theorem requires analytic tools and is beyond
the scope of this course.)
\end{define}

\begin{define}{Proposition 1.22 (Infinitely many primes congruent to $3$ modulo $4$)}[prop:1.22]
There are infinitely many primes of the form $4n+3$, where $n$ is a nonnegative integer.
\end{define}

\begin{define}{Lemma 1.23 (Product of numbers congruent to $1$ modulo $4$)}[lem:1.23]
Let $a,b\in\mathbb{Z}$. If $a\equiv 1\pmod{4}$ and $b\equiv 1\pmod{4}$, then $ab\equiv 1\pmod{4}$.
\end{define}

\begin{proof}
Write $a=4n_1+1$ and $b=4n_2+1$ with $n_1,n_2\in\mathbb{Z}$. Then
\[
ab=(4n_1+1)(4n_2+1)=16n_1n_2+4n_1+4n_2+1=4(4n_1n_2+n_1+n_2)+1,
\]
so $ab\equiv 1\pmod{4}$. This proves the lemma.
\end{proof}

\begin{proof}[Proof of Proposition \nameref{prop:1.22}]
Suppose, for contradiction, that there are only finitely many primes congruent to $3$ modulo $4$.
List them as
\[
P_1=3,\ P_2,\ P_3,\ \dots,\ P_r,
\]
all distinct and each $P_i\equiv 3\pmod{4}$.

Consider the integer
\[
N := 4P_1P_2\cdots P_r - 1.
\]
Note that $N\equiv -1\equiv 3\pmod{4}$.

Let $q$ be any prime divisor of $N$. We make two observations:

\begin{enumerate}
  \item For each $i=1,\dots,r$ we have $N\equiv -1\pmod{P_i}$, so $P_i\nmid N$. Thus $q$ is not equal to any of the $P_i$.
  \item Since $N\equiv 3\pmod{4}$, it is not possible that every prime divisor of $N$ is congruent to $1$ modulo $4$. 
  Indeed, a product of integers each congruent to $1\pmod{4}$ is itself congruent to $1\pmod{4}$ by Lemma \nameref{lem:1.23}, 
  contradicting $N\equiv 3\pmod{4}$. Hence some prime divisor $q$ of $N$ satisfies $q\equiv 3\pmod{4}$.
\end{enumerate}

Combining the two observations, $q$ is a prime congruent to $3$ modulo $4$ that is not among $P_1,\dots,P_r$, contradicting 
the assumed finiteness of such primes. Therefore there are infinitely many primes congruent to $3$ modulo $4$.
\end{proof}

\begin{define}{Remark (on Proposition \nameref{prop:1.22})}[rem:1.22]
The method above—building an integer with residue $3\pmod{4}$ whose prime divisors cannot all
be from a given finite list—gives a proof in this special case of Dirichlet's theorem. However,
this argument does not extend directly to all arithmetic progressions; for the full theorem one
needs deeper tools (Dirichlet characters and $L$-series).
\end{define}

\section*{Congruences}

\begin{define}{Historical remark (Gauss and congruences)}[rem:gauss]
Up to the beginning of the nineteenth century number theory consisted of many isolated
results. In 1801, Gauss, in his \emph{Disquisitiones Arithmeticae}, introduced the modern theory
of congruences and unified many isolated results in number theory.
\end{define}

\begin{define}{Definition (Congruence modulo $m$)}[def:congruence]
Let $a,b,m\in\mathbb{Z}$ with $m>0$. We say that \emph{$a$ is congruent to $b$ modulo $m$},
and write
\[
a\equiv b\pmod{m},
\]
if $m\mid (a-b)$. The integer $m$ is called the \emph{modulus} of the congruence.
\end{define}

\begin{define}{Remark (notation)}[rem:notation]
The notation $a\not\equiv b\pmod{m}$ means $m\nmid (a-b)$. The modulus is customarily taken
to be positive; we will assume $m>0$ in the sequel.
\end{define}

\begin{define}{Proposition 2.1 (Congruence modulo $m$ is an equivalence relation)}[prop:2.1]
Congruence modulo $m$ is reflexive, symmetric, and transitive on $\mathbb{Z}$; hence it is
an equivalence relation.
\end{define}

\begin{proof}
\emph{Reflexive:} For any $a\in\mathbb{Z}$, $a-a=0$ and $m\mid 0$, so $a\equiv a\pmod{m}$.

\emph{Symmetric:} If $a\equiv b\pmod{m}$ then $m\mid (a-b)$, hence $m\mid -(a-b)=(b-a)$, so
$b\equiv a\pmod{m}$.

\emph{Transitive:} If $a\equiv b\pmod{m}$ and $b\equiv c\pmod{m}$ then $m\mid (a-b)$ and
$m\mid (b-c)$, therefore $m\mid (a-b)+(b-c)=a-c$, so $a\equiv c\pmod{m}$.
\end{proof}

\begin{define}{Consequence 2.2 (Partition into residue classes)}[cor:2.2]
The set $\mathbb{Z}$ is partitioned into equivalence classes under congruence modulo $m$.
\end{define}

\begin{define}{Definition (Notation for residue class)}[def:resclass]
For $x\in\mathbb{Z}$, write $[x]_m$ (or simply $[x]$ when $m$ is understood) for the
equivalence class of $x$ modulo $m$:
\[
[x]_m=\{y\in\mathbb{Z}: y\equiv x\pmod{m}\}.
\]
(Do not confuse $[x]_m$ with the floor function.)
\end{define}

\subsection*{Example: The equivalence classes modulo 4}

\[
\begin{aligned}
[0]_4 &=\{x\in\mathbb{Z}: x\equiv 0\pmod{4}\}=\{4n: n\in\mathbb{Z}\}=\{\dots,-8,-4,0,4,8,\dots\},\\
[1]_4 &=\{x: x=4n+1,\ n\in\mathbb{Z}\}=\{\dots,-7,-3,1,5,9,\dots\},\\
[2]_4 &=\{x: x=4n+2,\ n\in\mathbb{Z}\}=\{\dots,-6,-2,2,6,10,\dots\},\\
[3]_4 &=\{x: x=4n+3,\ n\in\mathbb{Z}\}=\{\dots,-5,-1,3,7,11,\dots\}.
\end{aligned}
\]

Thus $\mathbb{Z}$ is partitioned into four residue classes modulo $4$, and every integer is
congruent to exactly one of $0,1,2,3$ modulo $4$.

\begin{define}{Definition (Complete residue system modulo $m$)}[def:completeRS]
A set $R\subset\mathbb{Z}$ is a \emph{complete residue system modulo $m$} if every integer
is congruent modulo $m$ to exactly one element of $R$.
\end{define}

\begin{define}{Example (complete residue systems)}[ex:completeRS]
The set $\{0,1,2,\dots,m-1\}$ is a complete residue system modulo $m$. Also, for example,
$\{6,-11,19,1988\}$ is a complete residue system modulo $4$, since
$6\equiv 2$, $-11\equiv 1$, $19\equiv 3$, and $1988\equiv 0\pmod{4}$.
\end{define}

\begin{define}{Proposition 2.3 (Standard complete residue system)}[prop:2.3]
The set $\{0,1,2,\dots,m-1\}$ is a complete residue system modulo $m$.
\end{define}

\begin{proof}
Let $a\in\mathbb{Z}$. By the Division Algorithm there exist unique $q,r\in\mathbb{Z}$
with $0\le r<m$ such that $a=mq+r$. Then $a-r=mq$, so $m\mid(a-r)$ and $a\equiv r\pmod{m}$,
with $r\in\{0,1,\dots,m-1\}$. This shows every integer is congruent to at least one element
of $\{0,1,\dots,m-1\}$.

To show uniqueness: suppose $r_1,r_2\in\{0,1,\dots,m-1\}$ and $r_1\equiv r_2\pmod{m}$.
Then $m\mid(r_1-r_2)$, but $-(m-1)\le r_1-r_2\le m-1$. The only multiple of $m$ in this range
is $0$, so $r_1-r_2=0$ and hence $r_1=r_2$. Thus each integer is congruent to exactly one
element of $\{0,1,\dots,m-1\}$.
\end{proof}

\begin{define}{Proposition 2.4 (Arithmetic with congruences)}[prop:2.4]
Let $a,b,c,d\in\mathbb{Z}$ and $m>0$. If $a\equiv b\pmod{m}$ and $c\equiv d\pmod{m}$, then
\begin{enumerate}
  \item $a+c\equiv b+d\pmod{m}$,
  \item $ac\equiv bd\pmod{m}$.
\end{enumerate}
\end{define}

\begin{proof}
From $a\equiv b\pmod{m}$ and $c\equiv d\pmod{m}$ we have $m\mid(a-b)$ and $m\mid(c-d)$.

For (1): $m\mid (a-b)+(c-d)=(a+c)-(b+d)$, hence $a+c\equiv b+d\pmod{m}$.

For (2): Note $m\mid(a-b)$ implies $m\mid c(a-b)$, and $m\mid(c-d)$ implies $m\mid b(c-d)$.
Add these divisibilities to get $m\mid c(a-b)+b(c-d)=ac-bd$, so $ac\equiv bd\pmod{m}$.
\end{proof}
