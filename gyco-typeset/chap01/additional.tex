\section*{Section 5: Prime and Composite Numbers}

\subsection*{Additional}
\textbf{Prime factorization}
\begin{itemize}
  \item the process of breaking down a number into its smallest building blocks--prime factors
\end{itemize}

\noindent\textgreater\ \kr{prime factorization(소인수분해) 말그대로 prime(소수)까지 분해한다는 거임}

\noindent\textgreater\ \kr{제일 기본적인 순서는 소인수분해할 숫자를 보는거임}

\noindent\textgreater\ \kr{그 숫자가 몇의 배수인지 봐야하는데 이건}
\begin{itemize}
  \item \kr{2의 배수: 짝수/일의 자리 숫자가 2, 4, 6, 8, 0인 수}
  \item \kr{3의 배수: 각 자리의 수의 합이 3의 배수}
  \item \kr{끝의 두 자리 수가 00이거나 4의 배수}
  \item \kr{5의 배수: 일의 자리 숫자가 0또는 5}
  \item \kr{7의 배수: 세자리수 abc일 때 ab-c*2가 7의 배수인수 (이건 솔직히 몰라도 됨)}
\end{itemize}

\noindent\textgreater\ \kr{일단 대충 저걸로 판단후 최소 값부터 계속 소인수분해 (ex: 60을 소인수 분해 하면 일단 2로 나눈다 왜냐하면 짝수이기 때문; 2*30; 30도 짝수이기 때문에 2로 다시 나눈다; 2*2*15; 15는 일의 자리가 5이기 때문에 5로 나눈다; 2*2*5*3; 와 다 소수네; 소인수분해 완료)}

\section*{Section 6: Greatest Common Factor \& Least Common Multiple}

\subsection*{Additional}
\textbf{How to find gcf and lcm}
\begin{itemize}
  \item use \textquotesingle\kr{ㄴ}\textquotesingle
\end{itemize}

\noindent\textgreater\ \kr{한국 학교에서 당연히 배웠겠지만 ‘ㄴ’자로 하면됨}

\noindent\textgreater\ \kr{‘ㄴ’자 방법이란 소인수분해랑 똑같은건데 두수 혹은 그 이상의 숫자들이 공유하는 약수로 다 나눈건데 이거는 꼭 소수로 안해도 됨}

\noindent\textgreater\ \kr{12와 18의 GCF(최대공배수)를 찾아봅시다}

\[
\begin{array}{r|cc}
6 & 12 & 18 \\
\hline
  & 2 & 3
\end{array}
\]

\noindent\textgreater\ \kr{위에처럼 마지막 숫자가 소수가 되면 끝}

\noindent\textgreater\ \kr{최대공약수의 경우 ‘ㄴ’자 옆에 있는 숫자들의 곱; 지금 같은 경우는 6}

\noindent\textgreater\ \kr{12와 18의 LCM(최소공배수)를 찾아봅니다}

\[
\begin{array}{r|cc}
6 & 12 & 18 \\
\hline
  & 2 & 3
\end{array}
\]

\noindent\textgreater\ \kr{최소공배수의 경우 ‘ㄴ’자 옆밑에 있는 모든 숫자들의 솝; 지금 같은 경우는 6*2*3 = 36}

\noindent\textgreater\ \kr{와 쉽죠}
