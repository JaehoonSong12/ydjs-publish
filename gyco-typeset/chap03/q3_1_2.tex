\subsection{Long Question (Answer)}
A class has 12 boys and 8 girls.
\begin{enumerate}[label=\alph*)]
  \item Write the ratio of boys to girls in simplest form.
  \item Write the ratio of girls to total students in simplest form.
\end{enumerate}

\textbf{Answer:}
\begin{enumerate}[label=\alph*)]
  \item Boys to girls = \textbf{12:8}, simplified to \textbf{3:2}.
  \item Girls to total = $8:20$, simplified to \textbf{2:5}.
\end{enumerate}
